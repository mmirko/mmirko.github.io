+func (arch *Arch) OnlyOne(curOp string, ops []string) bool {
+	sorted := make([]string, len(ops))
+	copy(sorted, ops)
+	sort.Strings(sorted)
+	for _, op := range sorted {
+		if op == curOp {
+			return true
+		} else {
+			for _, op2 := range arch.Conproc.Op {
+				if op2.Op_get_name() == op {
+					return false
+				}
+			}
+		}
+	}
+	return false
