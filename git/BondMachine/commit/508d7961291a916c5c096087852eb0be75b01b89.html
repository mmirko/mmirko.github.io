+func (bi *BasmInstance) bodyMacros(body *bmline.BasmBody) error {
+	for i := 0; i < len(body.Lines); i++ {
+		line := body.Lines[i]
+		op := line.Operation.GetValue()
+		if macro, ok := bi.macros[op]; ok {
+			if bi.debug {
+				fmt.Println(yellow("\t\t\tMacro: ") + op)
+			}
+			expArgs := macro.macroArgs
