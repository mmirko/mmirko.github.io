+func (b *BCOFData) Dump() string {
+	regBytes := (b.Rsize + 7) / 8
+	result := fmt.Sprintf("Rsize: %d\nId: %d\nSignature: %s\nPayload:", b.Rsize, b.Id, b.Signature)
+	for i, ch := range b.Payload {
+		if i%int(regBytes) == 0 {
+			result += "\n\t0x"
+		}
+		result += fmt.Sprintf("%02x", ch)
+	}
+	return result
+}
