      2 ; r0 pointer to the buffer beginning (es: mov r0, ram:buff)
      3 ; r2 lenght

      5         rsets8	r2, 0
      6 	rsets8	r3, 13
      7 getchar:

     12 	mov	ram:[r0], r1

     14         inc     r2
     15         j getchar
     16 end:
     17 	mov	r0,r2

     19 loop:
     20 	mov	vtm0:[r0], r3
