      1 ; This implements a simple sleep function
      2 ; in  r0 pointer to the outer loop counter
      3 ; in  r1 pointer to the inner loop counter
      4 ; in  r2 pointer to the temp loop counter
      5 
      6 %fragment sleep default_outer:r0 default_inner:r1 default_temp:r2
      7 outerloop:
      8 jz	{{ .Params.outer }}, endouter	
      9 	mov	{{ .Params.temp }} , {{ .Params.inner }}
     10 	dec 	{{ .Params.outer }}
     11 innerloop:
     12 	jz	{{ .Params.temp }}, endinner
     13 	dec	{{ .Params.temp }}
     14 	jmp	innerloop
     15 endinner:
     16 	jmp	outerloop
     17 endouter:
     18 	nop
     19 %endfragment
