newline.basm (2098B)

      1 ; Ths function go to a new line
      2 ; r0: pointer to the memory start (0 for vtm0) 
      3 ; r2: x position within the memory (video textual memory vtm0 or similar)
      4 %fragment newline default_out:vtm0 default_height:16 default_width:16
      5         mov     r1, {{ .Params.height }}
      6         dec     r1
      7         mov     r3, {{ .Params.width }}
      8         mult    r1, r3 ; this is the start cell of the last line
      9         mov     r3, r2
     10         sub     r3, r0 ; this is the current cell (offset from the start)
     11         cmprlt  r3, r1 ; if the current cell is less than the start of the last line don't scroll
     12         jcmp    dontscroll ; at this point r3 is the offset from the start of the memory, r1 is the current cell (no offset)
     13 scroll:
     14         mov     r4, r0
     15         mov     r5, {{ .Params.width }}
     16         add     r4, r5 ; This is the source
     17         mov     r5, r0 ; This is the destination
     18 scrollloop:
     19         mov     r6, {{ .Params.out }}:[r4]
     20         mov     {{ .Params.out }}:[r5], r6
     21         inc     r5
     22         inc     r4
     23         cmpr    r4, r2
     24         jcmp    scrollloopdone
     25         j       scrollloop
     26 scrollloopdone:
     27         mov     r5, {{ .Params.width }}
     28         sub     r2, r5
     29         sub     r3, r5
     30 dontscroll:
     31         ; find out the next line position
     32         mov     r1, 0
     33         mov     r4, {{ .Params.width }}
     34 findnextline:
     35         inc     r1
     36         mov     r5, r1
     37         mult    r5, r4
     38         cmprlt  r5, r3
     39         jcmp    findnextline
     40         mov     r1, 0x00
     41 fillblank:
     42         cmpr    r5, r3
     43         jcmp    fillblank2prep
     44         mov     r6, r3
     45         add     r6, r0
     46         mov     {{ .Params.out }}:[r6], r1
     47         inc     r3
     48         j       fillblank
     49 fillblank2prep:
     50         mov     r5, {{ .Params.width }}
     51         mov     r4, {{ .Params.height }}
     52         mult    r4, r5
     53         dec     r4
     54         mov     r5, r3
     55 fillblank2:
     56         cmpr    r5, r4
     57         jcmp    end
     58         mov     r7, r5
     59         add     r7, r0
     60         mov     {{ .Params.out }}:[r7], r1
     61         inc     r5
     62         j       fillblank2
     63 end:
     64         inc     r6
     65         mov     r2, r6
     66         mov     r4, r3
     67 %endfragment
