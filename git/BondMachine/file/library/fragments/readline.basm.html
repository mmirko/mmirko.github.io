readline.basm (673B)

      1 ; This reads a string from the keyboard the is supposed to be connected to kbd
      2 ; r0 pointer to the buffer beginning (es: mov r0, ram:buff)
      3 ; r2: pointer to the start of the memory where the string will be stored, it will be updated and it will point to the end
      4 %fragment readline default_out:vtm0 default_kbd:i0 default_feedout:vtm0 default_feedreg:r4
      5 	rsets8	r3, 13
      6 getchar:
      7 	i2rw	r1, {{ .Params.kbd }}
      8         mov     {{ .Params.out }}:[r2], r1
      9 	mov	ram:[r0], r1
     10 	mov	{{ .Params.feedout }}:[{{ .Params.feedreg }}], r1
     11 	inc	{{ .Params.feedreg }}
     12         inc     r0
     13         inc     r2
     14 	cmpr	r1, r3
     15 	jcmp	end
     16         j getchar
     17 end:
     18 	mov	r1, 0
     19 	mov	ram:[r0], r1
     20 %endfragment
