     48 func fixedPointImport(re *regexp.Regexp, input string) (*BMNumber, error) {
     49 	number := re.ReplaceAllString(input, "${number}")
     50 	ss := re.ReplaceAllString(input, "${s}")
     51 	fs := re.ReplaceAllString(input, "${f}")
     52 	s, _ := strconv.Atoi(ss)
     53 	f, _ := strconv.Atoi(fs)
     54 
     55 	if s < 1 || s > 32 {
     56 		return nil, errors.New("invalid s value for fixed point")
     57 	}
     58 
     59 	EventuallyCreateType("fps"+ss+"f"+fs, nil)
     60 
     61 	if numberNum, err := strconv.ParseFloat(number, 64); err != nil {
     62 		return nil, errors.New("invalid number for fixed point")
     63 	} else {
     64 		scale := float64(int(1) << uint(f))
     65 		fpNumber := int64(numberNum * scale)
     66 
     67 		// Create a temporary buffer to write the number to
     68 		buf := new(bytes.Buffer)
     69 		binary.Write(buf, binary.LittleEndian, fpNumber)
     70 
     71 		// Compute how many bytes we need to copy according to the number of bits (s)
     72 		toCopy := (s-1)/8 + 1
     73 
     74 		newNumber := BMNumber{}
     75 		newNumber.nType = FixedPoint{FixedPointName: "fps" + ss + "f" + fs, s: s, f: f}
     76 		newNumber.number = make([]byte, toCopy)
     77 		copy(newNumber.number, buf.Bytes()[0:toCopy])
     78 		newNumber.bits = s
     79 
     80 		// Use a mask to clear the eventually unused bits on the last byte
     81 		mask := byte(0xFF >> uint(8-(s-1)%8-1))
     82 		newNumber.number[toCopy-1] = newNumber.number[toCopy-1] & mask
     83 
     84 		return &newNumber, nil
     85 	}

     88 func (d FixedPoint) ExportString(n *BMNumber) (string, error) {
     89 	s := d.s
     90 	f := d.f

     95 	var isNegative bool
     96 	if n.number[len(n.number)-1]&(uint8(128)>>uint(8-(s-1)%8-1)) != 0 {
     97 		isNegative = true
     98 	}
     99 
    100 	copied := make([]byte, len(n.number))
    101 	copy(copied, n.number)
    102 
    103 	if isNegative {
    104 		// Use a mask to clear the eventually unused bits on the last byte
    105 		lastByte := (s-1)/8 + 1
    106 		mask := byte(0xFF >> uint(8-(s-1)%8-1))
    107 		copied[lastByte-1] = copied[lastByte-1] | ^mask
    108 
    109 		for i := len(copied); i < 8; i++ {
    110 			copied = append(copied, 0xFF)
    111 		}
    112 	} else {
    113 		for i := len(copied); i < 8; i++ {
    114 			copied = append(copied, 0x00)
    115 		}
    116 	}
    117 	var number int64
    118 	buf := bytes.NewReader(copied)
    119 
    120 	if err := binary.Read(buf, binary.LittleEndian, &number); err != nil {
    121 		return "", err
    122 	}
    123 
    124 	scale := float64(int(1) << uint(f))
    125 
    126 	numberF := float64(number) / scale
