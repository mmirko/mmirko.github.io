     48 func linearQuantizerImport(re *regexp.Regexp, input string) (*BMNumber, error) {
     49 	number := re.ReplaceAllString(input, "${number}")
     50 	ss := re.ReplaceAllString(input, "${s}")
     51 	ts := re.ReplaceAllString(input, "${t}")
     52 	s, _ := strconv.Atoi(ss)
     53 	t, _ := strconv.Atoi(ts)
     54 
     55 	if s < 1 || s > 32 {
     56 		return nil, errors.New("invalid s value for linear quantizer")
     57 	}
     58 
     59 	// Get the linear quantizer ranges struct
     60 	var lqRanges *map[int]LinearDataRange
     61 	for _, t := range AllDynamicalTypes {
     62 		if t.GetName() == "dyn_linear_quantizer" {
     63 			lqRanges = t.(DynLinearQuantizer).Ranges
     64 		}
     65 	}
     66 
     67 	if lqRanges != nil {
     68 		if _, ok := (*lqRanges)[t]; !ok {
     69 			return nil, errors.New("invalid type value for linear quantizer")
     70 		}
     71 	} else {
     72 		return nil, errors.New("linear quantizer ranges not found")
     73 	}
     74 
     75 	EventuallyCreateType("lqs"+ss+"t"+ts, nil)
     76 
     77 	bandNum := float64(int(1) << uint(s-1))
     78 	bandSize := ((*lqRanges)[t].Max) / bandNum
     79 
     80 	if numberNum, err := strconv.ParseFloat(number, 64); err != nil {
     81 		return nil, errors.New("invalid number for linear quantizer")
     82 	} else {
     83 		band := int64(numberNum / bandSize)
     84 		if band >= int64(bandNum) || band <= -int64(bandNum) {
     85 			return nil, errors.New("number out of range for linear quantizer")
     86 		}
     87 
     88 		// Create a temporary buffer to write the number to
     89 		buf := new(bytes.Buffer)
     90 		binary.Write(buf, binary.LittleEndian, band)
     91 
     92 		// Compute how many bytes we need to copy according to the number of bits (s)
     93 		toCopy := (s-1)/8 + 1
     94 
     95 		newNumber := BMNumber{}
     96 		newNumber.nType = LinearQuantizer{linearQuantizerName: "lqs" + ss + "t" + ts, s: s, t: t}
     97 		newNumber.number = make([]byte, toCopy)
     98 		copy(newNumber.number, buf.Bytes()[0:toCopy])
     99 		newNumber.bits = s
    100 
    101 		// Use a mask to clear the eventually unused bits on the last byte
    102 		mask := byte(0xFF >> uint(8-(s-1)%8-1))
    103 		newNumber.number[toCopy-1] = newNumber.number[toCopy-1] & mask
    104 
    105 		return &newNumber, nil
    106 	}

    109 func (d LinearQuantizer) ExportString(n *BMNumber) (string, error) {
    110 	s := d.s
    111 	t := d.t
    112 	ss := strconv.Itoa(s)
    113 	ts := strconv.Itoa(t)
    114 
    115 	// Find out if the number is negative
    116 	var isNegative bool
    117 	if n.number[len(n.number)-1]&(uint8(128)>>uint(8-(s-1)%8-1)) != 0 {
    118 		isNegative = true
    119 	}
    120 
    121 	copied := make([]byte, len(n.number))
    122 	copy(copied, n.number)
    123 
    124 	if isNegative {
    125 		// Use a mask to clear the eventually unused bits on the last byte
    126 		lastByte := (s-1)/8 + 1
    127 		mask := byte(0xFF >> uint(8-(s-1)%8-1))
    128 		copied[lastByte-1] = copied[lastByte-1] | ^mask
    129 
    130 		for i := len(copied); i < 8; i++ {
    131 			copied = append(copied, 0xFF)
    132 		}
    133 	} else {
    134 		for i := len(copied); i < 8; i++ {
    135 			copied = append(copied, 0x00)
    136 		}
    137 	}
    138 
    139 	var number int64
    140 	buf := bytes.NewReader(copied)
    141 	if err := binary.Read(buf, binary.LittleEndian, &number); err != nil {
    142 		return "", err
    143 	}
    144 
    145 	// Get the linear quantizer ranges struct
    146 	var lqRanges *map[int]LinearDataRange
    147 	for _, t := range AllDynamicalTypes {
    148 		if t.GetName() == "dyn_linear_quantizer" {
    149 			lqRanges = t.(DynLinearQuantizer).Ranges
    150 		}
    151 	}
    152 
    153 	bandNum := float64(int(1) << uint(s-1))
    154 	bandSize := (*lqRanges)[t].Max / bandNum
    155 
    156 	numberF := float64(number) * bandSize
    157 
    158 	result := "0lq<" + ss + "." + ts + ">" + strconv.FormatFloat(numberF, 'f', -1, 64)
    159 	return result, nil
