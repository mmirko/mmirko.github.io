     37 func (s *BMNumberServer) ExecRequest(w http.ResponseWriter, r *http.Request) {
     38 	reqBody, err := ioutil.ReadAll(r.Body)
     39 	if err != nil {
     40 		http.Error(w, "Failed to read request body", http.StatusBadRequest)
     41 		return
     42 	}
     43 
     44 	var bmNumbersRequest BMNumbersRequest
     45 	err = json.Unmarshal(reqBody, &bmNumbersRequest)
     46 	if err != nil {
     47 		http.Error(w, "Failed to unmarshal request body", http.StatusBadRequest)
     48 		return
     49 	}
     50 
     51 	var newType BMNumberType
     52 
     53 	if v := GetType(bmNumbersRequest.ReqType); v == nil {
     54 		http.Error(w, "Unknown type", http.StatusInternalServerError)
     55 		return
     56 	} else {
     57 		newType = v
     58 	}
     59 
     60 	var results []string
     61 
     62 	for i := 0; i < len(bmNumbersRequest.Numbers); i++ {
     63 		if output, err := ImportString(bmNumbersRequest.Numbers[i]); err != nil {
     64 			http.Error(w, fmt.Sprintf("Error: %v", err), http.StatusInternalServerError)
     65 			return
     66 		} else {
     67 
     68 			if bmNumbersRequest.Action == "convert" {
     69 				if err := newType.Convert(output); err != nil {
     70 					http.Error(w, fmt.Sprintf("Error: %v", err), http.StatusInternalServerError)
     71 					return
     72 				}
     73 			} else if bmNumbersRequest.Action == "cast" {
     74 				if err := CastType(output, newType); err != nil {
     75 					http.Error(w, fmt.Sprintf("Error: %v", err), http.StatusInternalServerError)
     76 					return
     77 				}
     78 			}
     79 
     80 			switch bmNumbersRequest.DumpMode {
     81 			case "native":
     82 				if value, err := output.ExportString(s.conf); err != nil {
     83 					http.Error(w, fmt.Sprintf("Error: %v", err), http.StatusInternalServerError)
     84 					return
     85 				} else {
     86 					results = append(results, value) // add a string to the slice
     87 				}
     88 			case "bin":
     89 				if value, err := output.ExportBinary(false); err != nil {
     90 					http.Error(w, fmt.Sprintf("Error: %v", err), http.StatusInternalServerError)
     91 					return
     92 				} else {
     93 					results = append(results, value) // add a string to the slice
     94 				}
     95 			case "unsigned":
     96 				if value, err := output.ExportUint64(); err != nil {
     97 					http.Error(w, fmt.Sprintf("Error: %v", err), http.StatusInternalServerError)
     98 					return
     99 				} else {
    100 					results = append(results, strconv.FormatUint(value, 10)) // add a string to the slice
    101 				}
    102 			default:
    103 				http.Error(w, "Unknown visualization format", http.StatusInternalServerError)
    104 				return
    105 			}
    106 		}
    107 	}
    108 
    109 	w.WriteHeader(http.StatusOK) // set HTTP status code to 200
    110 	json.NewEncoder(w).Encode(BMNumbersResponse{Numbers: results})
    111 }
