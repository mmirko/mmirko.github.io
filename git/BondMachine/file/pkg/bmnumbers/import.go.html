     27 func ImportBytes(input []byte, bits int) (*BMNumber, error) {

     29 	result.nType = Unsigned{}
     30 	result.bits = bits

     37 func ImportUint(input interface{}, optionalBits int) (*BMNumber, error) {

     50 	case uint32:
     51 		result.bits = 32

     54 		result.number[2] = uint8(input.(uint32) >> 16 & 0xFF)
     55 		result.number[1] = uint8(input.(uint32) >> 8 & 0xFF)
     56 		result.number[0] = uint8(input.(uint32) & 0xFF)
     57 	case uint64:
     58 		result.bits = 64
     59 		result.number = make([]byte, 8)
     60 		result.number[7] = uint8(input.(uint64) >> 56 & 0xFF)
     61 		result.number[6] = uint8(input.(uint64) >> 48 & 0xFF)
     62 		result.number[5] = uint8(input.(uint64) >> 40 & 0xFF)
     63 		result.number[4] = uint8(input.(uint64) >> 32 & 0xFF)
     64 		result.number[3] = uint8(input.(uint64) >> 24 & 0xFF)
     65 		result.number[2] = uint8(input.(uint64) >> 16 & 0xFF)
     66 		result.number[1] = uint8(input.(uint64) >> 8 & 0xFF)

     78 func LoadLinearDataRangesFromFile(filename string) error {
     79 
     80 	// Get the linear quantizer ranges struct
     81 	var lqRanges *map[int]LinearDataRange
     82 	for _, t := range AllDynamicalTypes {
     83 		if t.GetName() == "dyn_linear_quantizer" {
     84 			lqRanges = t.(DynLinearQuantizer).Ranges
     85 		}
     86 	}
     87 
     88 	splitted := strings.Split(filename, ",")
     89 	if len(splitted)%2 != 0 {
     90 		return errors.New("Error: Invalid linear data range files")
     91 	}
     92 
     93 	// Load a file for each index
     94 	for i := 0; i < len(splitted); i += 2 {
     95 		index, err := strconv.Atoi(splitted[i])
     96 		if err != nil {
     97 			return err
     98 		}
     99 
    100 		if index == 0 {
    101 			return errors.New("index cannot be 0 (reserved)")
    102 		}
    103 
    104 		// Check if the index is already present
    105 		if _, ok := (*lqRanges)[index]; ok {
    106 			return errors.New("index already present")
    107 		}
    108 
    109 		filename := splitted[i+1]
    110 
    111 		// Read all the lines of the file
    112 		f, err := os.Open(filename)
    113 		if err != nil {
    114 			return err
    115 		}
    116 
    117 		// Read all the lines of the file
    118 		var max float64
    119 		scanner := bufio.NewScanner(f)
    120 		first := true
    121 		for scanner.Scan() {
    122 			line := scanner.Text()
    123 
    124 			// Parse the max values
    125 			if val, err := strconv.ParseFloat(line, 64); err == nil {
    126 				val = math.Abs(val)
    127 				if first {
    128 					max = val
    129 					first = false
    130 				}
    131 
    132 				if val > max {
    133 					max = val
    134 				}
    135 			}
    136 		}
    137 
    138 		// Add the range to the map
    139 		(*lqRanges)[index] = LinearDataRange{Max: max}
    140 		f.Close()
    141 	}
    142 	return nil
    143 }
