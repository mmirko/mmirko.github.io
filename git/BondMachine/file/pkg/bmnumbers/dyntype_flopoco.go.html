     49 func floPoCoImport(re *regexp.Regexp, input string) (*BMNumber, error) {

     67 		return nil, err
     68 	}
     69 	defer os.RemoveAll(dir)
     70 
     71 	cmd := exec.Command(runCommand[0], runCommand[1:]...)
     72 	cmd.Dir = dir
     73 	if out, err := cmd.CombinedOutput(); err != nil {
     74 		return nil, err
     75 	} else {
     76 		resultReport := string(out)
     77 
     78 		// Parse every line of the report
     79 		for _, line := range strings.Split(resultReport, "\n") {
     80 			re = regexp.MustCompile(`[0-1]+`)

     82 				binNum := line
     83 				var nextByte string
     84 				newNumber := BMNumber{}
     85 				newNumber.number = make([]byte, (len(binNum)-1)/8+1)
     86 				newNumber.bits = len(binNum)
     87 				newNumber.nType = FloPoCo{floPoCoName: "flpe" + es + "f" + fs, e: e, f: f, instructions: i}
     88 
     89 				for i := 0; ; i++ {
     90 
     91 					if len(binNum) > 8 {
     92 						nextByte = binNum[len(binNum)-8:]
     93 						binNum = binNum[:len(binNum)-8]
     94 					} else if len(binNum) > 0 {
     95 						nextByte = binNum
     96 						binNum = ""
     97 					} else {
     98 						break
     99 					}
    100 
    101 					if val, err := strconv.ParseUint(nextByte, 2, 8); err == nil {
    102 						decoded := byte(val)

    105 						return nil, errors.New("invalid binary number" + input)
