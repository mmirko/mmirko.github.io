      9 func (rg *ReqRoot) decodeNode(node string) (*bmReqObj, error) {
     10 	if node == "/" {
     11 		return rg.obj, nil
     12 	}
     13 
     14 	seqlist := strings.Split(node, "/")
     15 	var currNode *bmReqObj = rg.obj
     16 	var lName string
     17 	var lReq string
     18 	for _, leaf := range seqlist {
     19 		currMap := currNode.getMap()
     20 		if leaf != "" {
     21 			decLeaf := strings.Split(leaf, ":")
     22 			if len(decLeaf) != 2 {
     23 				return nil, errors.New("malformed node")
     24 			}
     25 			lReq = decLeaf[0]
     26 			lName = decLeaf[1]
     27 
     28 			if reqset, ok := currMap[lReq]; ok {
     29 				if reqset.supportSub() {
     30 					if newNode, err := reqset.getSub(lName); err == nil {
     31 						if newNode != nil {
     32 							currNode = newNode
     33 						} else {
     34 							return nil, errors.New("node empty")
     35 						}
     36 					} else {
     37 						return nil, errors.New("node not exists")
     38 					}
     39 				} else {
     40 					return nil, errors.New("requirement Set does not support sub requirements")
     41 				}
     42 			} else {
     43 				return nil, errors.New("unknown Requirement Set")
     44 			}
     45 		}
     46 	}
     47 
     48 	return currNode, nil
     49 }

     51 func (rg *ReqRoot) run() {
     52 	ctx := rg.ctx
     53 	for {
     54 		select {
     55 		case <-ctx.Done():
     56 			return
     57 		case req := <-rg.ask:
     58 			resp := ReqResponse{Value: "", Error: nil}
     59 			switch req.Op {
     60 			case OpAdd:
     61 				if node, err := rg.decodeNode(req.Node); err == nil {
     62 					currMap := node.getMap()
     63 					if reqSet, ok := currMap[req.Name]; ok {
     64 						if reqSet.getType() != req.T {
     65 							resp.Error = errors.New("insert failed: Mismatch types")
     66 						} else {
     67 							if err := reqSet.insertReq(req.Value); err != nil {
     68 								resp.Error = errors.New("Insert failed: " + fmt.Sprint(err))
     69 							}
     70 						}
     71 					} else {
     72 						switch req.T {
     73 						case ObjectSet:
     74 							newReq := new(objectSet)
     75 							newReq.init()
     76 							newReq.setName(req.Name)
     77 							newReq.setType(ObjectSet)
     78 							if err := newReq.insertReq(req.Value); err == nil {
     79 								node.getMap()[req.Name] = newReq
     80 							} else {
     81 								resp.Error = errors.New("insert failed: " + fmt.Sprint(err))
     82 							}
     83 						case ObjectMax:
     84 							newReq := new(objectMax)
     85 							newReq.init()
     86 							newReq.setName(req.Name)
     87 							newReq.setType(ObjectMax)
     88 							if err := newReq.insertReq(req.Value); err == nil {
     89 								node.getMap()[req.Name] = newReq
     90 							} else {
     91 								resp.Error = errors.New("insert failed: " + fmt.Sprint(err))
     92 							}
     93 						default:
     94 							resp.Error = errors.New("unknown Type")
     95 						}
     96 					}
     97 				} else {
     98 					resp.Error = errors.New("node decoding failed: " + fmt.Sprint(err))
     99 				}
    100 			case OpCheck:
    101 				if node, err := rg.decodeNode(req.Node); err == nil {
    102 					currMap := node.getMap()
    103 					if reqSet, ok := currMap[req.Name]; ok {
    104 						if reqSet.getType() != req.T {
    105 							resp.Error = errors.New("check failed: Mismatch types")
    106 						} else {
    107 							if present, err := reqSet.checkReq(req.Value); err != nil {
    108 								resp.Error = errors.New("check failed: " + fmt.Sprint(err))
    109 							} else {
    110 								resp.Value = present
    111 							}
    112 						}
    113 					} else {
    114 					}
    115 				} else {
    116 					resp.Error = errors.New("node decoding failed: " + fmt.Sprint(err))
    117 				}
    118 			case OpGet:
    119 				if node, err := rg.decodeNode(req.Node); err == nil {
    120 					if reqSet, ok := node.getMap()[req.Name]; ok {
    121 						resp.Value = reqSet.getReqs()
    122 					} else {
    123 						resp.Error = errors.New("set of requirements not found \"" + req.Name + "\"")
    124 					}
    125 				} else {
    126 					resp.Error = errors.New("Node decoding failed: " + fmt.Sprint(err))
    127 				}
    128 			case OpDump:
    129 				if result, err := rg.recursiveDump(req.Node); err == nil {
    130 					resp.Value = result
    131 				} else {
    132 					resp.Error = err
    133 				}
    134 			case OpClone:
    135 				go rg.Clone(req.Node, req.Name)
    136 				resp.Value = "Clone started"
    137 			default:
    138 				resp.Error = errors.New("unknown Operation")
    139 			}
    140 
    141 			rg.answer <- resp
    142 		}
    143 	}
    144 }

    146 func (rg *ReqRoot) recursiveDump(node string) (string, error) {
    147 	if n, err := rg.decodeNode(node); err == nil {
    148 		result := ""
    149 		for name, set := range n.bmReqMap {
    150 			result += node[1:] + "/" + name + "[" + set.getReqs() + "]\n"
    151 			if set.supportSub() {
    152 				subs := set.listSub()
    153 				for _, sub := range subs {
    154 					sr, _ := rg.recursiveDump(node + "/" + name + ":" + sub)
    155 					list := strings.Split(sr, "\n")
    156 					for _, l := range list {
    157 						if l != "" {
    158 							result += l + "\n"
    159 						}
    160 					}
    161 				}
    162 			}
    163 		}
    164 		return result, nil
    165 	} else {
    166 		return "", errors.New("node decoding failed: " + fmt.Sprint(err))
    167 	}
    168 }

    170 func (rg *ReqRoot) Export(r *ExportedReqs, node string) error {
    171 	if n, err := rg.decodeNode(node); err == nil {
    172 		for name, set := range n.bmReqMap {
    173 			*r = append(*r, ExportedReq{Node: node, Type: set.getType(), Req: set.getReqs(), Name: name})
    174 			if set.supportSub() {
    175 				subs := set.listSub()
    176 				for _, sub := range subs {
    177 					if node == "/" {
    178 						node = node[1:]
    179 					}
    180 					rg.Export(r, node+"/"+name+":"+sub)
    181 				}
    182 			}
    183 		}
    184 		return nil
    185 	} else {
    186 		return errors.New("node decoding failed: " + fmt.Sprint(err))
    187 	}

    190 func Import(r *ExportedReqs) (*ReqRoot, error) {
    191 	rg := NewReqRoot()
    192 	for _, req := range *r {
    193 		switch req.Type {
    194 		case ObjectSet:
    195 			n := new(objectSet)
    196 			if err := n.importReqs(rg, req.Node, req.Name, req.Req); err != nil {
    197 				return nil, err
    198 			}
    199 		case ObjectMax:
    200 			n := new(objectMax)
    201 			if err := n.importReqs(rg, req.Node, req.Name, req.Req); err != nil {
    202 				return nil, err
    203 			}
    204 		}
    205 	}
    206 	return rg, nil
    207 }

    209 func (rg *ReqRoot) Clone(node string, nodeDes string) error {
    210 	if n, err := rg.decodeNode(node); err == nil {
    211 		for name, set := range n.bmReqMap {
    212 			if node == "/" {
    213 				node = node[1:]
    214 			}
    215 			switch set.getType() {
    216 			case ObjectSet:
    217 				n := new(objectSet)
    218 				if err := n.importReqs(rg, nodeDes, name, set.getReqs()); err != nil {
    219 					return err
    220 				}
    221 			case ObjectMax:
    222 				n := new(objectMax)
    223 				if err := n.importReqs(rg, nodeDes, name, set.getReqs()); err != nil {
    224 					return err
    225 				}
    226 			}
    227 			if set.supportSub() {
    228 				subs := set.listSub()
    229 				for _, sub := range subs {
    230 					if err := rg.Clone(node+"/"+name+":"+sub, nodeDes+"/"+name+":"+sub); err != nil {
    231 						return err
    232 					}
    233 				}
    234 			}
    235 		}
    236 		return nil
    237 	} else {
    238 		return errors.New("node decoding failed: " + fmt.Sprint(err))
    239 	}
