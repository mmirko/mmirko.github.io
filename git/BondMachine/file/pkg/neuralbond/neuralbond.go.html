     68 func (n *TrainedNet) Init(config *Config) error {
     69 	n.NetConfig = config
     70 	n.Neurons = make(map[string]*Neuron)
     71 
     72 	// List nb files in the neuron library path and load them
     73 	neuronFiles, err := os.ReadDir(n.NetConfig.NeuronLibPath)
     74 	if err != nil {
     75 		return err
     76 	}
     77 	for _, f := range neuronFiles {
     78 		if len(f.Name()) > 3 && f.Name()[len(f.Name())-3:] == ".nb" {
     79 			neuronFile, err := os.ReadFile(n.NetConfig.NeuronLibPath + "/" + f.Name())
     80 			if err != nil {
     81 				return err
     82 			}
     83 			neuron := new(Neuron)
     84 			if err := json.Unmarshal(neuronFile, neuron); err != nil {
     85 				return err
     86 			}
     87 			n.Neurons[f.Name()[0:len(f.Name())-3]] = neuron
     88 		}
     89 	}
     90 
     91 	return nil
     92 }

     94 func (n *TrainedNet) Normalize() {
     95 	for w, weight := range n.Weights {
     96 		downL, downP := weight.Layer-1, weight.PosPrevLayer
     97 		upL, upP := weight.Layer, weight.PosCurrLayer
     98 
     99 		for i, node := range n.Nodes {
    100 			if node.Layer == downL && node.Pos == downP {
    101 				n.Nodes[i].Outputs++
    102 			}
    103 			if node.Layer == upL && node.Pos == upP {
    104 				n.Nodes[i].Inputs++
    105 			}
    106 		}
    107 
    108 		sameDown := make([]int, 0)
    109 		sameUp := make([]int, 0)
    110 
    111 		for _, chWeight := range n.Weights {
    112 			if chWeight.Layer-1 == downL && chWeight.PosPrevLayer == downP {
    113 				sameDown = append(sameDown, chWeight.PosCurrLayer)
    114 			}
    115 			if chWeight.Layer == upL && chWeight.PosCurrLayer == upP {
    116 				sameUp = append(sameUp, chWeight.PosPrevLayer)
    117 			}
    118 		}
    119 
    120 		// Sort the sameDown and sameUp arrays
    121 		sort.Ints(sameDown)
    122 		sort.Ints(sameUp)
    123 
    124 		// fmt.Println(w, sameDown, sameUp)
    125 
    126 		for i, v := range sameDown {
    127 			if weight.PosCurrLayer == v {
    128 				n.Weights[w].RelPosDown = i
    129 				break
    130 			}
    131 		}
    132 
    133 		for i, v := range sameUp {
    134 			if weight.PosPrevLayer == v {
    135 				n.Weights[w].RelPosUp = i
    136 				break
    137 			}
    138 		}
    139 
    140 	}
    141 }

    143 func (n *TrainedNet) WriteBasm() (string, error) {
    144 	c := n.NetConfig
    145 	regSize := n.RegisterSize
    146 	result := fmt.Sprintf("%%meta bmdef     global registersize:%d\n", regSize)
    147 	switch n.IOMode {
    148 	case ASYNC:
    149 		result += fmt.Sprintf("%%meta bmdef     global iomode:async\n")
    150 	case SYNC:
    151 		result += fmt.Sprintf("%%meta bmdef     global iomode:sync\n")
    152 	}
    153 	switch n.OperatingMode {
    154 	case ROMCODE:
    155 		for _, node := range n.Nodes {
    156 			if node.Type == "input" {
    157 				result += fmt.Sprintf("%%meta cpdef node_0_%d romcode:terminal\n", node.Pos)
    158 				c.List[fmt.Sprintf("node_0_%d", node.Pos)] = "input"
    159 				result += fmt.Sprintf("%%meta iodef input_%d type:io\n", node.Pos)
    160 				result += fmt.Sprintf("%%meta ioatt input_%d cp:node_0_%d, type:input, index:0\n", node.Pos, node.Pos)
    161 				result += fmt.Sprintf("%%meta ioatt input_%d cp:bm, type:input, index:%d\n", node.Pos, node.Pos)
    162 			} else if node.Type == "output" {
    163 				result += fmt.Sprintf("%%meta cpdef node_%d_%d romcode:terminal\n", node.Layer, node.Pos)
    164 				c.List[fmt.Sprintf("node_%d_%d", node.Layer, node.Pos)] = "output"
    165 				result += fmt.Sprintf("%%meta iodef output_%d type:io\n", node.Pos)
    166 				result += fmt.Sprintf("%%meta ioatt output_%d cp:node_%d_%d, type:output, index:0\n", node.Pos, node.Layer, node.Pos)
    167 				result += fmt.Sprintf("%%meta ioatt output_%d cp:bm, type:output, index:%d\n", node.Pos, node.Pos)
    168 			} else {
    169 				if neuron, ok := n.Neurons[node.Type]; ok {
    170 					result += fmt.Sprintf("%%meta cpdef node_%d_%d romcode:%s", node.Layer, node.Pos, node.Type)
    171 					c.List[fmt.Sprintf("node_%d_%d", node.Layer, node.Pos)] = node.Type
    172 					for _, param := range neuron.Params {
    173 						switch param {
    174 						case "inputs":
    175 							result += fmt.Sprintf(", inputs:%d", node.Inputs)
    176 						case "outputs":
    177 							result += fmt.Sprintf(", outputs:%d", node.Outputs)
    178 						case "bias":
    179 							result += fmt.Sprintf(", bias:"+c.TypePrefix+"%f", node.Bias)
    180 						case "pos":
    181 							result += fmt.Sprintf(", pos:%d", node.Pos)
    182 						default:
    183 							if value, ok := c.Params[param]; ok {
    184 								result += fmt.Sprintf(", %s:%s", param, value)
    185 							} else {
    186 								return "", errors.New("Unknown parameter " + param)
    187 							}
    188 						}
    189 					}
    190 					result += "\n"
    191 				}
    192 			}
    193 		}
    194 
    195 		for _, weight := range n.Weights {
    196 			weightCP := fmt.Sprintf("weightcp_%d_%d__%d_%d", weight.Layer-1, weight.PosPrevLayer, weight.Layer, weight.PosCurrLayer)
    197 			downNode := fmt.Sprintf("node_%d_%d", weight.Layer-1, weight.PosPrevLayer)
    198 			upNode := fmt.Sprintf("node_%d_%d", weight.Layer, weight.PosCurrLayer)
    199 			result += fmt.Sprintf("%%meta cpdef %s romcode:weight, weight:"+c.TypePrefix+"%f\n", weightCP, weight.Value)
    200 			c.List[weightCP] = "weight"
    201 			result += fmt.Sprintf("%%meta iodef up%s type:io\n", weightCP)
    202 			result += fmt.Sprintf("%%meta iodef down%s type:io\n", weightCP)
    203 			result += fmt.Sprintf("%%meta ioatt down%s cp:%s, type:input, index:0\n", weightCP, weightCP)
    204 			result += fmt.Sprintf("%%meta ioatt down%s cp:%s, type:output, index:0\n", weightCP, downNode)
    205 			result += fmt.Sprintf("%%meta ioatt up%s cp:%s, type:input, index:%d\n", weightCP, upNode, weight.RelPosUp)
    206 			result += fmt.Sprintf("%%meta ioatt up%s cp:%s, type:output, index:0\n", weightCP, weightCP)
    207 		}
    208 	case FRAGMENT:
    209 		for _, node := range n.Nodes {
    210 			if node.Type == "input" {
    211 				result += fmt.Sprintf("%%meta fidef node_0_%d fragment:terminal\n", node.Pos)
    212 				c.List[fmt.Sprintf("node_0_%d", node.Pos)] = "input"
    213 				result += fmt.Sprintf("%%meta filinkdef input_%d type:fl\n", node.Pos)
    214 				result += fmt.Sprintf("%%meta filinkatt input_%d fi:node_0_%d, type:input, index:0\n", node.Pos, node.Pos)
    215 				result += fmt.Sprintf("%%meta filinkatt input_%d fi:ext, type:input, index:%d\n", node.Pos, node.Pos)
    216 			} else if node.Type == "output" {
    217 				result += fmt.Sprintf("%%meta fidef node_%d_%d fragment:terminal\n", node.Layer, node.Pos)
    218 				c.List[fmt.Sprintf("node_%d_%d", node.Layer, node.Pos)] = "output"
    219 				result += fmt.Sprintf("%%meta filinkdef output_%d type:fl\n", node.Pos)
    220 				result += fmt.Sprintf("%%meta filinkatt output_%d fi:node_%d_%d, type:output, index:0\n", node.Pos, node.Layer, node.Pos)
    221 				result += fmt.Sprintf("%%meta filinkatt output_%d fi:ext, type:output, index:%d\n", node.Pos, node.Pos)
    222 			} else {
    223 				if neuron, ok := n.Neurons[node.Type]; ok {
    224 					result += fmt.Sprintf("%%meta fidef node_%d_%d fragment:%s", node.Layer, node.Pos, node.Type)
    225 					c.List[fmt.Sprintf("node_%d_%d", node.Layer, node.Pos)] = node.Type
    226 					for _, param := range neuron.Params {
    227 						switch param {
    228 						case "inputs":
    229 							result += fmt.Sprintf(", inputs:%d", node.Inputs)
    230 						case "outputs":
    231 							result += fmt.Sprintf(", outputs:%d", node.Outputs)
    232 						case "bias":
    233 							result += fmt.Sprintf(", bias:"+c.TypePrefix+"%f", node.Bias)
    234 						case "pos":
    235 							result += fmt.Sprintf(", pos:%d", node.Pos)
    236 						default:
    237 							if value, ok := c.Params[param]; ok {
    238 								result += fmt.Sprintf(", %s:%s", param, value)
    239 							} else {
    240 								return "", errors.New("Unknown parameter " + param)
    241 							}
    242 						}
    243 					}
    244 					result += "\n"
    245 				}
    246 			}
    247 		}
    248 
    249 		for _, weight := range n.Weights {
    250 			weightFI := fmt.Sprintf("weightfi_%d_%d__%d_%d", weight.Layer-1, weight.PosPrevLayer, weight.Layer, weight.PosCurrLayer)
    251 			downNode := fmt.Sprintf("node_%d_%d", weight.Layer-1, weight.PosPrevLayer)
    252 			upNode := fmt.Sprintf("node_%d_%d", weight.Layer, weight.PosCurrLayer)
    253 			result += fmt.Sprintf("%%meta fidef %s fragment:weight", weightFI)
    254 			for _, param := range n.Neurons["weight"].Params {
    255 				switch param {
    256 				case "weight":
    257 					result += fmt.Sprintf(", weight:"+c.TypePrefix+"%f", weight.Value)
    258 				default:
    259 					if value, ok := c.Params[param]; ok {
    260 						result += fmt.Sprintf(", %s:%s", param, value)
    261 					} else {
    262 						return "", errors.New("Unknown parameter " + param)
    263 					}
    264 					result += "\n"
    265 				}
    266 			}
    267 			c.List[weightFI] = "weight"
    268 			result += fmt.Sprintf("%%meta filinkdef up%s type:fl\n", weightFI)
    269 			result += fmt.Sprintf("%%meta filinkdef down%s type:fl\n", weightFI)
    270 			result += fmt.Sprintf("%%meta filinkatt down%s fi:%s, type:input, index:0\n", weightFI, weightFI)
    271 			result += fmt.Sprintf("%%meta filinkatt down%s fi:%s, type:output, index:0\n", weightFI, downNode)
    272 			result += fmt.Sprintf("%%meta filinkatt up%s fi:%s, type:input, index:%d\n", weightFI, upNode, weight.RelPosUp)
    273 			result += fmt.Sprintf("%%meta filinkatt up%s fi:%s, type:output, index:0\n", weightFI, weightFI)
    274 		}
    275 
    276 		ProcessedNodes := make(map[string]struct{})
    277 		for node, _ := range c.List {
    278 			ProcessedNodes[node] = struct{}{}
    279 		}
    280 
    281 		// Processing pruned nodes
    282 		for _, pNode := range c.Pruned {
    283 			if _, ok := ProcessedNodes[pNode]; !ok {
    284 				return "", errors.New("Pruned node " + pNode + " not found")
    285 			} else {
    286 				result += fmt.Sprintf("%%meta fidef %s pruned:true\n", pNode)
    287 				delete(ProcessedNodes, pNode)
    288 			}
    289 		}
    290 
    291 		// Processing collapsed nodes
    292 		for _, cGroup := range c.Collapsed {
    293 			cName := ""
    294 			cCode := ""
    295 			for _, cNode := range cGroup {
    296 				if _, ok := ProcessedNodes[cNode]; !ok {
    297 					return "", errors.New("Collapsed node " + cNode + " not found")
    298 				} else {
    299 					cName += cNode + "_"
    300 					cCode += ":" + cNode
    301 					delete(ProcessedNodes, cNode)
    302 				}
    303 			}
    304 			result += fmt.Sprintf("%%meta cpdef %s fragcollapse%s\n", cName[:len(cName)-1], cCode)
    305 		}
    306 
    307 		// Processing remaining nodes
    308 		for node := range ProcessedNodes {
    309 			result += fmt.Sprintf("%%meta cpdef %s fragcollapse:%s\n", node, node)
    310 		}
    311 
    312 	default:
    313 		return "", errors.New("unknown operating mode")
    314 	}
    315 
    316 	return result, nil
