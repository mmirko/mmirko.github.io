     44 func (ev *BasmEvaluator) Visit(iProg *mel3program.Mel3Program) mel3program.Mel3Visitor {

     46 	debug := ev.Config.Debug

     66 			names := make([]string, arg_num)
     67 			for i, prog := range iProg.NextPrograms {
     68 				evaluators[i] = mel3program.ProgMux(ev, prog)
     69 				names[i] = nodeName + string(byte(97+i))
     70 				evaluators[i].(*BasmEvaluator).index = ev.index + string(byte(97+i))

     72 				evaluators[i].(*BasmEvaluator).group = ev.group

     82 						ev.MelBondConfig.CodeChan <- fmt.Sprintln("%meta filinkatt " + nodeName + "_" + names[i] + " fi:ext, type:output, index:" + iProg.ProgramValue)
     83 						ev.MelBondConfig.CodeChan <- fmt.Sprintln("%meta filinkatt " + nodeName + "_" + names[i] + " fi:" + names[i] + ", type:output, index:0")
     84 					} else {
     85 						ev.error = errors.New("an input is directly connected to an output")
     86 						return nil
     87 					}
     88 				} else {
     89 					ev.error = errors.New("no result")
     90 					return nil
     91 				}
     92 				ev.Result = result
     93 				return nil
     94 			}
     95 
     96 		case mel3program.B_IN_GROUP:
     97 			if ev.group != "" {
     98 				ev.error = errors.New("group already defined")
     99 				return nil
    100 			}
    101 
    102 			ev.group = iProg.ProgramValue
    103 			arg_num := len(iProg.NextPrograms) // It will be 1
    104 			evaluators := make([]mel3program.Mel3Visitor, arg_num)
    105 			for i, prog := range iProg.NextPrograms {
    106 				evaluators[i] = mel3program.ProgMux(ev, prog)
    107 				evaluators[i].(*BasmEvaluator).index = ev.index
    108 				evaluators[i].(*BasmEvaluator).groups = ev.groups
    109 				evaluators[i].(*BasmEvaluator).group = ev.group
    110 				evaluators[i].Visit(prog)
    111 				if evaluators[i].GetError() != nil {
    112 					ev.error = evaluators[i].GetError()
    113 					return nil
    114 				}
    115 				result := evaluators[i].GetResult()
    116 				if result == nil {
    117 					ev.error = errors.New("no result")
    118 					return nil
    119 				}
    120 				ev.Result = result
    121 				return nil
    122 			}
    123 
    124 		case mel3program.B_IN_UNGROUP:
    125 			if ev.group == "" {
    126 				ev.error = errors.New("no group defined")
    127 				return nil
    128 			}
    129 			ev.group = ""
    130 			arg_num := len(iProg.NextPrograms) // It will be 1
    131 			evaluators := make([]mel3program.Mel3Visitor, arg_num)
    132 			for i, prog := range iProg.NextPrograms {
    133 				evaluators[i] = mel3program.ProgMux(ev, prog)
    134 				evaluators[i].(*BasmEvaluator).index = ev.index
    135 				evaluators[i].(*BasmEvaluator).groups = ev.groups
    136 				evaluators[i].(*BasmEvaluator).group = ev.group
    137 				evaluators[i].Visit(prog)
    138 				if evaluators[i].GetError() != nil {
    139 					ev.error = evaluators[i].GetError()
    140 					return nil
    141 				}
    142 				result := evaluators[i].GetResult()

    144 					ev.error = errors.New("no result")
    145 					return nil
    146 				}
    147 				ev.Result = result
    148 				return nil
    149 			}
    150 		default:
    151 			ev.error = errors.New("unknown builtin")
    152 			return nil
    153 		}
    154 	} else {
    155 		implementation := ev.Implementation[iProg.LibraryID]
    156 
    157 		nodeCodeName := implementation.ImplName + "_" + implementation.ProgramNames[iProg.ProgramID]
    158 
    159 		if neuron, ok := ev.MelBondConfig.Neurons[nodeCodeName]; ok {
    160 
    161 			ev.MelBondConfig.CodeChan <- fmt.Sprint("%meta fidef " + nodeName + " fragment:" + nodeCodeName)
    162 
    163 			// groups management
    164 			myGroup := nodeName
    165 			if ev.group != "" {
    166 				myGroup = ev.group
    167 			}
    168 
    169 			if ev.groups == nil {

    181 			isFunctional := true

    184 				isFunctional = false
    185 			}
    186 
    187 			for _, param := range neuron.Params {

    189 					ev.MelBondConfig.CodeChan <- fmt.Sprintf(", %s:%s", implementation.ProgramNames[iProg.ProgramID], iProg.ProgramValue)
    190 					continue

    192 				switch param {

    194 					if value, ok := ev.MelBondConfig.Params[param]; ok {
    195 						ev.MelBondConfig.CodeChan <- fmt.Sprintf(", %s:%s", param, value)

    207 			names := make([]string, arg_num)
    208 			for i, prog := range iProg.NextPrograms {
    209 				evaluators[i] = mel3program.ProgMux(ev, prog)
    210 				names[i] = nodeName + string(byte(97+i))

    213 				evaluators[i].(*BasmEvaluator).group = ev.group
    214 				evaluators[i].Visit(prog)
    215 				if evaluators[i].GetError() != nil {
    216 					ev.error = evaluators[i].GetError()
    217 					return nil
    218 				}
    219 				result := evaluators[i].GetResult()

    221 				if result != nil {

    223 						ev.MelBondConfig.CodeChan <- fmt.Sprintln("%meta filinkatt " + nodeName + "_" + names[i] + " fi:" + nodeName + ", type:input, index:" + fmt.Sprint(i))
