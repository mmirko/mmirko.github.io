     51 func (n *MelBondConfig) Init() error {
     52 	n.Neurons = make(map[string]*Neuron)
     53 
     54 	// List nb files in the neuron library path and load them
     55 	neuronFiles, err := ioutil.ReadDir(n.NeuronLibPath)
     56 	if err != nil {
     57 		return err
     58 	}
     59 	for _, f := range neuronFiles {
     60 		if len(f.Name()) > 3 && f.Name()[len(f.Name())-3:] == ".nb" {
     61 			neuronFile, err := ioutil.ReadFile(n.NeuronLibPath + "/" + f.Name())
     62 			if err != nil {
     63 				return err
     64 			}
     65 			neuron := new(Neuron)
     66 			if err := json.Unmarshal(neuronFile, neuron); err != nil {
     67 				return err
     68 			}
     69 			n.Neurons[f.Name()[0:len(f.Name())-3]] = neuron
     70 		}
     71 	}
     72 
     73 	return nil
     74 }

     76 func (p *MelBondProgram) WriteBasm() (string, error) {
     77 
     78 	ctx, cancel := context.WithCancel(context.Background())
     79 	p.CodeChan = make(chan string)
     80 	p.CancelCtx = cancel
     81 
     82 	//Create the groups map and copy it to the environment
     83 
     84 	var groups interface{}
     85 	groups = make(Groups)
     86 	p.Mel3Object.Environment = &groups
     87 
     88 	go func() {
     89 		for {
     90 			select {
     91 			case <-ctx.Done():
     92 				return
     93 			case code := <-p.CodeChan:
     94 				p.BasmCode += code
     95 			}
     96 		}
     97 	}()
     98 
     99 	defer cancel()
    100 
    101 	if err := p.MelStringImport(p.Source); err != nil {
    102 		return "", err
    103 	} else {
    104 
    105 		regSize := p.RegisterSize
    106 		p.CodeChan <- fmt.Sprintf("%%meta bmdef     global registersize:%d\n", regSize)
    107 		switch p.IOMode {
    108 		case ASYNC:
    109 			p.CodeChan <- fmt.Sprintf("%%meta bmdef     global iomode:async\n")
    110 		case SYNC:
    111 			p.CodeChan <- fmt.Sprintf("%%meta bmdef     global iomode:sync\n")
    112 		}
    113 
    114 		if err := p.Compute(); err != nil {
    115 			return "", err
    116 		} else {
    117 			for gName, group := range groups.(Groups) {
    118 				nodeList := ""
    119 				for _, nName := range group {
    120 					nodeList += ":" + nName
    121 				}
    122 				p.CodeChan <- fmt.Sprintln("%meta cpdef  " + gName + " fragcollapse" + nodeList)
    123 			}
    124 			time.Sleep(1 * time.Second)
    125 			return p.BasmCode, nil
    126 		}
    127 	}
