    536 			result := make([]VarCell, len(bgfunct.Returns))
    537 			for i, cell := range bgfunct.Returns {
    538 				result[i] = cell
    539 			}
    540 
    541 			// Clean the inputs arguments VarCells
    542 			for vari, cell := range bgfunct.Vars {
    543 				if bg.In_debug() {
    544 					fmt.Println("Cleaning " + vari)
    545 				}
    546 
    547 				bg.Reqs <- VarReq{REQ_REMOVE, bg.CurrentRoutine, cell}
    548 				resp := <-bg.Answers
    549 				if resp.AnsType != ANS_OK {
    550 					bg.Set_faulty("Resource removal failed")
    551 					return []VarCell{}, false
    552 				}
    553 			}
    554 
    555 			return result, true
    556 		}
