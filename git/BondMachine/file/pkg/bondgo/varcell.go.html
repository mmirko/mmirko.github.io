     32 // The type to hold variables in processor objects (mem or registers or whatever) and in bondmachine objects
     33 type VarCell struct {
     34 	Vtype          *VarType // The variable type
     35 	Procobjtype    uint8    // 0 Register, 1 RAM, 2 output (notin vars) etc.
     36 	Id             int      // Local (within the processor) reference to the object among objects of the same type
     37 	Start_id       int      // Local (within the processor) reference to the object among objects of the same type (start data)
     38 	End_id         int      // Local (within the processor) reference to the object among objects of the same type (end data)
     39 	Global_id      int      // Global reference to the object among objects of the same type
     40 	Start_globalid int      // Global reference to the object among objects of the same type (start data)
     41 	End_globalid   int      // Global reference to the object among objects of the same type (end data)
     42 }

     60 // Variable type equality
     61 func Same_Type(t1 *VarType, t2 *VarType) bool {
     62 	if t1 != nil && t2 != nil {
     63 		if t1.MainType == t2.MainType {
     64 			switch t1.MainType {
     65 			case T_NAMED:
     66 				if t1.Name == t2.Name {
     67 					return true
     68 				}
     69 			case T_CHAN, T_STAR:
     70 				if len(t1.Values) == 1 && len(t2.Values) == 1 {
     71 					if Same_Type(t1.Values[0], t2.Values[0]) {
     72 						return true
     73 					}
     74 				}
     75 			case T_STRUCT:
     76 				// TODO
     77 			}
     78 		}
     79 	}
     80 	return false
     81 }

     83 // Type from string
     84 func Type_from_string(types string) (*VarType, error) {
     85 	t := strings.TrimSpace(types)
     86 	first_space := strings.Index(t, " ")
     87 	if first_space == -1 {
     88 		if t != "" {
     89 			switch t {
     90 			case "uint8", "uint16", "uint32", "uint64", "bool":
     91 				newtype := new(VarType)
     92 				newtype.MainType = T_NAMED
     93 				newtype.Name = t
     94 				newtype.Values = make([]*VarType, 0)
     95 				return newtype, nil
     96 			default:
     97 				return nil, errors.New("Type " + t + " unsupported")
     98 			}
     99 		}
    100 	} else {
    101 		mod := t[:first_space]
    102 		switch mod {
    103 		case "*":
    104 			if inner_type, err := Type_from_string(t[first_space:]); err == nil {
    105 				newtype := new(VarType)
    106 				newtype.MainType = T_STAR
    107 				newtype.Name = ""
    108 				newtype.Values = make([]*VarType, 1)
    109 				newtype.Values[0] = inner_type
    110 				return newtype, nil
    111 			}
    112 		case "chan":
    113 			if inner_type, err := Type_from_string(t[first_space:]); err == nil {
    114 				newtype := new(VarType)
    115 				newtype.MainType = T_CHAN
    116 				newtype.Name = ""
    117 				newtype.Values = make([]*VarType, 1)
    118 				newtype.Values[0] = inner_type
    119 				return newtype, nil
    120 			}
    121 		case "struct":
    122 			// TODO
    123 		}
    124 	}
    125 	return nil, errors.New("Type unknown")
    126 }

    128 // Type from ast
    129 func Type_from_ast(spec ast.Node) (*VarType, error) {
    130 	switch vtype := spec.(type) {
    131 	case *ast.Ident:
    132 		newtype := new(VarType)
    133 		newtype.MainType = T_NAMED
    134 		newtype.Name = vtype.Name
    135 		newtype.Values = make([]*VarType, 0)
    136 		return newtype, nil
    137 	case *ast.ChanType:
    138 		if inner_type, err := Type_from_ast(vtype.Value); err == nil {
    139 			newtype := new(VarType)
    140 			newtype.MainType = T_CHAN
    141 			newtype.Name = ""
    142 			newtype.Values = make([]*VarType, 1)
    143 			newtype.Values[0] = inner_type
    144 			return newtype, nil
    145 		}
    146 	case *ast.StarExpr:
    147 		if inner_type, err := Type_from_ast(vtype.X); err == nil {
    148 			newtype := new(VarType)
    149 			newtype.MainType = T_STAR
    150 			newtype.Name = ""
    151 			newtype.Values = make([]*VarType, 1)
    152 			newtype.Values[0] = inner_type
    153 			return newtype, nil
    154 		}
    155 	case *ast.StructType:
    156 	}
    157 	return nil, errors.New("Import failed")
    158 }

    160 // VarCell to string
    161 func (m VarCell) String() string {
    162 	result := "<"
    163 	switch m.Procobjtype {
    164 	case REGISTER:
    165 		result += "reg "
    166 	case INPUT:
    167 		result += "input "
    168 	case OUTPUT:
    169 		result += "output "
    170 	case MEMORY:
    171 		result += "mem "
    172 	case CHANNEL:
    173 		result += "chan "
    174 	}
    175 	result += strconv.Itoa(m.Id) + ">"
    176 	return result
