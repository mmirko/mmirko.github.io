     62 func (vr *VarReq) String() string {
     63 	result := ""
     64 	switch vr.ReqType {
     65 	case REQ_NEW:
     66 		result += "new "
     67 	case REQ_REMOVE:
     68 		result += "del "
     69 	}
     70 	result += "proc " + strconv.Itoa(vr.Processor_id) + " "
     71 	result += "cell " + vr.Cell.String()
     72 	// TODO Finish
     73 	return result
     74 }
