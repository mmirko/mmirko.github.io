    572 		newchannels := make([]uint32, 0)
    573 
    574 		for bio, ioid := range res.Map.Assoc {
    575 			if bio[0] == 'i' {
    576 				connected := false
    577 				multi := false
    578 
    579 				// Search the input among ext cluster outputs
    580 				if extc != nil {
    581 				extloop:
    582 					for _, peer := range extc.Peers {
    583 						for _, outp := range peer.Outputs {
    584 							if strconv.Itoa(int(outp)) == ioid {
    585 								if connected {
    586 									multi = true
    587 									break extloop
    588 								} else {
    589 									connected = true
    590 								}
    591 							}
    592 						}
    593 					}
    594 				}
    595 				// Search the input among other bondmachines
    596 			bmloop:
    597 				for otherid, otherres := range partialio {
    598 					// Not searching on the same BM
    599 					if otherid != myid {
    600 						for otherbio, otherioid := range otherres.Map.Assoc {
    601 							if ioid == otherioid {
    602 								if otherbio[0] == 'i' {
    603 									// That's ok to have other inputs connected to the same output
    604 								} else if otherbio[0] == 'o' {
    605 									if connected {
    606 										multi = true
    607 										break bmloop
    608 									} else {
    609 										connected = true
    610 									}
    611 
    612 								}
    613 							}
    614 						}
    615 					}
    616 				}
    617 
    618 				if multi {
    619 					return nil, nil, nil, nil, nil, errors.New("Input connected to multiple outputs")
    620 				}
    621 
    622 				if connected {
    623 					// The Input is within the cluster
    624 					newio.Assoc[bio] = ioid
    625 					ioid_n, _ := strconv.Atoi(ioid)
    626 					newinputs = append(newinputs, uint32(ioid_n))
    627 				} else {
    628 					// Its residual
    629 					newresidual.Map.Assoc[bio] = ioid
    630 				}
    631 
    632 			} else if bio[0] == 'o' {
    633 				connected := false
    634 				multioutput := false
    635 
    636 				// Search the output among ext cluster inputs
    637 				if extc != nil {
    638 					for _, peer := range extc.Peers {
    639 						for _, inp := range peer.Inputs {
    640 							if strconv.Itoa(int(inp)) == ioid {
    641 								connected = true
    642 								break
    643 							}
    644 						}
    645 
    646 						for _, outp := range peer.Outputs {
    647 							if strconv.Itoa(int(outp)) == ioid {
    648 								multioutput = true
    649 								break
    650 							}
    651 						}
    652 					}
    653 				}
    654 
    655 				for otherid, otherres := range partialio {
    656 					// Not searching on the same BM
    657 					if otherid != myid {
    658 						for otherbio, otherioid := range otherres.Map.Assoc {
    659 							if ioid == otherioid {
    660 								if otherbio[0] == 'o' {
    661 									multioutput = true
    662 								} else if otherbio[0] == 'i' {
    663 									connected = true
    664 								}
    665 							}
    666 						}
    667 					}
    668 				}
    669 
    670 				if multioutput {
    671 					return nil, nil, nil, nil, nil, errors.New("Multiple outputs have the same id")
    672 				}
    673 
    674 				if connected {
    675 					// The Output is within the cluster
    676 					newio.Assoc[bio] = ioid
    677 					ioid_n, _ := strconv.Atoi(ioid)
    678 					newoutputs = append(newoutputs, uint32(ioid_n))
    679 				} else {
    680 					// Its residual
    681 					newresidual.Map.Assoc[bio] = ioid
    682 				}
    683 
    684 			}
    685 		}
    686 
    687 		resultio = append(resultio, newio)
    688 		resultresi = append(resultresi, newresidual)
    689 
    690 		newpeer := udpbond.Peer{resultpeerids[myid], newchannels, newinputs, newoutputs}
    691 		resultcluster.Peers = append(resultcluster.Peers, newpeer)
    692 
    693 	}
    694 
    695 	return resultcluster, resultpeerids, resultbond, resultio, resultresi, nil
    696 }

    698 func Assembly_2_Processor(rsize int, asmcode []string) (*procbuilder.Machine, error) {
    699 
    700 	config := new(BondgoConfig)
    701 	config.Debug = false
    702 	config.Verbose = false
    703 	config.Mpm = false
    704 	config.Cascading_io = false
    705 
    706 	switch rsize {
    707 	case 8:
    708 		config.Rsize = uint8(8)
    709 		config.Basic_type = "uint8"
    710 		config.Basic_chantype = "chan uint8"
    711 	case 16:
    712 		config.Rsize = uint8(16)
    713 		config.Basic_type = "uint16"
    714 		config.Basic_chantype = "chan uint16"
    715 	case 32:
    716 		config.Rsize = uint8(32)
    717 		config.Basic_type = "uint32"
    718 		config.Basic_chantype = "chan uint32"
    719 	case 64:
    720 		config.Rsize = uint8(64)
    721 		config.Basic_type = "uint64"
    722 		config.Basic_chantype = "chan uint64"
    723 	default:
    724 		config.Rsize = uint8(8)
    725 		config.Basic_type = "uint8"
    726 		config.Basic_chantype = "chan uint8"
    727 	}
    728 
    729 	usagedone := make(chan bool)
    730 
    731 	results := new(BondgoResults) // Results go in here
    732 	results.Init_Results(config)
    733 
    734 	messages := new(BondgoMessages) // Compiler logs and errors
    735 	messages.Init_Messages(config)
    736 
    737 	reqmnts := new(BondgoRequirements) // The pointer to the requirements struct
    738 	reqmnts.Init_Requirements(config)
    739 
    740 	usagenotify := make(chan UsageNotify) // Used to notify the used resource
    741 
    742 	go reqmnts.Usage_Monitor(usagenotify, usagedone) // Spawn the usage monitor
    743 
    744 	run := new(BondgoRuninfo) // Running data
    745 	run.Init_Runinfo(config)
    746 
    747 	varreq := make(chan VarReq) // Variable request
    748 	varans := make(chan VarAns) // Variable response
    749 
    750 	functs := new(BondgoFunctions) // Functions
    751 	functs.Init_Functions(config, messages)
    752 
    753 	vars := make(map[string]VarCell)
    754 	returns := make([]VarCell, 0)
    755 
    756 	bgmain := &BondgoCheck{results, config, reqmnts, run, messages, functs, usagenotify, varreq, varans, nil, nil, vars, returns, "", "", "device_0", 0}
    757 
    758 	bgmain.Used <- UsageNotify{TR_PROC, 0, C_DEVICE, bgmain.CurrentDevice, I_NIL}
    759 
    760 	bgmain.Abstract_assembler(rsize, asmcode, usagenotify)
    761 
    762 	for _, currline := range asmcode {
    763 		bgmain.WriteLine(0, currline)
    764 	}
    765 
    766 	for procid, rout := range bgmain.Program {
    767 		// TODO Recheck
    768 		linesn := len(rout.Lines)
    769 		bgmain.Used <- UsageNotify{TR_PROC, procid, C_ROMSIZE, S_NIL, linesn}
    770 	}
    771 
    772 	bgmain.Used <- UsageNotify{TR_EXIT, 0, 0, S_NIL, I_NIL}
    773 	<-usagedone
    774 
    775 	if mymachine, ok := bgmain.Create_Connecting_Processor(rsize, 0); ok {
    776 		return mymachine, nil
    777 	} else {
    778 		return nil, errors.New("Creating processor failed")
    779 	}
    780 
    781 	return nil, nil
    782 }

    784 func MultiAsm2BondMachine(rsize int, aafile *Abs_assembly) (*bondmachine.Bondmachine, error) {
    785 	bmach := new(bondmachine.Bondmachine)
    786 	bmach.Rsize = uint8(rsize)
    787 	bmach.Init()
    788 
    789 	for i, proc_prog := range aafile.ProcProgs {
    790 		if mach, err := Assembly_2_Processor(rsize, strings.Split(proc_prog, "\n")); err == nil {
    791 			bmach.Domains = append(bmach.Domains, mach)
    792 			if _, ok := bmach.Add_processor(i); ok != nil {
    793 				return nil, errors.New("Attach processor failed")
    794 			}
    795 
    796 		} else {
    797 			return nil, errors.New("Creating processor failed")
    798 		}
    799 	}
    800 
    801 	for _, bond := range aafile.Bonds {
    802 		endpoints := strings.Split(bond, ",")
    803 		if endpoints[0][0] == 'i' {
    804 			for {
    805 				max := "i" + strconv.Itoa(bmach.Inputs)
    806 				if max <= endpoints[0] {
    807 					bmach.Add_input()
    808 				} else {
    809 					break
    810 				}
    811 			}
    812 		}
    813 		if endpoints[0][0] == 'o' {
    814 			for {
    815 				max := "o" + strconv.Itoa(bmach.Outputs)
    816 				if max <= endpoints[0] {
    817 					bmach.Add_output()
    818 				} else {
    819 					break
    820 				}
    821 			}
    822 		}
    823 
    824 		if endpoints[1][0] == 'i' {
    825 			for {
    826 				max := "i" + strconv.Itoa(bmach.Inputs)
    827 				if max <= endpoints[1] {
    828 					bmach.Add_input()
    829 				} else {
    830 					break
    831 				}
    832 			}
    833 		}
    834 		if endpoints[1][0] == 'o' {
    835 			for {
    836 				max := "o" + strconv.Itoa(bmach.Outputs)
    837 				if max <= endpoints[1] {
    838 					bmach.Add_output()
    839 				} else {
    840 					break
    841 				}
    842 			}
    843 		}
    844 
    845 		bmach.Add_bond(endpoints)
    846 
    847 	}
    848 
    849 	return bmach, nil
    850 }
