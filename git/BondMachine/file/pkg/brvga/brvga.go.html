     30 func NewBrvgaTextMemory(constraint string) (*BrvgaTextMemory, error) {
     31 	components := strings.Split(constraint, ":")
     32 	componentsN := len(components)
     33 	if (componentsN-1)%5 == 0 {
     34 		boxes := make([]cpTextMemory, 0)
     35 
     36 		for i := 1; i < componentsN; i = i + 5 {
     37 			newBox := cpTextMemory{}
     38 			if newCP, err := strconv.Atoi(components[i]); err == nil {
     39 				newBox.CpId = newCP
     40 			} else {
     41 				return nil, err
     42 			}
     43 			if newLeft, err := strconv.Atoi(components[i+1]); err == nil {
     44 				newBox.LeftPos = newLeft
     45 			} else {
     46 				return nil, err
     47 			}
     48 			if newTop, err := strconv.Atoi(components[i+2]); err == nil {
     49 				newBox.TopPos = newTop
     50 			} else {
     51 				return nil, err
     52 			}
     53 			if newWidth, err := strconv.Atoi(components[i+3]); err == nil {
     54 				newBox.Width = newWidth
     55 			} else {
     56 				return nil, err
     57 			}
     58 			if newHeight, err := strconv.Atoi(components[i+4]); err == nil {
     59 				newBox.Height = newHeight
     60 			} else {
     61 				return nil, err
     62 			}
     63 
     64 			memL := newBox.Width * newBox.Height
     65 			newBox.Mem = make([]byte, memL)
     66 			for j := 0; j < memL; j++ {
     67 				newBox.Mem[j] = 0x00

     70 			newBox.cpProtector = sync.RWMutex{}
     71 
     72 			boxes = append(boxes, newBox)
     73 		}
     74 
     75 		result := new(BrvgaTextMemory)
     76 		result.constraintString = constraint
     77 		result.Cps = boxes
     78 		return result, nil
     79 	}

    102 func (b *BrvgaTextMemory) GetCpMem(cpId int) ([]byte, error) {
    103 	for _, cp := range b.Cps {
    104 		if cp.CpId == cpId {
    105 			cp.cpProtector.RLock()
    106 			result := make([]byte, len(cp.Mem))
    107 			copy(result, cp.Mem)
    108 			cp.cpProtector.RUnlock()
    109 			return result, nil
    110 		}
    111 	}
    112 	return nil, errors.New("cp not found")

    155 func (b *BrvgaTextMemory) handleMessage(ctx context.Context, buf []byte) {
    156 	recv := &Textmemupdate{}
    157 
    158 	if err := proto.Unmarshal(buf, recv); err != nil {
    159 		return
    160 	}
    161 
    162 	for _, box := range b.Cps {
    163 		if box.CpId == int(recv.Cpid) {
    164 			box.cpProtector.Lock()
    165 			for _, seq := range recv.Seq {
    166 				copy(box.Mem[seq.Pos:seq.Pos+uint32(len(seq.Payload))], seq.Payload)
    167 			}
    168 			box.cpProtector.Unlock()
    169 		}
    170 	}
