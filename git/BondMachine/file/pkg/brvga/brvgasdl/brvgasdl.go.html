     33 func NewBrvgaSdlUnixSock(constraint string, sockPath string, headerPath string, fontsPath string) (*BrvgaSdl, error) {
     34 	result := new(BrvgaSdl)
     35 
     36 	// Create the brvga memory
     37 	textMem, err := brvga.NewBrvgaTextMemory(constraint)
     38 	if err != nil {
     39 		return nil, err
     40 	}
     41 	result.BrvgaTextMemory = textMem
     42 
     43 	// Create the canvas and window
     44 	wnd, canvas, err := sdlcanvas.CreateWindow(800, 600, "BondMachine")
     45 	if err != nil {
     46 		return nil, err
     47 	}
     48 	result.Window = wnd
     49 	result.Canvas = canvas
     50 
     51 	// Allocate the fonts array
     52 	result.Fonts = new(Fonts)
     53 	result.Fonts.fonts = make([]byte, 0)
     54 
     55 	// Allocate the header array
     56 	result.Header = new(Header)
     57 	result.Header.header = make([]byte, 0)
     58 
     59 	// Read the header
     60 	headerFile, err := os.Open(headerPath)
     61 	if err != nil {
     62 		return nil, err
     63 	}
     64 	defer headerFile.Close()
     65 
     66 	headerReader := bufio.NewScanner(headerFile)
     67 	for headerReader.Scan() {
     68 		if line := headerReader.Text(); len(line) > 0 {
     69 			if line[:2] == "0x" {
     70 				hexString := line[2:]
     71 				if decoded, err := hex.DecodeString(hexString); err == nil {
     72 					result.header = append(result.header, decoded...)
     73 				}
     74 			}
     75 		}
     76 	}
     77 
     78 	// Read the fonts file
     79 	fontsFile, err := os.Open(fontsPath)
     80 	if err != nil {
     81 		return nil, err
     82 	}
     83 	defer fontsFile.Close()
     84 
     85 	fontsReader := bufio.NewScanner(fontsFile)
     86 	for fontsReader.Scan() {
     87 		if line := fontsReader.Text(); len(line) > 0 {
     88 
     89 			if line[:2] == "0x" {
     90 				hexString := line[2:]
     91 				if decoded, err := hex.DecodeString(hexString); err == nil {
     92 					result.Fonts.fonts = append(result.Fonts.fonts, decoded...)
     93 				}
     94 			}
     95 		}
     96 	}
     97 
     98 	result.images = make(map[byte]*image.RGBA)
     99 
    100 	// Load 8x8 fonts
    101 	for c := 0; c < 128; c++ {
    102 
    103 		newChar := image.NewRGBA(image.Rectangle{Max: image.Point{X: 8, Y: 8}})
    104 
    105 		for i := 0; i < 8; i++ {
    106 			charLine := result.Fonts.fonts[c*8+i]
    107 
    108 			for j := 0; j < 8; j++ {
    109 				if charLine&byte((1<<j)) > 0 {
    110 					newChar.Set(j, i, color.White)
    111 				} else {
    112 					// newChar.Set(j, i, color.Black)
    113 				}
    114 			}
    115 		}
    116 
    117 		result.Fonts.images[byte(c)] = newChar
    118 	}
    119 
    120 	ctx, _ := context.WithCancel(context.Background())
    121 
    122 	go result.UNIXSockReceiver(ctx, "/tmp/brvga.sock")
    123 
    124 	return result, nil
    125 }
