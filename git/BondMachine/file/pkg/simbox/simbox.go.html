     45 func (rule Rule) String() string {
     46 	switch rule.Timec {
     47 	case TIMEC_ABS:
     48 		switch rule.Action {
     49 		case ACTION_SET:
     50 			return "absolute:" + strconv.Itoa(int(rule.Tick)) + ":set:" + rule.Object + ":" + rule.Extra
     51 		case ACTION_GET:
     52 			return "absolute:" + strconv.Itoa(int(rule.Tick)) + ":get:" + rule.Object + ":" + rule.Extra
     53 		case ACTION_SHOW:
     54 			return "absolute:" + strconv.Itoa(int(rule.Tick)) + ":show:" + rule.Object + ":" + rule.Extra
     55 		}
     56 	case TIMEC_NONE:
     57 		switch rule.Action {
     58 		case ACTION_CONFIG:
     59 			switch rule.Object {
     60 			case "get_all":
     61 				return "config:get_all:" + rule.Extra
     62 			case "get_all_internal":
     63 				return "config:get_all_internal:" + rule.Extra
     64 			default:
     65 				return "config:" + rule.Object
     66 			}
     67 		}
     68 	case TIMEC_REL:
     69 		switch rule.Action {
     70 		case ACTION_SET:
     71 			return "relative:" + strconv.Itoa(int(rule.Tick)) + ":set:" + rule.Object + ":" + rule.Extra
     72 		case ACTION_GET:
     73 			return "relative:" + strconv.Itoa(int(rule.Tick)) + ":get:" + rule.Object + ":" + rule.Extra
     74 		case ACTION_SHOW:
     75 			return "relative:" + strconv.Itoa(int(rule.Tick)) + ":show:" + rule.Object + ":" + rule.Extra
     76 		}
     77 	}
     78 	return ""
     79 }
