    139 func (bi *BasmInstance) SetActive(pass string) error {
    140 	for passN, v := range GetPassMnemonic() {
    141 		if v == pass {
    142 			if ch, ok := IsOptionalPass()[passN]; ok {
    143 				if ch {
    144 					bi.passes = bi.passes | passN
    145 					return nil
    146 				} else {
    147 					return errors.New("pass is not optional")
    148 				}
    149 			} else {
    150 				return errors.New("pass is not defined")
    151 			}
    152 		}
    153 	}
    154 	return errors.New("pass not found")
    155 }
