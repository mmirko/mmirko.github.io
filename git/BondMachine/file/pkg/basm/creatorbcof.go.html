     14 // Assembler2BCOF transform an assembled instance into a BCOF file
     15 func (bi *BasmInstance) Assembler2BCOF() error {
     16 
     17 	if bi.debug {
     18 		fmt.Println("\t" + green("BCOF metadata"))
     19 	}
     20 
     21 	registerSize := bi.global.GetMeta("registersize")
     22 
     23 	if registerSize == "" {
     24 		return errors.New("register size not specified")
     25 	}
     26 
     27 	var rSize uint8
     28 	if size, err := strconv.Atoi(registerSize); err == nil {
     29 		if 0 < size && size < 256 {
     30 			rSize = uint8(size)
     31 		} else {
     32 			return errors.New("wrong value for register size")
     33 		}
     34 	} else {
     35 		return errors.New("register size not valid")
     36 	}
     37 	if bi.debug {
     38 		fmt.Println("\t\t"+green("register size:"), rSize)
     39 	}
     40 
     41 	// Compute the register size padded to 8 bit multiples (the BCOF format requires bytes)
     42 	rSizePad := 8 * ((rSize + 7) / 8)
     43 
     44 	if bi.debug {
     45 		fmt.Println("\t\t"+green("register size padded:"), rSizePad)
     46 	}
     47 
     48 	if bi.debug {
     49 		fmt.Println("\t" + green("BCOF creation"))
     50 	}
     51 
     52 	bi.outBCOF = bcof.NewBCOF(uint32(rSize))
     53 	bi.outBCOF.SetId(1)
     54 	bi.outBCOF.SetSignature("bm")
     55 
     56 	idDone := make(map[int]struct{})
     57 
     58 	for _, cp := range bi.cps {
     59 		if bi.debug {
     60 			fmt.Println("\t\t" + green("CP: ") + yellow(cp.GetValue()))
     61 		}
     62 
     63 		ramCode := cp.GetMeta("ramcode")
     64 		if bi.debug {
     65 			if ramCode != "" {
     66 				fmt.Println("\t\t - " + green("ram code: ") + yellow(ramCode))
     67 			} else {
     68 				fmt.Println("\t\t - " + green("ram code: ") + yellow("not specified"))
     69 			}
     70 		}
     71 
     72 		ramData := cp.GetMeta("ramdata")
     73 		if bi.debug {
     74 			if ramData != "" {
     75 				fmt.Println("\t\t - " + green("ram data: ") + yellow(ramData))
     76 			} else {
     77 				fmt.Println("\t\t - " + green("ram data: ") + yellow("not specified"))
     78 			}
     79 		}
     80 
     81 		if ramCode == "" && ramData == "" {
     82 			if bi.debug {
     83 				fmt.Println("\t\t - " + purple("no ram code or ram data specified, skipping"))
     84 			}
     85 			continue
     86 		}
     87 
     88 		var program *procbuilder.Program
     89 
     90 		cpPayload := bcof.NewBCOFData(uint32(rSize))
     91 
     92 		// To identify the CP id we have to check the following:
     93 		// - if the BM came from this assembler run, the BMinfo file will contain the CP id and we have to use it
     94 		// - if the BM came from the CLI, and the BMinfo is provided, we have to use the CP id from the BMinfo file
     95 		// - if the BM came from the CLI, and the BMinfo is not provided, we will use the convention of the CP name in the form of "cpN" (where N is a number).
     96 
     97 		// Start with the instructions
     98 		// Search th ID in the BMinfo data
     99 		if bi.BMinfo != nil {
    100 			for id, name := range bi.BMinfo.CPNames {
    101 				if name == cp.GetValue() {
    102 					if _, ok := idDone[id]; ok {
    103 						return errors.New("CP id already used")
    104 					}
    105 					idDone[id] = struct{}{}
    106 					if bi.debug {
    107 						fmt.Println("\t\t - " + green("cpId: ") + yellow(id))
    108 					}
    109 
    110 					cpPayload.SetId(uint32(id))
    111 					cpPayload.SetSignature("cp" + strconv.Itoa(id)) // TODO: temporary name for the signature
    112 
    113 					prog := ""
    114 					if ramCode != "" {
    115 						for _, line := range bi.sections[ramCode].sectionBody.Lines {
    116 							prog += line.Operation.GetValue()
    117 							for _, arg := range line.Elements {
    118 								prog += " " + arg.GetValue()
    119 							}
    120 							prog += "\n"
    121 						}
    122 					}
    123 					// Get the machine
    124 					myMachine := bi.result
    125 					myArch := myMachine.Domains[myMachine.Processors[id]].Arch
    126 
    127 					if assembled, err := myArch.Assembler([]byte(prog)); err == nil {
    128 						program = &assembled
    129 					} else {
    130 						return err
    131 					}
    132 
    133 				}
    134 			}
    135 		} else {
    136 
    137 			re := regexp.MustCompile("^cp(?P<cpId>[0-9]+)$")
    138 			if re.MatchString(cp.GetValue()) {
    139 				cpId := re.ReplaceAllString(cp.GetValue(), "${cpId}")
    140 				if bi.debug {
    141 					fmt.Println("\t\t - " + green("cpId: ") + yellow(cpId))
    142 				}
    143 				//TODO
    144 			} else {
    145 				//TODO
    146 			}
    147 		}
    148 
    149 		// Continuing with the data
    150 		if ramData != "" {
    151 			for _, line := range bi.sections[ramData].sectionBody.Lines {
    152 				for _, arg := range line.Elements {
    153 					hexVal := arg.GetValue()
    154 					if n, err := bmnumbers.ImportString(hexVal); err == nil {
    155 						nS, _ := n.ExportBinaryNBits(int(rSizePad))
    156 						program.Slocs = append(program.Slocs, nS)
    157 					} else {
    158 						return err
    159 					}
    160 				}
    161 			}
    162 		}
    163 
    164 		// Put every line of the program (and data) in the payload
    165 		for _, line := range program.Slocs {
    166 			// Pad the line to the padded register size
    167 			for len(line) < int(rSizePad) {
    168 				line = "0" + line
    169 			}
    170 			// Read the binary number with bmnumbers
    171 			// fmt.Println("0b<" + strconv.Itoa(int(rSizePad)) + ">" + line)
    172 			num, _ := bmnumbers.ImportString("0b<" + strconv.Itoa(int(rSizePad)) + ">" + line)
    173 			cpPayload.Payload = append(cpPayload.Payload, num.GetBytes()...)
    174 		}
    175 
    176 		bi.outBCOF.AddData(cpPayload)
    177 
    178 		if bi.debug {
    179 			fmt.Println(green("\t\tBCOF entry created dump: "))
    180 			fmt.Println(green("\t\t----"))
    181 			fmt.Println(cpPayload.Dump())
    182 			fmt.Println(green("\t\t----"))
    183 		}
    184 	}
    185 
    186 	return nil
    187 }
