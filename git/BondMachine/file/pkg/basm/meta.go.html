     24 func (bi *BasmInstance) literalProcessor(meta string) error {
     25 
     26 	stripped := strings.TrimSpace(strings.TrimSpace(meta)[7:]) // Remove the %meta: part and the literal
     27 
     28 	re := regexp.MustCompile(`^(?P<meta>\w+)\s+(?P<value>.+)$`)
     29 	if re.MatchString(stripped) {
     30 		name := re.ReplaceAllString(stripped, "${meta}")
     31 		value := re.ReplaceAllString(stripped, "${value}")
     32 
     33 		if bi.isWithinFragment != "" {
     34 			if bi.debug {
     35 				fmt.Println("Adding %meta literal: ||" + name + "|| = ||" + value + "|| to " + bi.isWithinFragment)
     36 			}
     37 			bi.fragments[bi.isWithinFragment].fragmentBody.BasmMeta = bi.fragments[bi.isWithinFragment].fragmentBody.SetMeta(name, value)
     38 		}
     39 
     40 	} else {
     41 		return errors.New("Invalid %meta: " + meta)
     42 	}
     43 
     44 	return nil
     45 }

     47 func (bi *BasmInstance) metaProcessor(meta string) error {
     48 
     49 	// TODO Horrific and temporary code, a proper parser is desireable
     50 	space := regexp.MustCompile(`\s+`)
     51 
     52 	stripped := strings.TrimSpace(meta)
     53 	stripped = space.ReplaceAllString(stripped, " ")
     54 
     55 	splitted := strings.Split(stripped, " ")
     56 
     57 	if splitted[0] == "literal" {
     58 		return bi.literalProcessor(meta)
     59 	}
     60 
     61 	if len(splitted) < 3 {
     62 		return errors.New("Invalid %meta entry")
     63 	}
     64 
     65 	metaCommand := splitted[0]
     66 	metaObjId := splitted[1]
     67 	metaValues := strings.Join(splitted[2:], "")
     68 
     69 	metaValues = space.ReplaceAllString(metaValues, "")
     70 
     71 	pairs := strings.Split(metaValues, ",")
     72 
     73 	if bi.debug {
     74 		fmt.Printf("Meta command: %s, object: %s, values: %s\n", metaCommand, metaObjId, metaValues)
     75 	}
     76 
     77 	if len(pairs) == 0 {
     78 		return errors.New("Wrong format for %meta entry: " + metaCommand + " " + metaValues)
     79 	}
     80 
     81 	resultDict := make(map[string]string)
     82 
     83 	// Subsequent pairs are key=value and : is the separator between them. Multiple : are for multivalued keys or value containing :
     84 	for _, pair := range pairs {
     85 		keyVal := strings.Split(pair, ":")
     86 		if len(keyVal) < 2 {
     87 			return errors.New("Wrong format for %meta entry: " + metaCommand + " " + metaValues)
     88 		} else {
     89 			resultDict[keyVal[0]] = strings.Join(keyVal[1:], ":")
     90 		}
     91 	}
     92 	exCheck := false
     93 	switch metaCommand {
     94 	case "bmdef":
     95 		switch metaObjId {
     96 		case "global":
     97 			for key, value := range resultDict {
     98 				if err := bi.filteredMetaAdd(bi.global, key, value, "global"); err != nil {
     99 					return err
    100 				}
    101 			}
    102 		}
    103 	case "cpdef":
    104 		for _, cp := range bi.cps {
    105 			if cp.GetValue() == metaObjId {
    106 				exCheck = true
    107 				for key, value := range resultDict {
    108 					if err := bi.filteredMetaAdd(cp, key, value, metaCommand); err != nil {
    109 						return err
    110 					}
    111 				}
    112 				break
    113 			}
    114 		}
    115 	case "sodef":
    116 		for _, so := range bi.sos {
    117 			if so.GetValue() == metaObjId {
    118 				exCheck = true
    119 				for key, value := range resultDict {
    120 					if err := bi.filteredMetaAdd(so, key, value, metaCommand); err != nil {
    121 						return err
    122 					}
    123 				}
    124 				break
    125 			}
    126 		}
    127 	case "iodef":
    128 		for _, io := range bi.ios {
    129 			if io.GetValue() == metaObjId {
    130 				exCheck = true
    131 				for key, value := range resultDict {
    132 					if err := bi.filteredMetaAdd(io, key, value, metaCommand); err != nil {
    133 						return err
    134 					}
    135 				}
    136 				break
    137 			}
    138 		}
    139 	case "fidef":
    140 		for _, fi := range bi.fis {
    141 			if fi.GetValue() == metaObjId {
    142 				exCheck = true
    143 				for key, value := range resultDict {
    144 					if err := bi.filteredMetaAdd(fi, key, value, metaCommand); err != nil {
    145 						return err
    146 					}
    147 				}
    148 				break
    149 			}
    150 		}
    151 	case "filinkdef":
    152 		for _, fil := range bi.fiLinks {
    153 			if fil.GetValue() == metaObjId {
    154 				exCheck = true
    155 				for key, value := range resultDict {
    156 					if err := bi.filteredMetaAdd(fil, key, value, metaCommand); err != nil {
    157 						return err
    158 					}
    159 				}
    160 				break
    161 			}
    162 		}
    163 	case "soatt":
    164 	case "ioatt":
    165 	case "filinkatt":
    166 	default:
    167 		return errors.New("Unknown %meta entry: " + metaCommand)
    168 	}
    169 
    170 	if !exCheck {
    171 		newElem := new(bmline.BasmElement)
    172 		newElem.SetValue(metaObjId)
    173 		for key, value := range resultDict {
    174 			if err := bi.filteredMetaAdd(newElem, key, value, metaCommand); err != nil {
    175 				return err
    176 			}
    177 		}
    178 		switch metaCommand {
    179 		case "cpdef":
    180 			bi.cps = append(bi.cps, newElem)
    181 		case "sodef":
    182 			bi.sos = append(bi.sos, newElem)
    183 		case "iodef":
    184 			bi.ios = append(bi.ios, newElem)
    185 		case "fidef":
    186 			bi.fis = append(bi.fis, newElem)
    187 		case "filinkdef":
    188 			bi.fiLinks = append(bi.fiLinks, newElem)
    189 		case "soatt":
    190 			// TODO Include checks for consistent attach
    191 			bi.soAttach = append(bi.soAttach, newElem)
    192 		case "ioatt":
    193 			// TODO Include checks for consistent attach
    194 			bi.ioAttach = append(bi.ioAttach, newElem)
    195 		case "filinkatt":
    196 			// TODO Include checks for consistent attach
    197 			bi.fiLinkAttach = append(bi.fiLinkAttach, newElem)
    198 		}
    199 	}
    200 
    201 	return nil
