     65 declare -A SupportArray

     67 for i in `ls ../procbuilder/op_* | sort`
     68 do
     69 	opname=`basename $i | cut -d_ -f2- | cut -d. -f1`

    143 cat > ./reference/matrix.md << EOF
    144 # Support Matrix
    145 
    146 The following tables show the feature state in term of level of development for each instruction and instruction group.
    147 For each of them the support of the features is shown.
    148 
    149 The features are the following:
    150 | Feature | Description |
    151 | --- | --- |
    152 | hdl | The instruction can be translated to hardware description language |
    153 | asm | The instruction can be assembled by the assembler |
    154 | disasm | The instruction can be disassembled by the disassembler |
    155 | hlasm | The instruction can be assembled by the high-level assembler (Basm) |
    156 | asmmeta | The instruction has metadata for the assembler |
    157 | gosim | The instruction can be simulated in the Go-based simulator |
    158 | hdlsim | The instruction can be simulated in the hardware description language simulator |
    159 | mt | Instruction thread support |
    160 
    161 The possible support values are shown below:
    162 
    163 | Value | Meaning |
    164 | --- | --- |
    165 | ![ok](iconok.png) | The feature is fully implemented |
    166 | ![no](iconno.png) | The feature is not yet implemented |
    167 | ![testing](icontesting.png) | The feature is being tested |
    168 | ![partial](iconpartial.png) | The feature is partially implemented |
    169 | ![notapplicable](iconnotapplicable.png) | The feature is not applicable to the instruction |
