     22 func (bi *BasmInstance) GetLinks(t uint8, name string, index string, dir string) ([]string, error) {
     23 	result := make([]string, 0)
     24 	switch t {
     25 	case IOLINK:
     26 		for _, link := range bi.ioAttach {
     27 			if link.GetMeta("cp") == name && link.GetMeta("index") == index && link.GetMeta("type") == dir {
     28 				result = append(result, link.GetValue())
     29 			}
     30 		}
     31 	case FILINK:
     32 		for _, fi := range bi.fiLinkAttach {
     33 			if fi.GetMeta("fi") == name && fi.GetMeta("index") == index && fi.GetMeta("type") == dir {
     34 				result = append(result, fi.GetValue())
     35 			}
     36 		}
     37 	}
     38 	return result, nil
     39 }

     41 func (bi *BasmInstance) GetEndpoints(t uint8, name string) (in endpoint, out endpoint, err error) {
     42 	switch t {
     43 	case FILINK:
     44 		inok := false
     45 		outok := false
     46 		isExt := false
     47 		in = endpoint{}
     48 		out = endpoint{}
     49 		for i, fi := range bi.fiLinkAttach {
     50 			if fi.GetValue() == name {
     51 				dir := fi.GetMeta("type")
     52 				if fi.GetMeta("fi") == "ext" {
     53 					isExt = true
     54 					if dir == "input" {
     55 						dir = "output"
     56 					} else {
     57 						dir = "input"
     58 					}
     59 				}
     60 				switch dir {
     61 				case "input":
     62 					if inok {
     63 						return in, out, errors.New("Multiple input endpoints found")
     64 					}
     65 					in.name = fi.GetMeta("fi")
     66 					in.index = fi.GetMeta("index")
     67 					in.linkName = fi.GetValue()
     68 					if isExt {
     69 						in.typ = "output"
     70 					} else {
     71 						in.typ = "input"
     72 					}
     73 					in.seq = i
     74 					inok = true
     75 				case "output":
     76 					if outok {
     77 						return in, out, errors.New("Multiple output endpoints found")
     78 					}
     79 					out.name = fi.GetMeta("fi")
     80 					out.index = fi.GetMeta("index")
     81 					out.linkName = fi.GetValue()
     82 					if isExt {
     83 						out.typ = "input"
     84 					} else {
     85 						out.typ = "output"
     86 					}
     87 					out.seq = i
     88 					outok = true
     89 				default:
     90 					err = errors.New("Invalid direction")
     91 					return in, out, err
     92 				}
     93 			}
     94 		}
     95 		if inok && outok {
     96 			return in, out, nil
     97 		} else {
     98 			err = errors.New("Invalid endpoint")
     99 			return in, out, err
    100 		}
    101 	}
    102 	return in, out, errors.New("Invalid link type")
    103 }
