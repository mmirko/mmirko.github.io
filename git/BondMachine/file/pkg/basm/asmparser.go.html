     14 func idiotParser(s string) ([]string, int) {
     15 	// Convert tabs into spaces
     16 	tab := regexp.MustCompile(`\t`)
     17 	st := tab.ReplaceAllString(s, " ")
     18 	// Strip away all duplicates whitspace characters and comments
     19 	comment := regexp.MustCompile(`;.*`)
     20 	//space := regexp.MustCompile(`\s+`)
     21 	//stripped := strings.TrimSpace(space.ReplaceAllString(comment.ReplaceAllString(s, ""), " "))
     22 	stripped := strings.TrimSpace(comment.ReplaceAllString(st, ""))
     23 
     24 	// Splitting the line using spaces
     25 	splitted := strings.Split(stripped, " ")
     26 
     27 	return splitted, len(splitted)
     28 }
