    415 func (bi *BasmInstance) CreateConnectingProcessor(rSize uint8, cp *bmline.BasmElement, procid int, romCode string, romData string, ramCode string, ramData string, execMode string) (*procbuilder.Machine, error) {
    416 	myMachine := new(procbuilder.Machine)
    417 
    418 	myArch := &myMachine.Arch
    419 
    420 	myArch.Rsize = uint8(rSize)
    421 
    422 	myArch.Modes = make([]string, 1)
    423 	myArch.Modes[0] = execMode
    424 
    425 	var resp bmreqs.ReqResponse
    426 
    427 	tabs := "\t\t"
    428 
    429 	if bi.extra != "" {
    430 		tabs = bi.extra
    431 	}
    432 
    433 	// Processing Code sections: CP has to have at least one code section
    434 	// Getting the ROM code requirements
    435 
    436 	if romCode == "" && ramCode == "" {
    437 		return nil, errors.New("no code section specified, neither ROM nor RAM")
    438 	}
    439 
    440 	opCodesROM := make([]string, 0)
    441 
    442 	if romCode != "" {
    443 		resp = bi.rg.Requirement(bmreqs.ReqRequest{Node: "/code:romtexts/sections:" + romCode, Name: "opcodes", Op: bmreqs.OpGet})
    444 		if resp.Error != nil {
    445 			return nil, resp.Error
    446 		}
    447 		opCodesROM = strings.Split(resp.Value, ",")
    448 	}
    449 
    450 	opCodesRAM := make([]string, 0)
    451 
    452 	// Getting the RAM code requirements
    453 	if ramCode != "" {
    454 		resp = bi.rg.Requirement(bmreqs.ReqRequest{Node: "/code:ramtexts/sections:" + ramCode, Name: "opcodes", Op: bmreqs.OpGet})
    455 		if resp.Error != nil {
    456 			return nil, resp.Error
    457 		}
    458 		opCodesRAM = strings.Split(resp.Value, ",")
    459 	}
    460 	// The final list of opCodes is the union of the two lists
    461 	opCodes := make([]procbuilder.Opcode, 0)
    462 
    463 outer:
    464 	for _, op := range procbuilder.Allopcodes {
    465 		for _, opn := range opCodesROM {
    466 			if opn == op.Op_get_name() {
    467 				opCodes = append(opCodes, op)
    468 				continue outer
    469 			}
    470 		}
    471 		for _, opn := range opCodesRAM {
    472 			if opn == op.Op_get_name() {
    473 				opCodes = append(opCodes, op)
    474 				continue outer
    475 			}
    476 		}
    477 	}
    478 
    479 	sort.Sort(procbuilder.ByName(opCodes))
    480 
    481 	myArch.Op = opCodes
    482 
    483 	threads := 0
    484 	if cp.GetMeta("threaded") != "" {
    485 		if value, err := strconv.Atoi(cp.GetMeta("threaded")); err == nil {
    486 			threads = value
    487 		} else {
    488 			return nil, err
    489 		}
    490 	}
    491 
    492 	myArch.Threaded = threads
    493 
    494 	// TODO: check how it is used and if it is needed, eventually remove or substitute with the merge of the two lists
    495 	bi.rg.Clone("/code:romtexts/sections:"+romCode, "/bm:cps/id:"+strconv.Itoa(procid))
    496 
    497 	regs := make([]string, 0)
    498 
    499 	// Getting the registers requirements on the ROM code
    500 	if romCode != "" {
    501 		resp = bi.rg.Requirement(bmreqs.ReqRequest{Node: "/code:romtexts/sections:" + romCode, Name: "registers", Op: bmreqs.OpGet})
    502 		if resp.Error == nil {
    503 			regs = strings.Split(resp.Value, ",")
    504 		}
    505 	}
    506 
    507 	// Getting the registers requirements on the RAM code, appending to the previous list
    508 	if ramCode != "" {
    509 		resp = bi.rg.Requirement(bmreqs.ReqRequest{Node: "/code:ramtexts/sections:" + ramCode, Name: "registers", Op: bmreqs.OpGet})
    510 		if resp.Error == nil {
    511 			for _, reg := range strings.Split(resp.Value, ",") {
    512 				if !stringInSlice(reg, regs) {
    513 					regs = append(regs, reg)
    514 				}
    515 			}
    516 		}
    517 	}
    518 
    519 	if len(regs) == 0 {
    520 		return nil, errors.New("no registers found on ROM/RAM code")
    521 	}
    522 
    523 	// Sorting the registers list (ordering using the compareStrings function)
    524 	sort.Slice(regs, func(i, j int) bool {
    525 		return compareStrings(regs[i], regs[j])
    526 	})
    527 
    528 	// Getting the last register in the list
    529 	lastReg := regs[len(regs)-1]
    530 
    531 	// Getting the register number
    532 	regNum, _ := strconv.Atoi(lastReg[1:])
    533 
    534 	// To store up to the last register, we need regNum+1 registers
    535 	myArch.R = uint8(Needed_bits(regNum + 1))
    536 
    537 	// If the length of the register list is different from the number of registers, emit a warning
    538 	if len(regs) != regNum+1 {
    539 		bi.Warning("Register list is not complete, some registers are missing. This is not an error provided you know what you are doing.")
    540 	}
    541 
    542 	ins := make([]string, 0)
    543 
    544 	// Getting the Input requirements on the ROM code
    545 	if romCode != "" {
    546 		resp = bi.rg.Requirement(bmreqs.ReqRequest{Node: "/code:romtexts/sections:" + romCode, Name: "inputs", Op: bmreqs.OpGet})
    547 		if resp.Error == nil {
    548 			ins = strings.Split(resp.Value, ",")
    549 		}
    550 	}
    551 	// Getting the Input requirements on the RAM code, appending to the previous list
    552 	if ramCode != "" {
    553 		resp = bi.rg.Requirement(bmreqs.ReqRequest{Node: "/code:ramtexts/sections:" + ramCode, Name: "inputs", Op: bmreqs.OpGet})
    554 		if resp.Error == nil {
    555 			for _, in := range strings.Split(resp.Value, ",") {
    556 				if !stringInSlice(in, ins) {
    557 					ins = append(ins, in)
    558 				}
    559 			}
    560 		}
    561 	}
    562 
    563 	if len(ins) == 0 {
    564 		fmt.Println(tabs + " - " + purple("warning:") + " No inputs found on ROM/RAM code, assuming 0")
    565 		myArch.N = uint8(0)
    566 	} else {
    567 		// Sorting the inputs list (ordering using the compareStrings function)
    568 		sort.Slice(ins, func(i, j int) bool {
    569 			return compareStrings(ins[i], ins[j])
    570 		})
    571 
    572 		// Getting the last input in the list
    573 		lastIn := ins[len(ins)-1]
    574 
    575 		// Getting the input number
    576 		inNum, _ := strconv.Atoi(lastIn[1:])
    577 		// To store up to the last input, we need inNum+1 inputs
    578 		myArch.N = uint8(inNum + 1)
    579 
    580 		if len(ins) != inNum+1 {
    581 			fmt.Println(tabs + " - " + purple("warning:") + "Input list is not complete, some inputs are missing. This is not an error, but you are wasting resources.")
    582 		}
    583 	}
    584 
    585 	outs := make([]string, 0)
    586 
    587 	// Getting the Output requirements on the ROM code
    588 	if romCode != "" {
    589 		resp = bi.rg.Requirement(bmreqs.ReqRequest{Node: "/code:romtexts/sections:" + romCode, Name: "outputs", Op: bmreqs.OpGet})

    592 		}
    593 	}
    594 
    595 	// Getting the Output requirements on the RAM code, appending to the previous list

    598 		if resp.Error == nil {
    599 			for _, out := range strings.Split(resp.Value, ",") {
    600 				if !stringInSlice(out, outs) {
    601 					outs = append(outs, out)
    602 				}
    603 			}
    604 		}
    605 	}
    606 
    607 	if len(outs) == 0 {
    608 		fmt.Println(tabs + " - " + purple("warning:") + " No outputs found on ROM/RAM code, assuming 0")
    609 		myArch.M = uint8(0)
    610 	} else {
    611 		// Sorting the outputs list (ordering using the compareStrings function)
    612 		sort.Slice(outs, func(i, j int) bool {
    613 			return compareStrings(outs[i], outs[j])
    614 		})
    615 
    616 		// Getting the last output in the list
    617 		lastOut := outs[len(outs)-1]
    618 
    619 		// Getting the output number
    620 		outNum, _ := strconv.Atoi(lastOut[1:])
    621 		// To store up to the last output, we need outNum+1 outputs
    622 		myArch.M = uint8(outNum + 1)
    623 
    624 		if len(outs) != outNum+1 {
    625 			fmt.Println(tabs + " - " + purple("warning:") + "Output list is not complete, some outputs are missing. This is not an error, but you are wasting resources.")
    626 		}
    627 	}
    628 
    629 	// Here start the mess with the RAM/ROM size, word size, etc.
    630 	// Compute the word size and the RAM/ROM size, check the eventual cp options
    631 
    632 	// This is the sequence of check in order of priority (valid both for ROM and RAM):
    633 	// - Check if there is a cp option for the word size and use it overriding the other options. In the case of missing resources, the assembler will fail
    634 	// - TODO: Check requirements
    635 	//   - TODO From the direct opcodes
    636 	//   - TODO From the indirect opcodes

    638 
    639 	romCodeContrib := 0

    646 			}
    647 		} else {
    648 			return nil, err
    649 		}
    650 	} else if romCode != "" {
    651 		romCodeContrib = len(bi.sections[romCode].sectionBody.Lines)
    652 		myArch.O = uint8(Needed_bits(romCodeContrib))
    653 		if bi.debug {
    654 			fmt.Println(tabs + " - " + green("romsize (pre-data): ") + yellow(strconv.Itoa(Needed_bits(len(bi.sections[romCode].sectionBody.Lines)))))
    655 		}
    656 	} else {
    657 		romCodeContrib = 0
    658 		myArch.O = uint8(0)
    659 		if bi.debug {
    660 			fmt.Println(tabs + " - " + green("romsize (pre-data): ") + yellow("0"))
    661 		}
    662 	}
    663 
    664 	ramCodeContrib := 0
    665 
    666 	if cp.GetMeta("ramsize") != "" {
    667 		if val, err := strconv.Atoi(cp.GetMeta("ramsize")); err == nil {
    668 			ramCodeContrib = 2 ^ val
    669 			myArch.L = uint8(val)
    670 			if bi.debug {
    671 				fmt.Println(tabs + " - " + green("ramsize (cp config): ") + yellow(cp.GetMeta("ramsize")))
    672 			}
    673 		} else {
    674 			return nil, err
    675 		}
    676 
    677 	} else if ramCode != "" {
    678 		ramCodeContrib = len(bi.sections[ramCode].sectionBody.Lines)
    679 		myArch.L = uint8(Needed_bits(ramCodeContrib))
    680 		if bi.debug {
    681 			fmt.Println(tabs + " - " + green("ramsize (pre-data): ") + yellow(strconv.Itoa(Needed_bits(len(bi.sections[ramCode].sectionBody.Lines)))))
    682 		}
    683 	} else {
    684 		myArch.L = uint8(0)
    685 		ramCodeContrib = 0
    686 		if bi.debug {
    687 			fmt.Println(tabs + " - " + green("ramsize (pre-data): ") + yellow("0"))
    688 		}
    689 	}
    690 
    691 	// The shared constrains will be populated later from the basm metadata
    692 	myArch.Shared_constraints = ""
    693 
    694 	// This comports that program will be assembled later, after the bondmachine is created. Eventually, this will be done
    695 	// prog := ""
    696 
    697 	// for _, line := range bi.sections[romCode].sectionBody.Lines {
    698 	// 	prog += line.Operation.GetValue()
    699 	// 	for _, arg := range line.Elements {
    700 	// 		prog += " " + arg.GetValue()
    701 	// 	}
    702 	// 	prog += "\n"
    703 	// }
    704 
    705 	// if prog, err := myArch.Assembler([]byte(prog)); err == nil {
    706 	// 	myMachine.Program = prog
    707 	// } else {
    708 	// 	return nil, err
    709 	// }
    710 
    711 	// If there is a data section, we need to add it to the machine and update the myArch.O field prior to assembling
    712 	if romData != "" {
    713 		wordSize := myMachine.Max_word()
    714 		// fmt.Println("Word size: ", wordSize)
    715 		if wordSize < 8 {
    716 			return nil, errors.New("word size is too small")
    717 		}
    718 
    719 		data := make([]string, 0)
    720 
    721 		for _, line := range bi.sections[romData].sectionBody.Lines {
    722 			for _, arg := range line.Elements {
    723 				hexVal := arg.GetValue()
    724 				if n, err := bmnumbers.ImportString(hexVal); err == nil {
    725 					nS, _ := n.ExportBinaryNBits(int(wordSize))

    728 					return nil, err

    731 		}
    732 
    733 		myArch.O = uint8(Needed_bits(romCodeContrib + len(data)))
    734 	}
    735 
    736 	if ramData != "" {
    737 		rSize := myArch.Rsize
    738 		if rSize < 8 {
    739 			return nil, errors.New("register size is too small")
    740 		}
    741 
    742 		data := make([]string, 0)
    743 
    744 		for _, line := range bi.sections[ramData].sectionBody.Lines {
    745 			for _, arg := range line.Elements {

    750 				} else {
    751 					return nil, err
    752 				}
    753 			}
    754 		}
    755 
    756 		myArch.L = uint8(Needed_bits(ramCodeContrib + len(data)))
    757 	}
    758 
    759 	return myMachine, nil
    760 }

    771 func (bi *BasmInstance) CodeChoice(rSize uint8, i int, sh string) error {
    772 	if i >= len(bi.cps) || i < 0 {
    773 		return errors.New("index out of range")
    774 	}
    775 
    776 	cp := bi.cps[i]
    777 
    778 	if cp.GetMeta("romalternatives") != "" || cp.GetMeta("ramalternatives") != "" {
    779 		execMode := cp.GetMeta("execmode")
    780 		if execMode == "" {
    781 			execMode = bi.global.GetMeta("defaultexecmode")
    782 		}
    783 		if execMode == "" {
    784 			execMode = "ha"
    785 		}
    786 		romAlts := strings.Split(cp.GetMeta("romalternatives"), ":")
    787 		ramAlts := strings.Split(cp.GetMeta("ramalternatives"), ":")
    788 		if len(romAlts) == 0 {
    789 			romAlts = []string{""}
    790 		}
    791 		if len(ramAlts) == 0 {
    792 			ramAlts = []string{""}
    793 		}
    794 
    795 		romData := cp.GetMeta("romdata")
    796 		ramData := cp.GetMeta("ramdata")
    797 		romDataAlts := strings.Split(cp.GetMeta("romdataalternatives"), ":")
    798 		ramDataAlts := strings.Split(cp.GetMeta("ramdataalternatives"), ":")
    799 
    800 		params := make([]choiceParams, len(romAlts)*len(ramAlts))
    801 
    802 		if bi.debug {
    803 
    804 			fmt.Println("\t\t - " + green("ROM data: ") + yellow(romData))
    805 			fmt.Println("\t\t - " + green("RAM data: ") + yellow(ramData))
    806 			fmt.Println("\t\t - " + green("ROM alternatives: ") + yellow(strings.Join(romAlts, ", ")))
    807 			fmt.Println("\t\t - " + green("RAM alternatives: ") + yellow(strings.Join(ramAlts, ", ")))
    808 			fmt.Println("\t\t - " + green("ROM data alternatives: ") + yellow(strings.Join(romDataAlts, ", ")))
    809 			fmt.Println("\t\t - " + green("RAM data alternatives: ") + yellow(strings.Join(ramDataAlts, ", ")))
    810 		}
    811 
    812 		for ii, romAlt := range romAlts {
    813 			for jj, ramAlt := range ramAlts {
    814 
    815 				if bi.debug {
    816 					fmt.Println("\t\t - " + green("evaluating alternatives: ") + yellow(romAlt) + green(" - ") + yellow(ramAlt))
    817 				}
    818 
    819 				// Find the eventual common name
    820 				groupName := ""
    821 				if strings.HasPrefix(romAlt, "romcode") {
    822 					groupName = romAlt[7:]
    823 				} else if strings.HasPrefix(ramAlt, "ramcode") {
    824 					if groupName != ramAlt[7:] {
    825 						groupName = ""
    826 					}
    827 				} else {
    828 					groupName = ""
    829 				}
    830 
    831 				if bi.debug {
    832 					if groupName == "" {
    833 						fmt.Println("\t\t\t - " + green("no common group name"))
    834 					} else {
    835 						fmt.Println("\t\t\t - " + green("group name: ") + yellow(groupName))
    836 					}
    837 				}
    838 
    839 				if groupName != "" {
    840 					params[ii*len(ramAlts)+jj].groupName = groupName
    841 				}
    842 
    843 				romDataAlt := romData
    844 				ramDataAlt := ramData
    845 
    846 				if groupName != "" && len(romDataAlts) > 0 {
    847 					for _, rda := range romDataAlts {
    848 						if strings.HasSuffix(rda, groupName) {
    849 							romDataAlt = rda
    850 							break
    851 						}
    852 					}
    853 				}
    854 
    855 				if groupName != "" && len(ramDataAlts) > 0 {
    856 					for _, rda := range ramDataAlts {
    857 						if strings.HasSuffix(rda, groupName) {
    858 							ramDataAlt = rda
    859 							break
    860 						}
    861 					}
    862 				}
    863 
    864 				bi.extra = "\t\t\t"
    865 				tempCP, err := bi.CreateConnectingProcessor(rSize, cp, i, romAlt, romDataAlt, ramAlt, ramDataAlt, execMode)
    866 				bi.extra = ""
    867 
    868 				if err != nil {
    869 					return err
    870 				}
    871 
    872 				tempCP.Shared_constraints = sh
    873 				myArch := tempCP.Arch
    874 
    875 				prog := ""
    876 
    877 				romAltContrib := 0
    878 
    879 				params[ii*len(ramAlts)+jj].wordSize = myArch.Max_word()
    880 
    881 				if romAlt != "" {
    882 					for _, line := range bi.sections[romAlt].sectionBody.Lines {
    883 						prog += line.Operation.GetValue()
    884 						for _, arg := range line.Elements {
    885 							prog += " " + arg.GetValue()
    886 						}
    887 						prog += "\n"
    888 					}
    889 
    890 					if prog, err := myArch.Assembler([]byte(prog)); err == nil {
    891 						tempCP.Program = prog
    892 					} else {
    893 						return err
    894 					}
    895 
    896 					romAltContrib = len(bi.sections[romAlt].sectionBody.Lines)
    897 				}
    898 
    899 				if romData != "" {
    900 					wordSize := myArch.Max_word()
    901 
    902 					// fmt.Println("Word size: ", wordSize)
    903 					if wordSize < 8 {
    904 						return errors.New("word size is too small")
    905 					}
    906 
    907 					data := make([]string, 0)
    908 
    909 					for _, line := range bi.sections[romData].sectionBody.Lines {
    910 						for _, arg := range line.Elements {
    911 							hexVal := arg.GetValue()
    912 							if n, err := bmnumbers.ImportString(hexVal); err == nil {
    913 								nS, _ := n.ExportBinaryNBits(int(wordSize))
    914 								data = append(data, nS)
    915 							} else {
    916 								return err
    917 							}
    918 						}
    919 					}
    920 
    921 					myArch.O = uint8(Needed_bits(romAltContrib + len(data)))
    922 					tempCP.Data.Vars = data
    923 
    924 				}
    925 			}
    926 		}
    927 
    928 		// Choice of the sections
    929 		if bi.IsActive(bmconfig.ChooserMinWordSize) {
    930 			if bi.debug {
    931 				fmt.Println("\t\t - " + green("Choosing the code with the minimum word size"))
    932 			}
    933 			var ci string
    934 			var cj string
    935 			minSize := 1024
    936 			for ii, romAlt := range romAlts {
    937 				iiGuessedName := ""
    938 				if strings.HasPrefix(romAlt, "romcode") {
    939 					iiGuessedName = romAlt[7:]
    940 				}
    941 				for jj, ramAlt := range ramAlts {
    942 					jjGuessedName := ""
    943 					if strings.HasPrefix(ramAlt, "ramcode") {
    944 						jjGuessedName = ramAlt[7:]
    945 					}
    946 					if bi.IsActive(bmconfig.ChooserForceSameName) {
    947 						if iiGuessedName != "" && jjGuessedName != "" && iiGuessedName != jjGuessedName {
    948 							continue
    949 						}
    950 					}
    951 					if params[ii*len(ramAlts)+jj].wordSize < minSize {
    952 						minSize = params[ii*len(ramAlts)+jj].wordSize
    953 						ci = romAlt
    954 						cj = ramAlt
    955 					}
    956 				}
    957 			}
    958 
    959 			if bi.debug {
    960 				fmt.Println("\t\t - " + green("Chosen code: ") + yellow(ci) + green(" - ") + yellow(cj))
    961 			}
    962 
    963 			romGroupName := ""
    964 
    965 			if ci != "" {
    966 				cp.SetMeta("romcode", ci)
    967 				if strings.HasPrefix(ci, "romcode") {
    968 					romGroupName = ci[7:]
    969 				}
    970 			}
    971 
    972 			ramGroupName := ""
    973 
    974 			if cj != "" {
    975 				cp.SetMeta("ramcode", cj)
    976 				if strings.HasPrefix(cj, "ramcode") {
    977 					ramGroupName = cj[7:]
    978 				}
    979 			}
    980 
    981 			if cp.GetMeta("romdata") == "" {
    982 				chk := false
    983 				if romGroupName != "" {
    984 					if stringInSlice("romdata"+romGroupName, romDataAlts) {
    985 						cp.SetMeta("romdata", "romdata"+romGroupName)
    986 						cp.RmMeta("romdataalternatives")
    987 						chk = true
    988 					}
    989 				}
    990 				if !chk && ramGroupName != "" {
    991 					if stringInSlice("romdata"+ramGroupName, romDataAlts) {
    992 						cp.SetMeta("romdata", "romdata"+ramGroupName)
    993 						cp.RmMeta("romdataalternatives")
    994 					}
    995 				}
    996 			}
    997 
    998 			if cp.GetMeta("ramdata") == "" {
    999 				chk := false
   1000 				if ramGroupName != "" {
   1001 					if stringInSlice("ramdata"+ramGroupName, ramDataAlts) {
   1002 						cp.SetMeta("ramdata", "ramdata"+ramGroupName)
   1003 						cp.RmMeta("ramdataalternatives")
   1004 						chk = true
   1005 					}
   1006 				}
   1007 				if !chk && romGroupName != "" {
   1008 					if stringInSlice("ramdata"+romGroupName, ramDataAlts) {
   1009 						cp.SetMeta("ramdata", "ramdata"+romGroupName)
   1010 						cp.RmMeta("ramdataalternatives")
   1011 					}
   1012 				}
   1013 			}
   1014 
   1015 			cp.RmMeta("romalternatives")
   1016 			cp.RmMeta("ramalternatives")
   1017 			return nil
   1018 		}
   1019 
   1020 		return errors.New("unable to choose the code among the alternatives, a criteria is needed")
   1021 
   1022 	}
   1023 
   1024 	return nil
   1025 }

   1027 func (bi *BasmInstance) sharedPrecompute() ([]string, error) {
   1028 
   1029 	result := make([]string, len(bi.cps))
   1030 
   1031 	// Insert the Shared Objects into the BM and compose and hash of name,constraint
   1032 	constrains := make(map[string]string)
   1033 	soIndexes := make(map[string]string)
   1034 	for i, so := range bi.sos {
   1035 		if bi.debug {
   1036 			fmt.Print("\t\t" + green("SO: ") + yellow(so.GetValue()))
   1037 		}
   1038 		constraint := so.GetMeta("constraint")
   1039 		if bi.debug {
   1040 			fmt.Println(" - " + green("constraint: ") + yellow(constraint))
   1041 		}
   1042 		constrains[so.GetValue()] = constraint
   1043 		soIndexes[so.GetValue()] = strconv.Itoa(i)
   1044 	}
   1045 
   1046 	// Will keep track of the processed attach
   1047 	soattDone := make([]bool, len(bi.soAttach))
   1048 
   1049 	// Process every CP
   1050 	for i, cp := range bi.cps {
   1051 		cpName := cp.GetValue()
   1052 		expectedIndex := 0
   1053 		expectedIndexS := "0"
   1054 		cpConstraints := make([]string, 0)
   1055 		for {
   1056 			indexFound := false
   1057 			// Process every SO attach searching for the couple CP index
   1058 			for j, soatt := range bi.soAttach {
   1059 				if soatt.GetMeta("cp") == cpName && soatt.GetMeta("index") == expectedIndexS {
   1060 					endpoints := make([]string, 2)
   1061 					endpoints[0] = strconv.Itoa(i)
   1062 					endpoints[1] = soIndexes[soatt.GetValue()]
   1063 					indexFound = true
   1064 					soattDone[j] = true
   1065 					cpConstraints = append(cpConstraints, constrains[soatt.GetValue()])
   1066 					break
   1067 				}
   1068 			}
   1069 			if indexFound {
   1070 				expectedIndex += 1
   1071 				expectedIndexS = strconv.Itoa(expectedIndex)
   1072 			} else {
   1073 				break
   1074 			}
   1075 		}
   1076 		// compose the Shared constraint of every processor
   1077 		result[i] = strings.Join(cpConstraints, ",")
   1078 	}
   1079 
   1080 	for _, val := range soattDone {
   1081 		if !val {
   1082 			return nil, errors.New("processor SO index inconsistent")
   1083 		}
   1084 	}
   1085 
   1086 	return result, nil
   1087 }
