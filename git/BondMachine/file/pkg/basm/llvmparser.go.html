     13 func (bi *BasmInstance) LLVMGetLocalID(inst value.Value) (int64, error) {
     14 	switch inst := inst.(type) {
     15 	case *ir.InstAlloca:
     16 		return inst.LocalIdent.LocalID, nil
     17 	case *ir.InstLoad:
     18 		return inst.LocalIdent.LocalID, nil
     19 	case *ir.InstAdd:
     20 		return inst.LocalIdent.LocalID, nil
     21 	case *ir.Param:
     22 		return inst.LocalIdent.LocalID, nil
     23 	}
     24 	return 0, fmt.Errorf("Could not get local ID for %v", inst)
     25 }

     27 func (bi *BasmInstance) ParseAssemblyLLVM(filePath string) error {
     28 
     29 	if bi.debug {
     30 		fmt.Println(purple("Phase 0") + ": " + red("Reading LLVM IR file "+filePath))
     31 	}
     32 
     33 	m, err := asm.ParseFile(filePath)
     34 	if err != nil {
     35 		return err
     36 	}
     37 
     38 	for _, f := range m.Funcs {
     39 		name := f.GlobalIdent.Ident()
     40 		fName := strings.ReplaceAll(name, "@", "")
     41 		if bi.debug {
     42 			fmt.Println("\t" + red("Adding LLVM Function "+name+" as fragment "+fName))
     43 		}
     44 
     45 		assocMap := make(map[int64]string)
     46 
     47 		newFragment := new(BasmFragment)
     48 		newFragment.fragmentName = fName
     49 		newFragment.fragmentBody = new(bmline.BasmBody)
     50 		newFragment.fragmentBody.Lines = make([]*bmline.BasmLine, 0)
     51 
     52 		resin := make([]string, len(f.Params))
     53 
     54 		for i, param := range f.Params {
     55 			assocMap[param.LocalID] = "r" + fmt.Sprint(param.LocalID)
     56 			resin[i] = "r" + fmt.Sprint(param.LocalID)
     57 		}
     58 
     59 		newFragment.fragmentBody.BasmMeta = newFragment.fragmentBody.BasmMeta.SetMeta("template", "false")
     60 		newFragment.fragmentBody.BasmMeta = newFragment.fragmentBody.BasmMeta.SetMeta("llvm", "true")
     61 		if len(resin) > 0 {
     62 			newFragment.fragmentBody.BasmMeta = newFragment.fragmentBody.BasmMeta.SetMeta("resin", strings.Join(resin, ":"))
     63 		}
     64 
     65 		for _, bb := range f.Blocks {
     66 			for _, inst := range bb.Insts {
     67 				switch inst := inst.(type) {
     68 				case *ir.InstAlloca:
     69 					dest := "r" + fmt.Sprint(inst.LocalIdent.LocalID)
     70 
     71 					newLine := new(bmline.BasmLine)
     72 					newLine.Operation = new(bmline.BasmElement)
     73 					newLine.Operation.SetValue("clr")
     74 					newLine.Elements = make([]*bmline.BasmElement, 1)
     75 					newLine.Elements[0] = new(bmline.BasmElement)
     76 					newLine.Elements[0].SetValue(dest)
     77 
     78 					newFragment.fragmentBody.Lines = append(newFragment.fragmentBody.Lines, newLine)
     79 
     80 				case *ir.InstLoad:
     81 
     82 					dest := "r" + fmt.Sprint(inst.LocalIdent.LocalID)
     83 					srcP := inst.Src
     84 					id, err := bi.LLVMGetLocalID(srcP)
     85 					if err != nil {
     86 						return err
     87 					}
     88 					src := "r" + fmt.Sprint(id)
     89 					newLine := new(bmline.BasmLine)
     90 					newLine.Operation = new(bmline.BasmElement)
     91 					newLine.Operation.SetValue("mov")
     92 					newLine.Elements = make([]*bmline.BasmElement, 2)
     93 					newLine.Elements[0] = new(bmline.BasmElement)
     94 					newLine.Elements[0].SetValue(dest)
     95 					newLine.Elements[1] = new(bmline.BasmElement)
     96 					newLine.Elements[1].SetValue(src)
     97 
     98 					newFragment.fragmentBody.Lines = append(newFragment.fragmentBody.Lines, newLine)
     99 
    100 				case *ir.InstStore:
    101 					srcP := inst.Src
    102 					srcId, err := bi.LLVMGetLocalID(srcP)
    103 					if err != nil {
    104 						return err
    105 					}
    106 					src := "r" + fmt.Sprint(srcId)
    107 					destP := inst.Dst
    108 					destId, err := bi.LLVMGetLocalID(destP)
    109 					if err != nil {
    110 						return err
    111 					}
    112 					dest := "r" + fmt.Sprint(destId)
    113 					newLine := new(bmline.BasmLine)
    114 					newLine.Operation = new(bmline.BasmElement)
    115 					newLine.Operation.SetValue("mov")
    116 					newLine.Elements = make([]*bmline.BasmElement, 2)
    117 					newLine.Elements[0] = new(bmline.BasmElement)
    118 					newLine.Elements[0].SetValue(dest)
    119 					newLine.Elements[1] = new(bmline.BasmElement)
    120 					newLine.Elements[1].SetValue(src)
    121 
    122 					newFragment.fragmentBody.Lines = append(newFragment.fragmentBody.Lines, newLine)
    123 
    124 				case *ir.InstAdd:
    125 					var dest string
    126 					src1P := inst.X
    127 					src1Id, err := bi.LLVMGetLocalID(src1P)
    128 					if err != nil {
    129 						return err
    130 					}
    131 					src1 := "r" + fmt.Sprint(src1Id)
    132 					src2P := inst.Y
    133 					src2Id, err := bi.LLVMGetLocalID(src2P)
    134 					if err != nil {
    135 						return err
    136 					}
    137 					src2 := "r" + fmt.Sprint(src2Id)
    138 					destP := inst.LocalIdent
    139 					dest = "r" + fmt.Sprint(destP.LocalID)
    140 
    141 					newLine1 := new(bmline.BasmLine)
    142 					newLine1.Operation = new(bmline.BasmElement)
    143 					newLine1.Operation.SetValue("mov")
    144 					newLine1.Elements = make([]*bmline.BasmElement, 2)
    145 					newLine1.Elements[0] = new(bmline.BasmElement)
    146 					newLine1.Elements[0].SetValue(dest)
    147 					newLine1.Elements[1] = new(bmline.BasmElement)
    148 					newLine1.Elements[1].SetValue(src1)
    149 
    150 					newFragment.fragmentBody.Lines = append(newFragment.fragmentBody.Lines, newLine1)
    151 
    152 					newLine := new(bmline.BasmLine)
    153 					newLine.Operation = new(bmline.BasmElement)
    154 					newLine.Operation.SetValue("add")
    155 					newLine.Elements = make([]*bmline.BasmElement, 2)
    156 					newLine.Elements[0] = new(bmline.BasmElement)
    157 					newLine.Elements[0].SetValue(dest)
    158 					newLine.Elements[1] = new(bmline.BasmElement)
    159 					newLine.Elements[1].SetValue(src2)
    160 
    161 					newFragment.fragmentBody.Lines = append(newFragment.fragmentBody.Lines, newLine)
    162 
    163 				default:
    164 					return fmt.Errorf("unknown inst type %T", inst)
    165 
    166 				}
    167 
    168 				var ret string
    169 				retP := bb.Term
    170 				switch retP := retP.(type) {
    171 				case *ir.TermRet:
    172 					X := retP.X
    173 					XId, err := bi.LLVMGetLocalID(X)
    174 					if err != nil {
    175 						return err
    176 					}
    177 					ret = "r" + fmt.Sprint(XId)
    178 				}
    179 				newFragment.fragmentBody.BasmMeta = newFragment.fragmentBody.BasmMeta.SetMeta("resout", ret)
    180 
    181 			}
    182 		}
    183 		//pretty.Println(f)
    184 
    185 		bi.fragments[fName] = newFragment
    186 	}
    187 
    188 	return nil
