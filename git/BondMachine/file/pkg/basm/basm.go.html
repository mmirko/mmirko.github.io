     91 	bi.bm = bm

     97 	bi.isWithinFragment = ""

    101 	bi.fragments = make(map[string]*BasmFragment)
    102 	bi.chunks = make(map[string]*BasmChunk)
    103 	bi.symbols = make(map[string]int64)
    104 	bi.passes = passTemplateResolver |
    105 		passDynamicalInstructions |
    106 		passSymbolTagger1 |
    107 		passSymbolTagger2 |
    108 		passSymbolTagger3 |
    109 		passDataSections2Bytes |
    110 		passMetadataInfer1 |
    111 		passMetadataInfer2 |
    112 		passMetadataInfer3 |
    113 		passEntryPoints |
    114 		passSymbolsResolver |
    115 		passMatcherResolver |
    116 		// passFragmentAnalyzer |
    117 		passFragmentPruner |
    118 		passFragmentComposer |
    119 		passFragmentOptimizer1 |
    120 		passMemComposer |
    121 		passSectionCleaner |
    122 		passCallResolver |
    123 		passMacroResolver |
    124 		// passTemplateFinalizer |
    125 		0
    126 
    127 	bi.matchers = make([]*bmline.BasmLine, 0)
    128 	bi.dynMatchers = make([]*bmline.BasmLine, 0)
    129 	bi.matchersOps = make([]procbuilder.Opcode, 0)
    130 	bi.dynMatcherOps = make([]procbuilder.DynamicInstruction, 0)
    131 
    132 	bi.rg = bmreqs.NewReqRoot()
    133 
    134 	bi.cps = make([]*bmline.BasmElement, 0)
    135 	bi.sos = make([]*bmline.BasmElement, 0)
    136 	bi.ios = make([]*bmline.BasmElement, 0)
    137 	bi.ioAttach = make([]*bmline.BasmElement, 0)
    138 	bi.soAttach = make([]*bmline.BasmElement, 0)
    139 	bi.fis = make([]*bmline.BasmElement, 0)
    140 	bi.fiLinks = make([]*bmline.BasmElement, 0)
    141 	bi.fiLinkAttach = make([]*bmline.BasmElement, 0)
    142 	bi.global = new(bmline.BasmElement)
    143 	bi.global.SetValue("global")
    144 
    145 	if bi.debug {
    146 		fmt.Println(purple("Init") + ":")
    147 	}
    148 	for _, op := range procbuilder.Allopcodes {
    149 		if bi.debug {
    150 			fmt.Println(purple("\tExamining opcode: ") + op.Op_get_name())
    151 		}
    152 		for _, line := range op.HLAssemblerMatch(nil) {
    153 			if mt, err := bmline.Text2BasmLine(line); err == nil {
    154 				bi.matchers = append(bi.matchers, mt)
    155 				bi.matchersOps = append(bi.matchersOps, op)
    156 			} else {
    157 				bi.Warning(err)
    158 			}
    159 		}
    160 	}
    161 
    162 	for _, dyn := range procbuilder.AllDynamicalInstructions {
    163 		if bi.debug {
    164 			fmt.Println(purple("\tExamining dyn opcode: ") + dyn.GetName())
    165 		}
    166 		for _, line := range dyn.HLAssemblerGeneratorMatch(bi.BmConfig) {
    167 			if mt, err := bmline.Text2BasmLine(line); err == nil {
    168 				bi.dynMatchers = append(bi.dynMatchers, mt)
    169 				bi.dynMatcherOps = append(bi.dynMatcherOps, dyn)
    170 			} else {
    171 				bi.Warning(err)
    172 			}
    173 		}
    174 	}
    175 }

    195 // Run the assembler
    196 func (bi *BasmInstance) RunAssembler() error {
    197 	step := uint64(1)
    198 	names := getPassFunctionName()
    199 
    200 	bi.rg.Requirement(bmreqs.ReqRequest{Node: "/", T: bmreqs.ObjectSet, Name: "code", Value: "romtexts", Op: bmreqs.OpAdd})
    201 	bi.rg.Requirement(bmreqs.ReqRequest{Node: "/", T: bmreqs.ObjectSet, Name: "code", Value: "ramtexts", Op: bmreqs.OpAdd})
    202 	bi.rg.Requirement(bmreqs.ReqRequest{Node: "/", T: bmreqs.ObjectSet, Name: "code", Value: "romdatas", Op: bmreqs.OpAdd})
    203 	bi.rg.Requirement(bmreqs.ReqRequest{Node: "/", T: bmreqs.ObjectSet, Name: "code", Value: "ramdatas", Op: bmreqs.OpAdd})
    204 
    205 	if bi.debug {
    206 		fmt.Println(purple("Pre phase 1 ") + bi.String())
    207 	}
    208 
    209 	passes := getPassFunction()
    210 	for i := 0; i < 64; i++ {
    211 		if bi.ActivePass(step) {
    212 			if bi.debug {
    213 				fmt.Println(purple("Phase "+strconv.Itoa(i+1)) + ": " + red(names[step], " started"))
    214 			}
    215 			currentPass := passes[step]
    216 			if err := currentPass(bi); err != nil {
    217 				return err
    218 			} else {
    219 				if bi.debug {
    220 					fmt.Println(purple("Phase "+strconv.Itoa(i+1)) + ": " + red(names[step], " completed"))
    221 					fmt.Println(purple("Post phase "+strconv.Itoa(i+1)) + " " + bi.String())
    222 				}
    223 			}
    224 		} else {
    225 			if bi.debug {
    226 				fmt.Println(purple("Phase "+strconv.Itoa(i+1)) + ": " + red(names[step], " is disabled"))
    227 			}
    228 		}
    229 		if step == LAST_PASS {
    230 			break
    231 		}
    232 		step = step << 1
    233 	}
    234 
    235 	if bi.debug {
    236 		fmt.Println(bi.rg.Requirement(bmreqs.ReqRequest{Node: "/", Op: bmreqs.OpDump}))
    237 	}
    238 
    239 	return nil
    240 }

    264 func (m *BasmSection) writeText() string {
    265 	result := ""
    266 	for _, line := range m.sectionBody.Lines {
    267 		if line.GetMeta("symbol") != "" {
    268 			result += line.GetMeta("symbol") + ":\n"
    269 		}
    270 		result += line.Operation.GetValue() + " "
    271 		for _, element := range line.Elements {
    272 			result += element.GetValue()
    273 			if element != line.Elements[len(line.Elements)-1] {
    274 				result += ", "
    275 			}
    276 		}
    277 		result += "\n"
    278 	}
    279 	return result
    280 }
