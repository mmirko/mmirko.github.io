     19 func createBasicTemplateData() *templateData {
     20 	result := new(templateData)
     21 	result.Params = make(map[string]string)
     22 	funcMap := template.FuncMap{
     23 		"inc": func(i int) int {
     24 			return i + 1
     25 		},
     26 		"incs": func(i string) string {
     27 			j, _ := strconv.Atoi(i)
     28 			return strconv.Itoa(j + 1)
     29 		},
     30 		"dec": func(i int) int {
     31 			return i - 1
     32 		},
     33 		"decs": func(i string) string {
     34 			j, _ := strconv.Atoi(i)
     35 			return strconv.Itoa(j - 1)
     36 		},
     37 		"add": func(i, j int) int {
     38 			return i + j
     39 		},
     40 		"adds": func(i, j string) string {
     41 			k, _ := strconv.Atoi(i)
     42 			l, _ := strconv.Atoi(j)
     43 			return strconv.Itoa(k + l)
     44 		},
     45 		"intRange": func(startS, endS string) []int {
     46 			start, _ := strconv.Atoi(startS)
     47 			end, _ := strconv.Atoi(endS)
     48 
     49 			n := end - start
     50 			result := make([]int, n)
     51 			for i := 0; i < n; i++ {
     52 				result[i] = start + i
     53 			}
     54 			return result
     55 		},
     56 		"atoi": func(s string) int {
     57 			i, _ := strconv.Atoi(s)
     58 			return i
     59 		},
     60 	}
     61 
     62 	result.funcMap = funcMap
     63 	return result
     64 }

     66 func (bi *BasmInstance) templateAutoMark() error {
     67 	// Set the template meta to true for all the sections and fragments that contains the template mark
     68 	for _, section := range bi.sections {
     69 		if section.isTemplate() {
     70 			section.sectionBody.BasmMeta = section.sectionBody.SetMeta("template", "true")
     71 		} else {
     72 			section.sectionBody.BasmMeta = section.sectionBody.SetMeta("template", "false")
     73 		}
     74 	}
     75 
     76 	for _, fragment := range bi.fragments {
     77 		if fragment.isTemplate() {
     78 			fragment.fragmentBody.BasmMeta = fragment.fragmentBody.SetMeta("template", "true")
     79 		} else {
     80 			fragment.fragmentBody.BasmMeta = fragment.fragmentBody.SetMeta("template", "false")
     81 		}
     82 	}
     83 	return nil
     84 }

     86 func isTemplate(s string) bool {
     87 	pattern := `{{[^{}]*}}`
     88 	regex := regexp.MustCompile(pattern)
     89 	matches := regex.FindAllString(s, -1)
     90 	return len(matches) > 0
     91 }

    101 func applyTemplate(e *bmline.BasmElement, params map[string]string) error {
    102 	// Search for default values
    103 	for k, v := range params {
    104 		if strings.HasPrefix(k, "default_") {
    105 			key := strings.TrimPrefix(k, "default_")
    106 			if _, ok := params[key]; !ok {
    107 				params[key] = v
    108 			}
    109 		}
    110 	}
    111 
    112 	td := createBasicTemplateData()
    113 
    114 	for key, value := range params {
    115 		td.Params[key] = value
    116 	}
    117 
    118 	var f bytes.Buffer
    119 
    120 	t, err := template.New("template").Funcs(td.funcMap).Parse(e.GetValue())
    121 	if err != nil {
    122 		return err
    123 	}
    124 
    125 	err = t.Execute(&f, *td)
    126 	if err != nil {
    127 		return err
    128 	}
    129 
    130 	e.SetValue(f.String())
    131 
    132 	// fmt.Println(params)
    133 
    134 	return nil
    135 }
