    102 func dbDataConverter(line string) ([]string, error) {
    103 	trimmed := strings.TrimSpace(line)
    104 
    105 	result := make([]string, 0)
    106 
    107 	withinString := false
    108 	curElem := ""
    109 	curNumeric := true
    110 
    111 	for i := 0; i < len(trimmed); i++ {
    112 		ch := trimmed[i]
    113 		switch ch {
    114 		case '"':
    115 			withinString = !withinString
    116 			if withinString {
    117 				curNumeric = false
    118 			}
    119 		case ',':
    120 			if withinString {
    121 				curElem += string(ch)
    122 			} else {
    123 				if curElem != "" {
    124 					if converted, err := object2Bytes(curElem, curNumeric); err != nil {
    125 						return []string{}, err
    126 					} else {
    127 						result = append(result, converted...)
    128 						curElem = ""
    129 						curNumeric = true
    130 					}
    131 				}
    132 			}
    133 		default:
    134 			if withinString {
    135 				curElem += string(ch)
    136 			} else {
    137 				curElem += strings.TrimSpace(string(ch))
    138 			}
    139 		}
    140 	}
    141 
    142 	if withinString {
    143 		return []string{}, errors.New("a string has not been closed")
    144 	}
    145 
    146 	if converted, err := object2Bytes(curElem, curNumeric); err != nil {
    147 		return []string{}, err
    148 	} else {
    149 		result = append(result, converted...)
    150 	}
    151 
    152 	return result, nil
    153 }

    189 func compareStrings(str1, str2 string) bool {
    190 	// Extract the numbers from the strings
    191 	num1, err1 := extractNumber(str1)
    192 	num2, err2 := extractNumber(str2)
    193 
    194 	// If both strings contain numbers, compare them as numbers
    195 	if err1 == nil && err2 == nil {
    196 		return num1 < num2
    197 	}
    198 
    199 	// Otherwise, compare them as strings
    200 	return str1 < str2
    201 }

    203 func extractNumber(str string) (int, error) {
    204 	// Find the index of the first digit
    205 	tokens := strings.FieldsFunc(str, func(r rune) bool {
    206 		return !('0' <= r && r <= '9')
    207 	})
    208 
    209 	// If no numeric tokens were found, return an error
    210 	if len(tokens) == 0 {
    211 		return 0, fmt.Errorf("No numeric tokens found in %s", str)
    212 	}
    213 
    214 	// Otherwise, return the last token as an integer
    215 	return strconv.Atoi(tokens[len(tokens)-1])
    216 }
