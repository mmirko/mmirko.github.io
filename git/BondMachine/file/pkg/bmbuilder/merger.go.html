     34 func (e BMEndPoint) String() string {
     35 	switch e.BType {
     36 	case FinalBMInput:
     37 		return fmt.Sprintf("i%d", e.BNum)
     38 	case FinalBMOutput:
     39 		return fmt.Sprintf("o%d", e.BNum)
     40 	case FirstBMInput:
     41 		return fmt.Sprintf("bm1i%d", e.BNum)
     42 	case FirstBMOutput:
     43 		return fmt.Sprintf("bm1o%d", e.BNum)
     44 	case SecondBMInput:
     45 		return fmt.Sprintf("bm2i%d", e.BNum)
     46 	case SecondBMOutput:
     47 		return fmt.Sprintf("bm2o%d", e.BNum)
     48 	}
     49 	return fmt.Sprintf("Unknown(%d)", e.BNum)
     50 }

     76 func followLink(bm1 *bondmachine.Bondmachine, bm2 *bondmachine.Bondmachine, result *bondmachine.Bondmachine, intOutMap chMap, endPoint BMEndPoint) (int, error) {
     77 	// This function finds out the internal output that corresponds to the given endpoint
     78 
     79 	switch endPoint.BType {
     80 	case FinalBMInput:
     81 		for i, bond := range result.Internal_outputs {
     82 			if bond.Map_to == bondmachine.BMINPUT && bond.Res_id == endPoint.BNum {
     83 				return i, nil
     84 			}
     85 		}
     86 	case FirstBMOutput:
     87 		for i, bond := range bm1.Internal_inputs {
     88 			if bond.Map_to == bondmachine.BMOUTPUT && bond.Res_id == endPoint.BNum {
     89 				if bm1.Links[i] == -1 {
     90 					return -1, nil
     91 				}
     92 				return intOutMap.getNewID(1, bm1.Links[i]), nil
     93 			}
     94 		}
     95 	case SecondBMOutput:
     96 		for i, bond := range bm2.Internal_inputs {
     97 			if bond.Map_to == bondmachine.BMOUTPUT && bond.Res_id == endPoint.BNum {
     98 				if bm2.Links[i] == -1 {
     99 					return -1, nil
    100 				}
    101 				return intOutMap.getNewID(2, bm2.Links[i]), nil
    102 			}
    103 		}
    104 	}
    105 	return 0, errors.New("endpoint not Found")
    106 }

    108 func (bld *BMBuilder) BMMerge(bm1 *bondmachine.Bondmachine, bm2 *bondmachine.Bondmachine, l *BMConnections) (*bondmachine.Bondmachine, error) {
    109 	// Merge two bondmachines
    110 	// bm1 and bm2 are the bondmachines to merge
    111 	// Returns the merged bondmachine
    112 
    113 	if bm1.Rsize != bm2.Rsize {
    114 		return nil, errors.New("bondmachines have different register sizes")
    115 	}
    116 
    117 	if bld.debug {
    118 		fmt.Println("Connections:\n", l)
    119 	}
    120 
    121 	result := new(bondmachine.Bondmachine)
    122 	result.Rsize = bm1.Rsize
    123 
    124 	result.Init()
    125 
    126 	mergedInputs := 0
    127 	mergedOutputs := 0
    128 	for _, link := range l.Links {
    129 		if link.E1.BType == FinalBMInput {
    130 			if link.E1.BNum >= mergedInputs {
    131 				mergedInputs = link.E1.BNum + 1
    132 			}
    133 		}
    134 		if link.E1.BType == FinalBMOutput {
    135 			if link.E1.BNum >= mergedOutputs {
    136 				mergedOutputs = link.E1.BNum + 1
    137 			}
    138 		}
    139 
    140 		if link.E2.BType == FinalBMInput {
    141 			if link.E2.BNum >= mergedInputs {
    142 				mergedInputs = link.E2.BNum + 1
    143 			}
    144 		}
    145 		if link.E2.BType == FinalBMOutput {
    146 			if link.E2.BNum >= mergedOutputs {
    147 				mergedOutputs = link.E2.BNum + 1
    148 			}
    149 		}
    150 	}
    151 
    152 	result.Inputs = mergedInputs
    153 	result.Outputs = mergedOutputs
    154 
    155 	if bld.debug {
    156 		fmt.Printf("Inputs of the merged bondmachine: %d\n", mergedInputs)
    157 		fmt.Printf("Outputs of the merged bondmachine: %d\n", mergedOutputs)
    158 	}
    159 
    160 	// Processing domains
    161 	domainsMap := make(chMap)
    162 
    163 	result.Domains = make([]*procbuilder.Machine, len(bm1.Domains)+len(bm2.Domains))
    164 
    165 	for i, dom := range bm1.Domains {
    166 		result.Domains[i] = dom
    167 		domainsMap[i] = bmOrigin{originBM: 1, originID: i}
    168 	}
    169 	for i, dom := range bm2.Domains {
    170 		result.Domains[i+len(bm1.Domains)] = dom
    171 		domainsMap[i+len(bm1.Domains)] = bmOrigin{originBM: 2, originID: i}
    172 	}
    173 
    174 	if bld.debug {
    175 		fmt.Println("Domains map: ", domainsMap)
    176 		fmt.Println("Domains: ", result.Domains)
    177 	}
    178 
    179 	// Processing processors
    180 	procsMap := make(chMap)
    181 
    182 	result.Processors = make([]int, len(bm1.Processors)+len(bm2.Processors))
    183 
    184 	for i, domId := range bm1.Processors {
    185 		result.Processors[i] = domainsMap.getNewID(1, i)
    186 		procsMap[i] = bmOrigin{originBM: 1, originID: domId}
    187 	}
    188 
    189 	for i, domId := range bm2.Processors {
    190 		result.Processors[i+len(bm1.Processors)] = domainsMap.getNewID(2, domId)
    191 		procsMap[i+len(bm1.Processors)] = bmOrigin{originBM: 2, originID: i}
    192 	}
    193 
    194 	if bld.debug {
    195 		fmt.Println("Processors map: ", procsMap)
    196 		fmt.Println("Processors: ", result.Processors)
    197 	}
    198 
    199 	if bld.debug {
    200 		fmt.Println("BM 1 internal inputs: ", bm1.Internal_inputs)
    201 		fmt.Println("BM 1 internal outputs: ", bm1.Internal_outputs)
    202 		fmt.Println("BM 2 internal inputs: ", bm2.Internal_inputs)
    203 		fmt.Println("BM 2 internal outputs: ", bm2.Internal_outputs)
    204 		fmt.Println("BM 1 links: ", bm1.Links)
    205 		fmt.Println("BM 2 links: ", bm2.Links)
    206 	}
    207 
    208 	// Processing shared objects
    209 	soMap := make(chMap)
    210 	result.Shared_objects = make([]bondmachine.Shared_instance, len(bm1.Shared_objects)+len(bm2.Shared_objects))
    211 
    212 	for i, so := range bm1.Shared_objects {
    213 		result.Shared_objects[i] = so
    214 		soMap[i] = bmOrigin{originBM: 1, originID: i}
    215 	}
    216 
    217 	for i, so := range bm2.Shared_objects {
    218 		result.Shared_objects[i+len(bm1.Shared_objects)] = so
    219 		soMap[i+len(bm1.Shared_objects)] = bmOrigin{originBM: 2, originID: i}
    220 	}
    221 
    222 	result.Shared_links = make([]bondmachine.Shared_instance_list, len(result.Processors))
    223 	for i, _ := range result.Processors {
    224 		soList := make(bondmachine.Shared_instance_list, 0)
    225 		bm := procsMap[i].originBM
    226 		origProc := procsMap[i].originID
    227 		switch bm {
    228 		case 1:
    229 			for _, so := range bm1.Shared_links[origProc] {
    230 				soList = append(soList, soMap.getNewID(1, so))
    231 			}
    232 		case 2:
    233 			for _, so := range bm2.Shared_links[origProc] {
    234 				soList = append(soList, soMap.getNewID(2, so))

    236 		}
    237 		result.Shared_links[i] = soList
    238 	}

    241

    244 
    245 	for i := 0; i < mergedInputs; i++ {
    246 		newBond := bondmachine.Bond{Map_to: bondmachine.BMINPUT, Res_id: i, Ext_id: 0}

    260 			intOutMap[no] = bmOrigin{originBM: 1, originID: i}

    262 		}
    263 	}

    265 	for i, bond := range bm2.Internal_outputs {
    266 		switch bond.Map_to {

    268 			if bld.debug {
    269 				fmt.Println("Ignoring internal output ", bond, " from BM 2")

    271 		case bondmachine.CPOUTPUT:

    273 			result.Internal_outputs = append(result.Internal_outputs, newBond)

    281 		fmt.Println("\t", intOutMap)

    285 	intInMap := make(chMap)
    286 	result.Internal_inputs = make([]bondmachine.Bond, mergedOutputs)
    287 
    288 	for i := 0; i < mergedOutputs; i++ {
    289 		newBond := bondmachine.Bond{Map_to: bondmachine.BMOUTPUT, Res_id: i, Ext_id: 0}
    290 		result.Internal_inputs[i] = newBond
    291 	}
    292

    294 	for i, bond := range bm1.Internal_inputs {
    295 		switch bond.Map_to {
    296 		case bondmachine.BMOUTPUT:
    297 			if bld.debug {
    298 				fmt.Println("Ignoring internal input ", bond, " from BM 1")
    299 			}
    300 
    301 		case bondmachine.CPINPUT:
    302 			newBond := bondmachine.Bond{Map_to: bondmachine.CPINPUT, Res_id: procsMap.getNewID(1, bond.Res_id), Ext_id: bond.Ext_id}
    303 			result.Internal_inputs = append(result.Internal_inputs, newBond)

    306 		}
    307 	}
    308 
    309 	for i, bond := range bm2.Internal_inputs {
    310 		switch bond.Map_to {
    311 		case bondmachine.BMOUTPUT:
    312 			if bld.debug {
    313 				fmt.Println("Ignoring internal input ", bond, " from BM 2")
    314 			}
    315 		case bondmachine.CPINPUT:
    316 			newBond := bondmachine.Bond{Map_to: bondmachine.CPINPUT, Res_id: procsMap.getNewID(2, bond.Res_id), Ext_id: bond.Ext_id}
    317 			result.Internal_inputs = append(result.Internal_inputs, newBond)

    321 	}
    322

    332 			fmt.Println("Following internal input: ", bond)
    333 		}
    334 
    335 		switch bond.Map_to {

    337 			for _, link := range l.Links {
    338 				if link.E2.BType == FinalBMOutput && link.E2.BNum == bond.Res_id {

    346 					}
    347 
    348 					break
    349 				}
    350 
    351 				if link.E1.BType == FinalBMOutput && link.E1.BNum == bond.Res_id {
