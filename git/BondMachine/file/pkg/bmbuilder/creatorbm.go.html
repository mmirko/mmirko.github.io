      9 func (bi *BMBuilder) BuildBondMachine() error {
     10 	if bi.debug {
     11 		fmt.Println("\t" + green("BondMachine metadata"))
     12 	}
     13 
     14 	registerSize := bi.global.GetMeta("registersize")
     15 
     16 	if registerSize == "" {
     17 		return errors.New("register size not specified")
     18 	}
     19 
     20 	var rSize uint8
     21 	if size, err := strconv.Atoi(registerSize); err == nil {
     22 		if 0 < size && size < 256 {
     23 			rSize = uint8(size)
     24 		} else {
     25 			return errors.New("wrong value for register size")
     26 		}
     27 	} else {
     28 		return errors.New("register size not valid")
     29 	}
     30 	if bi.debug {
     31 		fmt.Println("\t\t"+green("register size:"), rSize)
     32 	}
     33 
     34 	// Get the main block
     35 	mainBlock := bi.global.GetMeta("main")
     36 
     37 	if mainBlock == "" {
     38 		return errors.New("main block not specified")
     39 	}
     40 
     41 	if block, ok := bi.blocks[mainBlock]; ok {
     42 		if bi.debug {
     43 			fmt.Println("\t\t"+green("main block:"), mainBlock)
     44 		}
     45 
     46 		switch block.blockType {
     47 		case blockSequential:
     48 			return bi.BuildSequentialBlock(block)
     49 		default:
     50 			return errors.New("unknown block type for main block")
     51 		}
     52 
     53 	} else {
     54 		return errors.New("main block not found")
     55 	}
     56 
     57 	return nil
     58 }

     60 func (bi *BMBuilder) BuildSequentialBlock(block *BMBuilderBlock) error {
     61 	if bi.debug {
     62 		fmt.Println("\t" + green("Building Sequential block"))
     63 	}
     64 
     65 	BMNum := len(block.blockBMs)
     66 	if BMNum == 0 {
     67 		return errors.New("no BMs in the block")
     68 	}
     69 
     70 	if BMNum == 1 {
     71 		bi.result = block.blockBMs[0]
     72 		return nil
     73 	}
     74 
     75 	bm1 := block.blockBMs[0]
     76 
     77 	for i := 1; i < BMNum; i++ {
     78 		bm2 := block.blockBMs[i]
     79 		con := block.blockConn[i-1]
     80 
     81 		// The connections are not defined, so we need to create them using the default connector
     82 		if con == nil {
     83 			if bm1.Outputs != bm2.Inputs {
     84 				return errors.New("BM outputs and inputs do not match")
     85 			}
     86 
     87 			con = new(BMConnections)
     88 			con.Links = make([]BMLink, 0)
     89 
     90 			// Set the bm1 inputs as final inputs
     91 			for j := 0; j < bm1.Inputs; j++ {
     92 				e1 := BMEndPoint{BType: FinalBMInput, BNum: j}
     93 				e2 := BMEndPoint{BType: FirstBMInput, BNum: j}
     94 				con.Links = append(con.Links, BMLink{E1: e1, E2: e2})
     95 			}
     96 
     97 			// Set the bm1 to bm2 connections
     98 			for j := 0; j < bm1.Outputs; j++ {
     99 				e1 := BMEndPoint{BType: FirstBMOutput, BNum: j}
    100 				e2 := BMEndPoint{BType: SecondBMInput, BNum: j}
    101 				con.Links = append(con.Links, BMLink{E1: e1, E2: e2})
    102 			}
    103 
    104 			// Set the bm2 outputs as final outputs
    105 			for j := 0; j < bm2.Outputs; j++ {
    106 				e1 := BMEndPoint{BType: SecondBMOutput, BNum: j}
    107 				e2 := BMEndPoint{BType: FinalBMOutput, BNum: j}
    108 				con.Links = append(con.Links, BMLink{E1: e1, E2: e2})
    109 			}
    110 		}
    111 
    112 		// Merge the two bondmachines
    113 		if bm, err := bi.BMMerge(bm1, bm2, con); err == nil {
    114 			bm1 = bm
    115 		} else {
    116 			return err
    117 		}
    118 
    119 	}
    120 
    121 	bi.result = bm1
    122 
    123 	return nil
    124 }
