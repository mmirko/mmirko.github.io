     58 // QasmToBmMatrices converts a QASM file to a list of BmMatrixSquareComplex, the input is a BasmBody with all the metadata and the list of quantum instructions
     59 func (sim *BmQSimulator) QasmToBmMatrices(qasm *bmline.BasmBody) ([]*bmmatrix.BmMatrixSquareComplex, error) {
     60 
     61 	result := make([]*bmmatrix.BmMatrixSquareComplex, 0)
     62 
     63 	// Get the qbits and their names
     64 	qbits := qasm.GetMeta("qbits")
     65 
     66 	if qbits == "" {
     67 		return nil, fmt.Errorf("no qbits defined")
     68 	}
     69 
     70 	for i, qbit := range strings.Split(qbits, ":") {
     71 		if _, ok := sim.qbitsNum[qbit]; ok {
     72 			return nil, fmt.Errorf("qbit %s already defined", qbit)
     73 		} else {
     74 			sim.qbits = append(sim.qbits, qbit)
     75 			sim.qbitsNum[qbit] = i
     76 		}
     77 	}
     78 
     79 	curOp := make([]*bmline.BasmLine, 0)
     80 	curQBits := make(map[int]struct{})
     81 
     82 	for i := 0; i < len(qasm.Lines); {
     83 		line := qasm.Lines[i]
     84 		op := line.Operation.GetValue()
     85 
     86 		// Check if the operation is ready to form a matrix
     87 		nextOp := false
     88 
     89 		// Include the qbits in the operation to the currQbits map
     90 		for _, arg := range line.Elements {
     91 			argName := arg.GetValue()
     92 			// Check if the argument is a qbit, otherwise ignore it
     93 			if qbitN, ok := sim.qbitsNum[argName]; ok {
     94 				if _, ok := curQBits[qbitN]; ok {
     95 					nextOp = true
     96 					break
     97 				} else {
     98 					curQBits[sim.qbitsNum[argName]] = struct{}{}
     99 				}
    100 			}
    101 
    102 		}
    103 
    104 		if op == "nextop" {
    105 			nextOp = true
    106 		}
    107 
    108 		if i == len(qasm.Lines)-1 && !nextOp {
    109 			if sim.debug {
    110 				fmt.Println("\tProcessing line", i, ":", line.String())
    111 			}
    112 			curOp = append(curOp, line)
    113 			i++
    114 			nextOp = true
    115 		}
    116 
    117 		// If the operation is ready to form a matrix, create the matrix
    118 		if nextOp {
    119 
    120 			if sim.debug {
    121 				fmt.Println(red("\tNew operation") + " (ready to form a matrix)")
    122 			}
    123 
    124 			// Create the matrix
    125 			if len(curOp) > 0 {
    126 				// Create the matrix
    127 				if m, err := sim.BmMatrixFromOperation(curOp); err != nil {
    128 					return nil, fmt.Errorf("error creating matrix from operation: %v", err)
    129 				} else {
    130 					if m != nil {
    131 						result = append(result, m)
    132 					}
    133 				}
    134 			}
    135 
    136 			// Reset the operation and the qbits
    137 			curOp = make([]*bmline.BasmLine, 0)
    138 			curQBits = make(map[int]struct{})
    139 		} else {
    140 			if sim.debug {
    141 				fmt.Println("\tProcessing line", i, ":", line.String())
    142 			}
    143 			curOp = append(curOp, line)
    144 			i++
    145 		}
    146 	}
    147 	return result, nil
    148 }

    155 func (sim *BmQSimulator) BmMatrixFromOperation(op []*bmline.BasmLine) (*bmmatrix.BmMatrixSquareComplex, error) {
    156 	// Let prepare the matrix that will be tensor-producted to form the final matrix
    157 	var result *bmmatrix.BmMatrixSquareComplex
    158 
    159 	swaps := make([]swap, 0)
    160 	// loop over the qbits each qbit will have only one (or zero) operation within the op list
    161 	// Every line where the qbits in not in sequence will be reordered swapping the qbits
    162 	// and swapped back after the matrix is created
    163 
    164 	localQBits := make([]string, len(sim.qbits))
    165 	copy(localQBits, sim.qbits)
    166 
    167 	if sim.debug {
    168 		ops := ""
    169 		for _, l := range op {
    170 			ops += l.String() + ", "
    171 		}
    172 		fmt.Println("\t\tBuilding matrix from operation" + ops)
    173 	}
    174 
    175 	for q := 0; q < len(localQBits); q++ {
    176 
    177 		if sim.debug {
    178 			fmt.Println("\t\tAnalyzing qbit number:", q, "name:", localQBits[q])
    179 			fmt.Println("\t\t\tCurrent QBits order:", localQBits)
    180 			fmt.Println("\t\t\tCurrent QBits swaps:", swaps)
    181 		}
    182 
    183 		qbit := localQBits[q]
    184 		// Find the operation for the qbit
    185 		found := false
    186 		fundLine := -1
    187 		for i, line := range op {
    188 			for _, arg := range line.Elements {
    189 				argName := arg.GetValue()
    190 				if argName == qbit {
    191 					found = true
    192 					fundLine = i
    193 					break
    194 				}
    195 			}
    196 			if found {
    197 				break
    198 			}
    199 		}
    200 
    201 		if !found {
    202 			if sim.debug {
    203 				fmt.Println("\t\t\tNo operation for qbit", q, "adding identity matrix")
    204 			}
    205 			// No operation for the qbit, lets add an identity matrix
    206 			// Create the identity matrix
    207 			ident := bmmatrix.IdentityComplex(2)
    208 			if result == nil {
    209 				result = ident
    210 			} else {
    211 				result = bmmatrix.TensorProductComplex(result, ident)
    212 			}
    213 			if sim.debug {
    214 				fmt.Println("\t\t\tPartial Matrix:")
    215 				fmt.Println(result)
    216 			}
    217 		} else {
    218 			if sim.debug {
    219 				fmt.Print("\t\t\tOperation for qbit ", localQBits[q], " found:"+op[fundLine].String())
    220 			}
    221 			argNumQBits := 0
    222 			for _, arg := range op[fundLine].Elements {
    223 				if _, ok := sim.qbitsNum[arg.GetValue()]; ok {
    224 					argNumQBits++
    225 				}
    226 			}
    227 			if sim.debug {
    228 				fmt.Println(" the operation has", argNumQBits, "qbits")
    229 			}
    230 
    231 			if argNumQBits == 1 {
    232 				// Single qbit operation
    233 				// Create the matrix
    234 				if matrix, err := sim.MatrixFromOp(op[fundLine]); err != nil {
    235 					return nil, fmt.Errorf("error creating matrix from operation: %v", err)
    236 				} else {
    237 					if result == nil {
    238 						result = matrix
    239 					} else {
    240 						result = bmmatrix.TensorProductComplex(result, matrix)
    241 					}
    242 					if sim.debug {
    243 						fmt.Println("\t\t\tPartial Matrix:")
    244 						fmt.Println(result)
    245 					}
    246 				}
    247 			} else {
    248 				// Multi qbit operation
    249 				// The order of the qbits in the operation is important, we need to reorder the qbits in the operation
    250 				// if they are not in sequence by swapping the qbits and then swapping them back after the matrix is created
    251 
    252 				localOrder := make([]int, len(op[fundLine].Elements))
    253 				for i, arg := range op[fundLine].Elements {
    254 					argName := arg.GetValue()
    255 					if _, ok := sim.qbitsNum[argName]; ok {
    256 						localOrder[i] = sim.qbitsNum[argName]
    257 					} else {
    258 						// Leaving out the arguments that are not qbits
    259 						localOrder[i] = -1
    260 					}
    261 				}
    262 
    263 				for i, lq := range localOrder {
    264 					if lq != q && lq != -1 {
    265 						// Swap the qbits
    266 						localQBits[q], localQBits[lq] = localQBits[lq], localQBits[q]
    267 						// Add the swap to the list
    268 						swaps = append(swaps, swap{q, lq})
    269 						if sim.debug {
    270 							fmt.Printf("Swapping qbits %d and %d\n", q, lq)
    271 						}
    272 						// Swap the localOrder if needed
    273 						for j, lq2 := range localOrder {
    274 							if lq2 == q {
    275 								localOrder[j] = lq
    276 							} else if lq2 == lq {
    277 								localOrder[j] = q
    278 							}
    279 						}
    280 						if sim.debug {
    281 							fmt.Println("newLocalOrder:", localOrder)
    282 						}
    283 
    284 					}
    285 					if i != len(localOrder)-1 {
    286 						q++
    287 					}
    288 				}
    289 
    290 				// Create the matrix
    291 				if matrix, err := sim.MatrixFromOp(op[fundLine]); err != nil {
    292 					return nil, fmt.Errorf("error creating matrix from operation: %v", err)
    293 				} else {
    294 					if result == nil {
    295 						result = matrix
    296 					} else {
    297 						result = bmmatrix.TensorProductComplex(result, matrix)
    298 					}
    299 					if sim.debug {
    300 						fmt.Println("\t\t\tPartial Matrix:")
    301 						fmt.Println(result)
    302 					}
    303 				}
    304 			}
    305 		}
    306 	}
    307 
    308 	if len(swaps) == 0 {
    309 		if sim.debug {
    310 			fmt.Println("\t\tNo swaps needed")
    311 		}
    312 	} else {
    313 		if sim.debug {
    314 			fmt.Println("\t\tPerforming the swaps back")
    315 		}
    316 
    317 		for i := range len(swaps) {
    318 			s := swaps[len(swaps)-1-i]
    319 			baseSwaps := swaps2baseSwaps(s, len(sim.qbits))
    320 			for _, bs := range baseSwaps {
    321 				result = bmmatrix.SwapRowsColsComplex(result, bs.s1, bs.s2)
    322 			}
    323 			if sim.debug {
    324 				fmt.Println("Swapping qbits", s.s1, "and", s.s2)
    325 				fmt.Println("\t\t\tPartial Matrix:")
    326 				fmt.Println(result)
    327 			}
    328 		}
    329 	}
    330 	return result, nil
    331 }

    333 func (sim *BmQSimulator) SwapQbits(m *bmmatrix.BmMatrixSquareComplex, s1, s2 int) *bmmatrix.BmMatrixSquareComplex {
    334 	s := swap{s1, s2}
    335 	baseSwaps := swaps2baseSwaps(s, len(sim.qbits))
    336 	// fmt.Println("baseSwaps:", baseSwaps)
    337 	for _, bs := range baseSwaps {
    338 		m = bmmatrix.SwapRowsColsComplex(m, bs.s1, bs.s2)
    339 	}
    340 	return m
    341 }

    343 func swaps2baseSwaps(s swap, n int) []swap {
    344 	baseNum := uint64(1 << n)
    345 	// fmt.Println("baseNum:", baseNum)
    346 
    347 	iDone := make(map[uint64]struct{})
    348 	result := make([]swap, 0)
    349 
    350 	s1 := s.s1
    351 	s2 := s.s2
    352 
    353 	max := uint64(1 << (n - 1))
    354 
    355 	mask1 := uint64(max >> s1)
    356 	mask2 := uint64(max >> s2)
    357 
    358 	// fmt.Println("mask1:", int2bin(int(mask1), n))
    359 	// fmt.Println("mask2:", int2bin(int(mask2), n))
    360 
    361 	for i := uint64(0); i < baseNum; i++ {
    362 		// fmt.Println(i, int2bin(int(i), n), i&mask1>>(n-1-s1), i&mask2>>(n-1-s2))
    363 		if _, ok := iDone[i]; !ok {
    364 			if i&mask1>>(n-1-s1) != i&mask2>>(n-1-s2) {
    365 				num1 := i
    366 				num2 := i ^ mask1 ^ mask2
    367 				iDone[num1] = struct{}{}
    368 				iDone[num2] = struct{}{}
    369 				result = append(result, swap{int(num1), int(num2)})
    370 			}
    371 		}
    372 	}
    373 
    374 	return result
    375 }

    386 func (sim *BmQSimulator) MatrixFromOp(line *bmline.BasmLine) (*bmmatrix.BmMatrixSquareComplex, error) {
    387 	op := line.Operation.GetValue()
    388 	op = strings.ToLower(op)
    389 	switch op {
    390 	case "h", "hadamard":
    391 		return bmmatrix.Hadamard(), nil
    392 	case "x", "paulix":
    393 		return bmmatrix.PauliX(), nil
    394 	case "y", "pauliy":
    395 		return bmmatrix.PauliY(), nil
    396 	case "z", "pauliz":
    397 		return bmmatrix.PauliZ(), nil
    398 	case "cx", "cnot", "xor":
    399 		return bmmatrix.CNot(), nil
    400 	case "s", "p":
    401 		return bmmatrix.S(), nil
    402 	case "v", "sx":
    403 		return bmmatrix.Sx(), nil
    404 	case "t":
    405 		return bmmatrix.T(), nil
    406 	case "xnor":
    407 		return bmmatrix.XNor(), nil
    408 	case "cz", "cphase", "csign", "cpf":
    409 		return bmmatrix.Cphase(), nil
    410 	case "dcnot":
    411 		return bmmatrix.Dcnot(), nil
    412 	case "swap":
    413 		return bmmatrix.Swap(), nil
    414 	case "iswap":
    415 		return bmmatrix.Iswap(), nil
    416 	case "phase", "ph":
    417 		return sim.Phase(line)
    418 	case "r":
    419 		return sim.P(line)
    420 	case "rx":
    421 		return sim.RX(line)
    422 	case "ry":
    423 		return sim.RY(line)
    424 	case "rz":
    425 		return sim.RZ(line)
    426 	case "zero", "input":
    427 	// Ignore the zero and the input operations
    428 	default:
    429 		return nil, fmt.Errorf("unknown operation %s", op)
    430 	}
    431 	return nil, nil
    432 }

    434 func (sim *BmQSimulator) EmitBMAPIMaps(hwflavor string) (string, error) {
    435 	if sim != nil && sim.qbits != nil {
    436 		var ioNum int
    437 		switch hwflavor {
    438 		case "seq_hardcoded_real":
    439 			ioNum = int(math.Pow(float64(2), float64(len(sim.qbits))))
    440 		case "seq_hardcoded_complex", "seq_hardcoded_addtree_complex":
    441 			ioNum = int(math.Pow(float64(2), float64(len(sim.qbits)))) * 2
    442 		}
    443 		newMap := new(IOmap)
    444 		newMap.Assoc = make(map[string]string)
    445 		for i := 0; i < ioNum; i++ {
    446 			newMap.Assoc[fmt.Sprintf("i%d", i)] = fmt.Sprintf("%d", i)
    447 			newMap.Assoc[fmt.Sprintf("o%d", i)] = fmt.Sprintf("%d", i)
    448 		}
    449 		if jData, err := json.Marshal(newMap); err != nil {
    450 			return "", fmt.Errorf("error marshalling the map: %v", err)
    451 		} else {
    452 			return string(jData), nil
    453 		}
    454 	}
    455 	return "", fmt.Errorf("no qbits defined")
    456 }

    458 func (sim *BmQSimulator) RunSoftwareSimulation() error {
    459 	// TODO Include the checks
    460 	sim.Outputs = make([]StateArray, len(sim.Inputs))
    461 	for i, input := range sim.Inputs {
    462 		if len(input.Vector) != sim.StateSize() {
    463 			return fmt.Errorf("input %d has %d vectors, expected %d", i, len(input.Vector), sim.StateSize())
    464 		}
    465 
    466 		curState := make([]bmmatrix.Complex32, sim.StateSize())
    467 		copy(curState, input.Vector)
    468 
    469 		for j, mtx := range sim.Mtx {
    470 			if mtx == nil {
    471 				return fmt.Errorf("matrix %d is nil", j)
    472 			}
    473 			if len(mtx.Data) != sim.StateSize() {
    474 				return fmt.Errorf("matrix %d has %d vectors, expected %d", j, len(mtx.Data), sim.StateSize())
    475 			}
    476 
    477 			if newState, err := bmmatrix.MatrixVectorProductComplex(mtx, curState); err != nil {
    478 				return fmt.Errorf("error running the software simulation: %v", err)
    479 			} else {
    480 				curState = newState
    481 			}
    482 		}
    483 
    484 		sim.Outputs[i] = StateArray{Vector: curState}
    485 
    486 	}
    487 	return nil
