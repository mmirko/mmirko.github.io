     18 func nth_assoc(assoc string, seq int) string {
     19 	re := regexp.MustCompile("\\[(?P<to>[0-9]+):(?P<from>[0-9]+)\\] +(?P<name>[a-zA-Z0-9]+)")
     20 	if re.MatchString(assoc) {
     21 		to_string := re.ReplaceAllString(assoc, "${to}")
     22 		from_string := re.ReplaceAllString(assoc, "${from}")
     23 		name_string := re.ReplaceAllString(assoc, "${name}")
     24 		to, _ := strconv.Atoi(to_string)
     25 		from, _ := strconv.Atoi(from_string)
     26 		step := 1
     27 		if from > to {
     28 			step = -1
     29 		}
     30 		pos := from + seq*step
     31 
     32 		return name_string + "[" + strconv.Itoa(pos) + "]"
     33 	}
     34 	return assoc
     35 }
