     26 module krnl_bondmachine_rtl_axi_write_master #(
     27   parameter integer C_ADDR_WIDTH       = 64,
     28   parameter integer C_DATA_WIDTH       = 32,
     29   parameter integer C_MAX_LENGTH_WIDTH = 32,
     30   parameter integer C_BURST_LEN        = 256,
     31   parameter integer C_LOG_BURST_LEN    = 8
     32 )
     33 (
     34   // Control interface
     35   input wire                          ctrl_start,
     36   input wire [C_ADDR_WIDTH-1:0]       ctrl_offset,
     37   input wire [C_MAX_LENGTH_WIDTH-1:0] ctrl_length,
     38   output wire                         ctrl_done,
     39 
     40   // AXI4-Stream interface
     41   input wire                          s_tvalid,
     42   input wire  [C_DATA_WIDTH-1:0]      s_tdata,
     43   output wire                         s_tready,
     44 
     45   // AXI Interface 
     46   input wire                          aclk,
     47   input wire                          areset,
     48 
     49   output wire [C_ADDR_WIDTH-1:0]      awaddr,
     50   output wire [7:0]                   awlen,
     51   output wire [2:0]                   awsize,
     52   output wire                         awvalid,
     53   input wire                          awready,
     54 
     55   output wire [C_DATA_WIDTH-1:0]      wdata,
     56   output wire [C_DATA_WIDTH/8-1:0]    wstrb,
     57   output wire                         wlast,
     58   output wire                         wvalid,
     59   input wire                          wready,
     60 
     61   input wire [1:0]                    bresp,
     62   input wire                          bvalid,
     63   output wire                         bready
     64 );
     65 
     66 timeunit 1ps; 
     67 timeprecision 1ps; 
     68 
     69 /////////////////////////////////////////////////////////////////////////////
     70 // Local Parameters
     71 /////////////////////////////////////////////////////////////////////////////
     72 localparam integer LP_LOG_MAX_W_TO_AW = 8; // Allow up to 256 outstanding w to aw transactions
     73 localparam integer LP_TRANSACTION_CNTR_WIDTH = C_MAX_LENGTH_WIDTH-C_LOG_BURST_LEN;
     74 
     75 /////////////////////////////////////////////////////////////////////////////
     76 // Variables
     77 /////////////////////////////////////////////////////////////////////////////
     78 logic [LP_TRANSACTION_CNTR_WIDTH-1:0] num_full_bursts;
     79 logic                                 num_partial_bursts;
     80 logic                                 start    = 1'b0;
     81 logic [LP_TRANSACTION_CNTR_WIDTH-1:0] num_transactions;
     82 logic                                 has_partial_burst;
     83 logic [C_LOG_BURST_LEN-1:0]           final_burst_len;
     84 logic                                 single_transaction;
     85 
     86 logic                                 wxfer;       // Unregistered write data transfer
     87 logic                                 wfirst = 1'b1;
     88 logic                                 load_burst_cntr;
     89 logic [C_LOG_BURST_LEN-1:0]           wxfers_to_go;  // Used for simulation debug
     90 logic [LP_TRANSACTION_CNTR_WIDTH-1:0] w_transactions_to_go;
     91 logic                                 w_final_transaction;
     92 logic                                 w_almost_final_transaction = 1'b0;
     93 
     94 logic                                 awxfer;
     95 logic                                 awvalid_r    = 1'b0; 
     96 logic [C_ADDR_WIDTH-1:0]              addr;
     97 logic                                 wfirst_d1    = 1'b0; 
     98 logic                                 wfirst_pulse = 1'b0;
     99 logic [LP_LOG_MAX_W_TO_AW-1:0]        dbg_w_to_aw_outstanding;
    100 logic                                 idle_aw;
    101 logic [LP_TRANSACTION_CNTR_WIDTH-1:0] aw_transactions_to_go;
    102 logic                                 aw_final_transaction;
    103 
    104 wire                                  bxfer;
    105 logic [LP_TRANSACTION_CNTR_WIDTH-1:0] b_transactions_to_go;
    106 logic                                 b_final_transaction;
    107 
    108 /////////////////////////////////////////////////////////////////////////////
    109 // Control logic
    110 /////////////////////////////////////////////////////////////////////////////
    111 // Count the number of transfers and assert done when the last bvalid is received.
    112 assign num_full_bursts = ctrl_length[C_LOG_BURST_LEN+:C_MAX_LENGTH_WIDTH-C_LOG_BURST_LEN];
    113 assign num_partial_bursts = ctrl_length[0+:C_LOG_BURST_LEN] ? 1'b1 : 1'b0; 
    114 
    115 always @(posedge aclk) begin 
    116   start <= ctrl_start;
    117   num_transactions <= (num_partial_bursts == 1'b0) ? num_full_bursts - 1'b1 : num_full_bursts;
    118   has_partial_burst <= num_partial_bursts;
    119   final_burst_len <=  ctrl_length[0+:C_LOG_BURST_LEN] - 1'b1;
    120 end
    121 
    122 assign ctrl_done = bxfer & b_final_transaction;
    123 assign single_transaction = (num_transactions == {LP_TRANSACTION_CNTR_WIDTH{1'b0}}) ? 1'b1 : 1'b0;
    124 
    125 /////////////////////////////////////////////////////////////////////////////
    126 // AXI Write Data Channel
    127 /////////////////////////////////////////////////////////////////////////////
    128 assign wvalid        = s_tvalid;
    129 assign wdata         = s_tdata;
    130 assign wstrb         = {(C_DATA_WIDTH/8){1'b1}};
    131 assign s_tready = wready;
    132 
    133 assign wxfer = wvalid & wready; 
    134 
    135 always @(posedge aclk) begin 
    136   if (areset) begin 
    137     wfirst <= 1'b1;
    138   end
    139   else begin
    140     wfirst <= wxfer ? wlast : wfirst;
    141   end
    142 end
    143 
    144 // Load burst counter with partial burst if on final transaction or if there is only 1 transaction
    145 assign load_burst_cntr = (wxfer & wlast & w_almost_final_transaction) || (start & single_transaction);
    146 
    147 krnl_bondmachine_rtl_counter #(
    148   .C_WIDTH ( C_LOG_BURST_LEN         ) ,
    149   .C_INIT  ( {C_LOG_BURST_LEN{1'b1}} ) 
    150 )
    151 inst_burst_cntr ( 
    152   .clk        ( aclk            ) ,
    153   .clken      ( 1'b1            ) ,
    154   .rst        ( areset          ) ,
    155   .load       ( load_burst_cntr ) ,
    156   .incr       ( 1'b0            ) ,
    157   .decr       ( wxfer           ) ,
    158   .load_value ( final_burst_len ) ,
    159   .count      ( wxfers_to_go    ) ,
    160   .is_zero    ( wlast           ) 
    161 );
    162 
    163 krnl_bondmachine_rtl_counter #(
    164   .C_WIDTH ( LP_TRANSACTION_CNTR_WIDTH         ) ,
    165   .C_INIT  ( {LP_TRANSACTION_CNTR_WIDTH{1'b0}} ) 
    166 )
    167 inst_w_transaction_cntr ( 
    168   .clk        ( aclk                 ) ,
    169   .clken      ( 1'b1                 ) ,
    170   .rst        ( areset               ) ,
    171   .load       ( start                ) ,
    172   .incr       ( 1'b0                 ) ,
    173   .decr       ( wxfer & wlast        ) ,
    174   .load_value ( num_transactions     ) ,
    175   .count      ( w_transactions_to_go ) ,
    176   .is_zero    ( w_final_transaction  ) 
    177 );
    178 
    179 always @(posedge aclk) begin
    180   w_almost_final_transaction <= (w_transactions_to_go == 1) ? 1'b1 : 1'b0;
    181 end
    182 
    183 /////////////////////////////////////////////////////////////////////////////
    184 // AXI Write Address Channel
    185 /////////////////////////////////////////////////////////////////////////////
    186 // The address channel samples the data channel and send out transactions when 
    187 // first beat of wdata is asserted. This ensures that address requests are not 
    188 // sent without data on the way.
    189 
    190 assign awvalid = awvalid_r;
    191 assign awxfer = awvalid & awready;
    192 
    193 always @(posedge aclk) begin 
    194   if (areset) begin 
    195     awvalid_r <= 1'b0;
    196   end
    197   else begin
    198     awvalid_r <= ~idle_aw & ~awvalid_r ? 1'b1 : 
    199                  awready               ? 1'b0 : 
    200                                          awvalid_r;
    201   end
    202 end
    203 
    204 assign awaddr = addr;
    205 
    206 always @(posedge aclk) begin 
    207   addr <= ctrl_start ? ctrl_offset :
    208           awxfer     ? addr + C_BURST_LEN*C_DATA_WIDTH/8 : 
    209                        addr;
    210 end
    211 
    212 assign awlen   = aw_final_transaction || (start & single_transaction) ? final_burst_len : C_BURST_LEN - 1;
    213 assign awsize  = $clog2((C_DATA_WIDTH/8));
    214 
    215 krnl_bondmachine_rtl_counter #(
    216   .C_WIDTH (LP_LOG_MAX_W_TO_AW),
    217   .C_INIT ({LP_LOG_MAX_W_TO_AW{1'b0}})
    218 )
    219 inst_w_to_aw_cntr ( 
    220   .clk        ( aclk                    ) ,
    221   .clken      ( 1'b1                    ) ,
    222   .rst        ( areset                  ) ,
    223   .load       ( 1'b0                    ) ,
    224   .incr       ( wfirst_pulse            ) ,
    225   .decr       ( awxfer                  ) ,
    226   .load_value (                         ) ,
    227   .count      ( dbg_w_to_aw_outstanding ) ,
    228   .is_zero    ( idle_aw                 ) 
    229 );
    230 
    231 always @(posedge aclk) begin 
    232   wfirst_d1 <= wvalid & wfirst; 
    233 end
    234 
    235 always @(posedge aclk) begin 
    236   wfirst_pulse <= wvalid & wfirst & ~wfirst_d1;
    237 end
    238 
    239 krnl_bondmachine_rtl_counter #(
    240   .C_WIDTH ( LP_TRANSACTION_CNTR_WIDTH         ) ,
    241   .C_INIT  ( {LP_TRANSACTION_CNTR_WIDTH{1'b0}} ) 
    242 )
    243 inst_aw_transaction_cntr ( 
    244   .clk        ( aclk                   ) ,
    245   .clken      ( 1'b1                   ) ,
    246   .rst        ( areset                 ) ,
    247   .load       ( start                  ) ,
    248   .incr       ( 1'b0                   ) ,
    249   .decr       ( awxfer                 ) ,
    250   .load_value ( num_transactions       ) ,
    251   .count      ( aw_transactions_to_go  ) ,
    252   .is_zero    ( aw_final_transaction   ) 
    253 );
    254 
    255 /////////////////////////////////////////////////////////////////////////////
    256 // AXI Write Response Channel
    257 /////////////////////////////////////////////////////////////////////////////
    258 
    259 assign bready = 1'b1;
    260 assign bxfer = bready & bvalid;
    261 
    262 krnl_bondmachine_rtl_counter #(
    263   .C_WIDTH ( LP_TRANSACTION_CNTR_WIDTH         ) ,
    264   .C_INIT  ( {LP_TRANSACTION_CNTR_WIDTH{1'b0}} ) 
    265 )
    266 inst_b_transaction_cntr ( 
    267   .clk        ( aclk                 ) ,
    268   .clken      ( 1'b1                 ) ,
    269   .rst        ( areset               ) ,
    270   .load       ( start                ) ,
    271   .incr       ( 1'b0                 ) ,
    272   .decr       ( bxfer                ) ,
    273   .load_value ( num_transactions     ) ,
    274   .count      ( b_transactions_to_go ) ,
    275   .is_zero    ( b_final_transaction  ) 
    276 );
