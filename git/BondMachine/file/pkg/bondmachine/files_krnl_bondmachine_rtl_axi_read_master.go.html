     21 ///////////////////////////////////////////////////////////////////////////////
     22 // Description: This is a multi-threaded AXI4 read master.  Each channel will
     23 // issue commands on a different IDs.  As a result data may arrive out of 
     24 // order.  The amount of data requested is equal to the ctrl_length variable.
     25 // Prog full is set and sampled such that the FIFO will never overflow.  Thus 
     26 // rready can be always asserted for better timing.

     30 
     31 module krnl_bondmachine_rtl_axi_read_master #( 
     32   parameter integer C_ID_WIDTH         = 1,   // Must be >= $clog2(C_NUM_CHANNELS)
     33   parameter integer C_ADDR_WIDTH       = 64,
     34   parameter integer C_DATA_WIDTH       = 32,
     35   parameter integer C_NUM_CHANNELS     = 1,   // Only 2 tested.
     36   parameter integer C_LENGTH_WIDTH     = 32,  
     37   parameter integer C_BURST_LEN        = 256, // Max AXI burst length for read commands
     38   parameter integer C_LOG_BURST_LEN    = 8,
     39   parameter integer C_MAX_OUTSTANDING  = 3 
     40 )
     41 (
     42   // System signals
     43   input  wire                                          aclk,
     44   input  wire                                          areset,
     45   // Control signals 
     46   input  wire                                          ctrl_start, 
     47   output wire                                          ctrl_done, 
     48   input  wire [C_NUM_CHANNELS-1:0][C_ADDR_WIDTH-1:0]   ctrl_offset,
     49   input  wire                     [C_LENGTH_WIDTH-1:0] ctrl_length,
     50   input  wire [C_NUM_CHANNELS-1:0]                     ctrl_prog_full,
     51   // AXI4 master interface                             
     52   output wire                                          arvalid,
     53   input  wire                                          arready,
     54   output wire [C_ADDR_WIDTH-1:0]                       araddr,
     55   output wire [C_ID_WIDTH-1:0]                         arid,
     56   output wire [7:0]                                    arlen,
     57   output wire [2:0]                                    arsize,
     58   input  wire                                          rvalid,
     59   output wire                                          rready,
     60   input  wire [C_DATA_WIDTH - 1:0]                     rdata,
     61   input  wire                                          rlast,
     62   input  wire [C_ID_WIDTH - 1:0]                       rid,
     63   input  wire [1:0]                                    rresp,
     64   // AXI4-Stream master interface, 1 interface per channel.
     65   output wire [C_NUM_CHANNELS-1:0]                     m_tvalid,
     66   input  wire [C_NUM_CHANNELS-1:0]                     m_tready,
     67   output wire [C_NUM_CHANNELS-1:0][C_DATA_WIDTH-1:0]   m_tdata
     68 );
     69 
     70 timeunit 1ps; 
     71 timeprecision 1ps; 
     72 
     73 ///////////////////////////////////////////////////////////////////////////////
     74 // Local Parameters
     75 ///////////////////////////////////////////////////////////////////////////////
     76 localparam integer LP_MAX_OUTSTANDING_CNTR_WIDTH = $clog2(C_MAX_OUTSTANDING+1); 
     77 localparam integer LP_TRANSACTION_CNTR_WIDTH = C_LENGTH_WIDTH-C_LOG_BURST_LEN;
     78 
     79 ///////////////////////////////////////////////////////////////////////////////
     80 // Variables
     81 ///////////////////////////////////////////////////////////////////////////////
     82 // Control logic
     83 logic [C_NUM_CHANNELS-1:0]            done = '0;
     84 logic [LP_TRANSACTION_CNTR_WIDTH-1:0] num_full_bursts;
     85 logic                                 num_partial_bursts;
     86 logic                                 start    = 1'b0;
     87 logic [LP_TRANSACTION_CNTR_WIDTH-1:0] num_transactions;
     88 logic                                 has_partial_burst;
     89 logic [C_LOG_BURST_LEN-1:0]           final_burst_len;
     90 logic                                 single_transaction;
     91 logic                                 ar_idle = 1'b1;
     92 logic                                 ar_done;
     93 // AXI Read Address Channel
     94 logic                                                         fifo_stall;
     95 logic                                                         arxfer;
     96 logic                                                         arvalid_r = 1'b0; 
     97 logic [C_NUM_CHANNELS-1:0][C_ADDR_WIDTH-1:0]                  addr;
     98 logic [C_ID_WIDTH-1:0]                                        id = {C_ID_WIDTH{1'b1}};
     99 logic [LP_TRANSACTION_CNTR_WIDTH-1:0]                         ar_transactions_to_go;
    100 logic                                                         ar_final_transaction;
    101 logic [C_NUM_CHANNELS-1:0]                                    incr_ar_to_r_cnt;
    102 logic [C_NUM_CHANNELS-1:0]                                    decr_ar_to_r_cnt;
    103 logic [C_NUM_CHANNELS-1:0]                                    stall_ar;
    104 logic [C_NUM_CHANNELS-1:0][LP_MAX_OUTSTANDING_CNTR_WIDTH-1:0] outstanding_vacancy_count;
    105 // AXI Data Channel
    106 logic [C_NUM_CHANNELS-1:0]                                tvalid;
    107 logic [C_NUM_CHANNELS-1:0][C_DATA_WIDTH-1:0]              tdata;
    108 logic                                                     rxfer;
    109 logic [C_NUM_CHANNELS-1:0]                                decr_r_transaction_cntr;
    110 logic [C_NUM_CHANNELS-1:0][LP_TRANSACTION_CNTR_WIDTH-1:0] r_transactions_to_go;
    111 logic [C_NUM_CHANNELS-1:0]                                r_final_transaction;
    112 ///////////////////////////////////////////////////////////////////////////////
    113 // Control Logic 
    114 ///////////////////////////////////////////////////////////////////////////////
    115 
    116 always @(posedge aclk) begin
    117   for (int i = 0; i < C_NUM_CHANNELS; i++) begin 
    118     done[i] <= rxfer & rlast & (rid == i) & r_final_transaction[i] ? 1'b1 : 
    119           ctrl_done ? 1'b0 : done[i]; 
    120   end
    121 end
    122 assign ctrl_done = &done;
    123 
    124 // Determine how many full burst to issue and if there are any partial bursts.
    125 assign num_full_bursts = ctrl_length[C_LOG_BURST_LEN+:C_LENGTH_WIDTH-C_LOG_BURST_LEN];
    126 assign num_partial_bursts = ctrl_length[0+:C_LOG_BURST_LEN] ? 1'b1 : 1'b0; 
    127 
    128 always @(posedge aclk) begin 
    129   start <= ctrl_start;
    130   num_transactions <= (num_partial_bursts == 1'b0) ? num_full_bursts - 1'b1 : num_full_bursts;
    131   has_partial_burst <= num_partial_bursts;
    132   final_burst_len <=  ctrl_length[0+:C_LOG_BURST_LEN] - 1'b1;
    133 end
    134 
    135 // Special case if there is only 1 AXI transaction. 
    136 assign single_transaction = (num_transactions == {LP_TRANSACTION_CNTR_WIDTH{1'b0}}) ? 1'b1 : 1'b0;
    137 
    138 ///////////////////////////////////////////////////////////////////////////////
    139 // AXI Read Address Channel
    140 ///////////////////////////////////////////////////////////////////////////////
    141 assign arvalid = arvalid_r;
    142 assign araddr = addr[id];
    143 assign arlen  = ar_final_transaction || (start & single_transaction) ? final_burst_len : C_BURST_LEN - 1;
    144 assign arsize = $clog2((C_DATA_WIDTH/8));
    145 assign arid   = id;
    146 
    147 assign arxfer = arvalid & arready;
    148 assign fifo_stall = ctrl_prog_full[id];
    149 
    150 always @(posedge aclk) begin 
    151   if (areset) begin 
    152     arvalid_r <= 1'b0;
    153   end
    154   else begin
    155     arvalid_r <= ~ar_idle & ~stall_ar[id] & ~arvalid_r & ~fifo_stall ? 1'b1 : 
    156                  arready ? 1'b0 : arvalid_r;
    157   end
    158 end
    159 
    160 // When ar_idle, there are no transactions to issue.
    161 always @(posedge aclk) begin 
    162   if (areset) begin 
    163     ar_idle <= 1'b1; 
    164   end
    165   else begin 
    166     ar_idle <= start   ? 1'b0 :
    167                ar_done ? 1'b1 : 
    168                          ar_idle;
    169   end
    170 end
    171 
    172 // each channel is assigned a different id. The transactions are interleaved.
    173 always @(posedge aclk) begin 
    174   if (start) begin 
    175     id <= {C_ID_WIDTH{1'b1}};
    176   end
    177   else begin
    178     id <= arxfer ? id - 1'b1 : id; 
    179   end
    180 end
    181 
    182 
    183 // Increment to next address after each transaction is issued.
    184 always @(posedge aclk) begin 
    185   for (int i = 0; i < C_NUM_CHANNELS; i++) begin
    186     addr[i] <= ctrl_start          ? ctrl_offset[i] :
    187                arxfer && (id == i) ? addr[i] + C_BURST_LEN*C_DATA_WIDTH/8 : 
    188                                      addr[i];
    189   end
    190 end
    191 
    192 // Counts down the number of transactions to send.
    193 krnl_bondmachine_rtl_counter #(
    194   .C_WIDTH ( LP_TRANSACTION_CNTR_WIDTH         ) ,
    195   .C_INIT  ( {LP_TRANSACTION_CNTR_WIDTH{1'b0}} ) 
    196 )
    197 inst_ar_transaction_cntr ( 
    198   .clk        ( aclk                   ) ,
    199   .clken      ( 1'b1                   ) ,
    200   .rst        ( areset                 ) ,
    201   .load       ( start                  ) ,
    202   .incr       ( 1'b0                   ) ,
    203   .decr       ( arxfer && id == '0     ) ,
    204   .load_value ( num_transactions       ) ,
    205   .count      ( ar_transactions_to_go  ) ,
    206   .is_zero    ( ar_final_transaction   ) 
    207 );
    208 
    209 assign ar_done = ar_final_transaction && arxfer && id == 1'b0;
    210 
    211 always_comb begin 
    212   for (int i = 0; i < C_NUM_CHANNELS; i++) begin 
    213     incr_ar_to_r_cnt[i] = rxfer & rlast & (rid == i);
    214     decr_ar_to_r_cnt[i] = arxfer & (arid == i);
    215   end
    216 end
    217 
    218 // Keeps track of the number of outstanding transactions. Stalls 
    219 // when the value is reached so that the FIFO won't overflow.
    220 krnl_bondmachine_rtl_counter #(
    221   .C_WIDTH ( LP_MAX_OUTSTANDING_CNTR_WIDTH                       ) ,
    222   .C_INIT  ( C_MAX_OUTSTANDING[0+:LP_MAX_OUTSTANDING_CNTR_WIDTH] ) 
    223 )
    224 inst_ar_to_r_transaction_cntr[C_NUM_CHANNELS-1:0] ( 
    225   .clk        ( aclk                           ) ,
    226   .clken      ( 1'b1                           ) ,
    227   .rst        ( areset                         ) ,
    228   .load       ( 1'b0                           ) ,
    229   .incr       ( incr_ar_to_r_cnt               ) ,
    230   .decr       ( decr_ar_to_r_cnt               ) ,
    231   .load_value ( {LP_MAX_OUTSTANDING_CNTR_WIDTH{1'b0}} ) ,
    232   .count      ( outstanding_vacancy_count      ) ,
    233   .is_zero    ( stall_ar                       ) 
    234 );
    235 
    236 ///////////////////////////////////////////////////////////////////////////////
    237 // AXI Read Channel
    238 ///////////////////////////////////////////////////////////////////////////////
    239 assign m_tvalid = tvalid;
    240 assign m_tdata = tdata;
    241 
    242 always_comb begin 
    243   for (int i = 0; i < C_NUM_CHANNELS; i++) begin
    244     tvalid[i] = rvalid && (rid == i); 
    245     tdata[i] = rdata;
    246   end
    247 end
    248 
    249 // rready can remain high for optimal timing because ar transactions are not issued
    250 // unless there is enough space in the FIFO.
    251 assign rready = 1'b1;
    252 assign rxfer = rready & rvalid;
    253 
    254 always_comb begin 
    255   for (int i = 0; i < C_NUM_CHANNELS; i++) begin 
    256     decr_r_transaction_cntr[i] = rxfer & rlast & (rid == i);
    257   end
    258 end
    259 krnl_bondmachine_rtl_counter #(
    260   .C_WIDTH ( LP_TRANSACTION_CNTR_WIDTH         ) ,
    261   .C_INIT  ( {LP_TRANSACTION_CNTR_WIDTH{1'b0}} ) 
    262 )
    263 inst_r_transaction_cntr[C_NUM_CHANNELS-1:0] ( 
    264   .clk        ( aclk                          ) ,
    265   .clken      ( 1'b1                          ) ,
    266   .rst        ( areset                        ) ,
    267   .load       ( start                         ) ,
    268   .incr       ( 1'b0                          ) ,
    269   .decr       ( decr_r_transaction_cntr       ) ,
    270   .load_value ( num_transactions              ) ,
    271   .count      ( r_transactions_to_go          ) ,
    272   .is_zero    ( r_final_transaction           ) 
    273 );
    274 
    275 
    276 endmodule : krnl_bondmachine_rtl_axi_read_master
