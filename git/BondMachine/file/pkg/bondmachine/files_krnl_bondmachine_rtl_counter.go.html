     27   parameter integer C_WIDTH  = 4,
     28   parameter [C_WIDTH-1:0] C_INIT = {C_WIDTH{1'b0}}
     29 )
     30 ( 
     31   input  wire               clk,
     32   input  wire               clken,
     33   input  wire               rst,
     34   input  wire               load,
     35   input  wire               incr,
     36   input  wire               decr,
     37   input  wire [C_WIDTH-1:0] load_value,
     38   output wire [C_WIDTH-1:0] count,
     39   output wire               is_zero
     40 );
     41 
     42   localparam [C_WIDTH-1:0] LP_ZERO = {C_WIDTH{1'b0}};
     43   localparam [C_WIDTH-1:0] LP_ONE = {{"{{"}}C_WIDTH-1{1'b0{{"}}"}},1'b1};
     44   localparam [C_WIDTH-1:0] LP_MAX = {C_WIDTH{1'b1}};
     45 
     46   reg [C_WIDTH-1:0] count_r = C_INIT;
     47   reg   is_zero_r = (C_INIT == LP_ZERO);
     48 
     49   assign count = count_r;
     50 
     51   always @(posedge clk) begin 
     52     if (rst) begin 
     53       count_r <= C_INIT;
     54     end
     55     else if (clken) begin 
     56       if (load) begin
     57         count_r <= load_value; 
     58       end
     59       else if (incr & ~decr) begin
     60         count_r <= count_r + 1'b1;
     61       end
     62       else if (~incr & decr) begin 
     63         count_r <= count_r - 1'b1;
     64       end
     65       else 
     66         count_r <= count_r;
     67     end
     68   end
     69 
     70   assign is_zero = is_zero_r;
     71

     73     if (rst) begin 
     74       is_zero_r <= (C_INIT == LP_ZERO);
     75     end
     76     else if (clken) begin 
     77       if (load) begin 
     78         is_zero_r <= (load_value == LP_ZERO);
