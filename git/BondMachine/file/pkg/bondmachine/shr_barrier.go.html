    104 	result += "module " + barrier_name + "(clk, reset" + subresult + ");\n"
    105 	result += "\n"
    106 	result += "	//--------------Input Ports-----------------------\n"

    135 	}
    136 	result += "\n"
    137

    139 	result += "		done = 1'b0;\n"

    144 	result += "	end\n"
    145 	result += "\n"
    146 
    147 	result += "	always @ (posedge clock) begin\n"

    164 		result += "		if (!done & !timeout & (" + orlist + "))\n"
    165 		result += "			counter <= counter + 1'b1;\n"
    166 		result += "	end\n"
    167 		result += "\n"
    168 	}
    169 
    170 	result += "	always @(posedge clock) begin\n"
    171 	result += "		if (" + andlist + ")\n"
    172 	result += "			done <= 1;\n"
    173 	if has_tout {
    174 		result += "		else begin\n"
    175 		result += "			if (counter == " + strconv.Itoa(toutlen) + "'b" + binarytout + ")\n"

    184 		for _, so_id := range solist {
    185 			if so_id == so_index {

    192 		}
    193 	}
