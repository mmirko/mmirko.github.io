    309 func (bmach *Bondmachine) Del_input(iid int) error {
    310 	if iid < bmach.Inputs {
    311 		bondtoremove := Bond{0, iid, 0}
    312 		// 1 - remove all the bonds using the input (an input is within Internal_outputs)
    313 		for i, linked := range bmach.Links {
    314 			if linked != -1 {
    315 				chkbond := bmach.Internal_outputs[linked]
    316 				if chkbond.Map_to == 0 && chkbond.Res_id == bondtoremove.Res_id {
    317 					bmach.Links[i] = -1
    318 				}
    319 			}
    320 		}
    321 
    322 		// 2 - Remove the internal output
    323 		// 3 - lower by 1 all ids > iid in bonds
    324 		newinternal := make([]Bond, len(bmach.Internal_outputs)-1)
    325 		j := 0
    326 		keeppos := -1
    327 		for i, bond := range bmach.Internal_outputs {
    328 			if bond.Map_to == 0 {
    329 				if bond.Res_id == bondtoremove.Res_id {
    330 					j--
    331 					// Keep the position of the bond removed from Internal_outputs
    332 					keeppos = i
    333 				} else if bond.Res_id > bondtoremove.Res_id {
    334 					newinternal[j] = Bond{0, bond.Res_id - 1, 0}
    335 				} else {
    336 					newinternal[j] = bond
    337 				}
    338 			} else {
    339 				newinternal[j] = bond
    340 			}
    341 			j++
    342 		}
    343 
    344 		bmach.Internal_outputs = newinternal
    345 
    346 		// 4 - Swift links
    347 		if keeppos > -1 {
    348 			for i, linked := range bmach.Links {
    349 				if linked > keeppos {
    350 					bmach.Links[i] = linked - 1
    351 				}
    352 			}
    353 		}
    354 
    355 		// 5 - Lower Inputs
    356 		bmach.Inputs--
    357 	} else {
    358 		return Prerror{"Input id outside limit"}
    359 	}
    360 	return nil
    361 }

    615 	var e1 string
    616 
    617 	for _, outp := range bmach.Internal_outputs {
    618 		if outp.String() == endpoints[0] {
    619 			e0 = endpoints[0]
    620 		}
    621 		if outp.String() == endpoints[1] {
    622 			e1 = endpoints[1]
    623 		}
    624 	}
    625 
    626 	if e0 != "" && e1 != "" {
    627 		mybcore := new(procbuilder.Machine)
    628 
    629 		myarch := &mybcore.Arch
    630 
    631 		myarch.Rsize = uint8(bmach.Rsize)
    632 
    633 		myarch.Modes = make([]string, 1)
    634 		myarch.Modes[0] = "ha"
    635 
    636 		opcodes := make([]procbuilder.Opcode, 0)
    637 
    638 		for _, op := range procbuilder.Allopcodes {
    639 			for _, opn := range []string{"sic", "r2o", "j"} {
    640 				if opn == op.Op_get_name() {
    641 					opcodes = append(opcodes, op)
    642 					break
    643 				}
    644 			}
    645 		}
    646 
    647 		sort.Sort(procbuilder.ByName(opcodes))
    648 
    649 		myarch.Op = opcodes
    650 
    651 		myarch.R = uint8(1)
    652 		myarch.L = uint8(0)
    653 		myarch.N = uint8(2)
    654 		myarch.M = uint8(1)
    655 		myarch.O = uint8(2)
    656 		myarch.Shared_constraints = ""
    657 
    658 		prog := "sic r0 i0\nsic r0 i1\nr2o r0 o0\nj 0\n"
    659 
    660 		if prog, err := myarch.Assembler([]byte(prog)); err == nil {
    661 			mybcore.Program = prog
    662 		} else {
    663 			fmt.Println(err)
    664 		}
    665 
    666 		bmach.Domains = append(bmach.Domains, mybcore)
    667 		bmach.Add_processor(len(bmach.Domains) - 1)
    668 		newpnum := strconv.Itoa(len(bmach.Processors) - 1)
    669 		bmach.Add_bond([]string{"p" + newpnum + "i0", e0})
    670 		bmach.Add_bond([]string{"p" + newpnum + "i1", e1})
    671 		bmach.Add_output()
    672 		newonum := strconv.Itoa(bmach.Outputs - 1)
    673 		bmach.Add_bond([]string{"p" + newpnum + "o0", "o" + newonum})
    674 	} else {
    675 		return Prerror{"Benchmark core endpoints has to be internal outputs"}
    676 	}
    677 	return nil
    678 }
