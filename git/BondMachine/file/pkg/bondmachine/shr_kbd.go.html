     45 func (op Kbd) Instantiate(s string) (Shared_instance, bool) {
     46 	if strings.HasPrefix(s, "kbd:") {
     47 		result := new(Kbd_instance)
     48 		result.Shared_element = op
     49 		components := strings.Split(s, ":")
     50 		componentsN := len(components)
     51 		if componentsN == 2 {
     52 			if depth, ok := strconv.Atoi(components[1]); ok == nil {
     53 				result.Depth = depth
     54 			}
     55 			return result, true
     56 		}
     57 	}
     58 	return nil, false
     59 }

     72 func (sm Kbd_instance) Write_verilog(bmach *Bondmachine, soIndex int, kbdName string, flavor string) string {
     73 
     74 	result := ""
     75 
     76 	// Compute the receivers and senders of the kbd, senders will be the writers of the fifo, receivers will be the readers of the fifo
     77 	receivers := make([]string, 0)
     78 
     79 	for numProcessor, soList := range bmach.Shared_links {
     80 		for _, soId := range soList {
     81 			if soId == soIndex {
     82 				for _, op := range bmach.Domains[bmach.Processors[numProcessor]].Op {
     83 					switch op.Op_get_name() {
     84 					case "k2r":
     85 						receivers = append(receivers, "p"+strconv.Itoa(numProcessor)+"kbd_recv")
     86 						continue
     87 					}
     88 				}
     89 			}
     90 		}
     91 	}
     92 
     93 	// Create the fifo where the CPs will read from
     94 	rFifo := bmstack.CreateBasicStack()
     95 	rFifo.ModuleName = kbdName + "rfifo"
     96 	rFifo.DataSize = 8
     97 	rFifo.Depth = sm.Depth
     98 	rFifo.MemType = "FIFO"
     99 	rFifo.Senders = []string{"kbdwriter"}
    100 	rFifo.Receivers = receivers
    101 
    102 	r, _ := rFifo.WriteHDL()
    103 
    104 	result += r
    105 
    106 	// TODO : add the kbd module
    107 
    108 	return result
    109 
    110 }

    119 				result += "\twire p" + strconv.Itoa(procId) + soName + "receiverAck;\n"

    122 			}
    123 		}
    124 	}
