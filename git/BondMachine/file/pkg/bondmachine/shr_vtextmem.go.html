     44 func (op Vtextmem) Instantiate(s string) (Shared_instance, bool) {
     45 	if strings.HasPrefix(s, "vtextmem:") {
     46 		components := strings.Split(s, ":")
     47 		componentsN := len(components)
     48 		if (componentsN-1)%5 == 0 {
     49 			boxes := make([]GraphBox, 0)
     50 
     51 			for i := 1; i < componentsN; i = i + 5 {
     52 				newBox := GraphBox{}
     53 				if newCP, err := strconv.Atoi(components[i]); err == nil {
     54 					newBox.CP = newCP
     55 				} else {
     56 					return nil, false
     57 				}
     58 				if newLeft, err := strconv.Atoi(components[i+1]); err == nil {
     59 					newBox.Left = newLeft
     60 				} else {
     61 					return nil, false
     62 				}
     63 				if newTop, err := strconv.Atoi(components[i+2]); err == nil {
     64 					newBox.Top = newTop
     65 				} else {
     66 					return nil, false
     67 				}
     68 				if newWidth, err := strconv.Atoi(components[i+3]); err == nil {
     69 					newBox.Width = newWidth
     70 				} else {
     71 					return nil, false
     72 				}
     73 				if newHeight, err := strconv.Atoi(components[i+4]); err == nil {
     74 					newBox.Height = newHeight
     75 				} else {
     76 					return nil, false
     77 				}
     78 
     79 				boxes = append(boxes, newBox)
     80 			}
     81 
     82 			result := new(Vtextmem_instance)
     83 			result.Shared_element = op
     84 			result.Boxes = boxes
     85 			return *result, true
     86 		}
     87 	}
     88 	return nil, false
     89 }
