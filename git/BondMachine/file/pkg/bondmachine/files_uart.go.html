     86         RX_READ_BITS        = 3'd3,

    127     function integer log2(input integer M);
    128         integer i;
    129     begin
    130         log2 = 1;
    131         for (i = 0; 2**i <= M; i = i + 1)
    132             log2 = i + 1;

    138     always @(posedge clk) begin
    139         if (rst) begin
    140             recv_state = RX_IDLE;
    141             tx_state = TX_IDLE;
    142         end
    143                               
    144         // Countdown timers for the receiving and transmitting
    145         // state machines are decremented.
    146         
    147         if(rx_clk) begin
    148             rx_clk = rx_clk - 1'd1;
    149         end
    150     
    151         if(tx_clk) begin
    152             tx_clk = tx_clk - 1'd1;
    153         end
    154         
    155     
    156 //** Receive state machine ************************************
    157 
    158         case (recv_state)
    159             RX_IDLE: begin
    160                 // A low pulse on the receive line indicates the
    161                 // start of data.
    162                 if (!rx) begin
    163                     // Wait 1/2 of the bit period
    164                     rx_clk = one_baud_cnt / 2;
    165                     recv_state = RX_CHECK_START;
    166                 end
    167             end
    168             
    169             RX_CHECK_START: begin
    170                 if (!rx_clk) begin
    171                     // Check the pulse is still there
    172                     if (!rx) begin
    173                         // Pulse still there - good
    174                         // Wait the bit period plus 3/8 of the next
    175                         rx_clk = (one_baud_cnt / 2) + (one_baud_cnt * 3) / 8; 
    176                         rx_bits_remaining = 8;  
    177                         recv_state = RX_SAMPLE_BITS;
    178                         rx_samples = 0;
    179                         rx_sample_countdown = 5;
    180                     end else begin
    181                         // Pulse lasted less than half the period -
    182                         // not a valid transmission.
    183                         recv_state = RX_ERROR;
    184                     end
    185                 end
    186             end
    187             
    188             RX_SAMPLE_BITS: begin
    189                 // sample the rx line multiple times 
    190                 if (!rx_clk) begin
    191                     if (rx) begin
    192                         rx_samples =  rx_samples + 1'd1;
    193                     end
    194                     rx_clk = one_baud_cnt / 8;
    195                     rx_sample_countdown = rx_sample_countdown -1'd1;
    196                     recv_state = rx_sample_countdown ? RX_SAMPLE_BITS : RX_READ_BITS;
    197                 end
    198             end
    199             
    200             RX_READ_BITS: begin
    201                 if (!rx_clk) begin
    202                     // Should be finished sampling the pulse here.
    203                     // Update and prep for next
    204                     if (rx_samples > 3) begin
    205                         rx_data = {1'd1, rx_data[7:1]};
    206                     end else begin
    207                         rx_data = {1'd0, rx_data[7:1]};
    208                     end
    209                     
    210                     rx_clk = (one_baud_cnt * 3) / 8;
    211                     rx_samples = 0;
    212                     rx_sample_countdown = 5;
    213                     rx_bits_remaining = rx_bits_remaining - 1'd1;
    214                     
    215                     if(rx_bits_remaining)begin
    216                         recv_state = RX_SAMPLE_BITS;
    217                     end else begin
    218                         recv_state = RX_CHECK_STOP;
    219                         rx_clk = one_baud_cnt / 2;
    220                     end
    221                 end
    222             end
    223             
    224             RX_CHECK_STOP: begin
    225                 if (!rx_clk) begin
    226                     // Should resume half-way through the stop bit
    227                     // This should be high - if not, reject the
    228                     // transmission and signal an error.
    229                     recv_state = rx ? RX_RECEIVED : RX_ERROR;
    230                 end
    231             end
    232             
    233 
    234             
    235             RX_ERROR: begin
    236                 // There was an error receiving.
    237                 // Raises the recv_error flag for one clock
    238                 // cycle while in this state and then waits
    239                 // 2 bit periods before accepting another
    240                 // transmission.
    241                 rx_clk = 8 * sys_clk_freq / (baud_rate);
    242                 recv_state = RX_DELAY_RESTART;
    243             end
    244             
    245     // why is this state needed?  Why not go to idle and wait for next? 
    246     
    247             RX_DELAY_RESTART: begin
    248                 // Waits a set number of cycles before accepting
    249                 // another transmission.
    250                 recv_state = rx_clk ? RX_DELAY_RESTART : RX_IDLE;
    251             end
    252             
    253             
    254             RX_RECEIVED: begin
    255                 // Successfully received a byte.
    256                 // Raises the received flag for one clock
    257                 // cycle while in this state.
    258                 recv_state = RX_IDLE;
    259             end
    260             
    261         endcase
    262         
    263         
    264 //** Transmit state machine ***********************************
    265 
    266         case (tx_state)
    267             TX_IDLE: begin
    268                 if (transmit) begin
    269                     // If the transmit flag is raised in the idle
    270                     // state, start transmitting the current content
    271                     // of the tx_byte input.
    272                     tx_data = tx_byte;
    273                     // Send the initial, low pulse of 1 bit period
    274                     // to signal the start, followed by the data
    275                   //  tx_clk_divider =  clock_divide;                                
    276                     tx_clk = one_baud_cnt;
    277                     tx_out = 0;
    278                     tx_bits_remaining = 8;
    279                     tx_state = TX_SENDING;
    280                 end
    281             end
    282             
    283             TX_SENDING: begin
    284                 if (!tx_clk) begin
    285                     if (tx_bits_remaining) begin
    286                         tx_bits_remaining = tx_bits_remaining - 1'd1;
    287                         tx_out = tx_data[0];
    288                         tx_data = {1'b0, tx_data[7:1]};
    289                         tx_clk = one_baud_cnt;
    290                         tx_state = TX_SENDING;
    291                     end else begin
    292                         // Set delay to send out 2 stop bits.
    293                         tx_out = 1;
    294                         tx_clk = 16 * one_baud_cnt;// tx_countdown = 16;
    295                         tx_state = TX_DELAY_RESTART;
    296                     end
    297                 end
    298             end
    299             
    300             TX_DELAY_RESTART: begin
    301                 // Wait until tx_countdown reaches the end before
    302                 // we send another transmission. This covers the
    303                 // "stop bit" delay.
    304                 tx_state = tx_clk ? TX_DELAY_RESTART : TX_RECOVER;// TX_IDLE;
    305             end
    306             
    307             TX_RECOVER: begin
    308                 // Wait unitil the transmit line is deactivated.  This prevents repeated characters
    309                 tx_state = transmit ? TX_RECOVER : TX_IDLE;
    310            
    311             end
    312             
    313         endcase
    314     end
