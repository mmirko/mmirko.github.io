    173 func (op R2owaa) Assembler(arch *Arch, words []string) (string, error) {
    174 	opbits := arch.Opcodes_bits()
    175 	outbits := arch.Outputs_bits()
    176 	rom_word := arch.Max_word()
    177 
    178 	reg_num := 2
    179 	reg_num = reg_num << (arch.R - 1)
    180 
    181 	if len(words) != 2 {
    182 		return "", Prerror{"Wrong arguments number"}
    183 	}
    184 
    185 	result := ""
    186 	for i := 0; i < reg_num; i++ {
    187 		if words[0] == strings.ToLower(Get_register_name(i)) {
    188 			result += zeros_prefix(int(arch.R), get_binary(i))
    189 			break
    190 		}
    191 	}
    192 
    193 	if result == "" {
    194 		return "", Prerror{"Unknown register name " + words[0]}
    195 	}
    196 
    197 	if partial, err := Process_output(words[1], int(arch.M)); err == nil {
    198 		result += zeros_prefix(outbits, partial)
    199 	} else {
    200 		return "", Prerror{err.Error()}
    201 	}
    202 
    203 	for i := opbits + int(arch.R) + outbits; i < rom_word; i++ {
    204 		result += "0"
    205 	}
    206 
    207 	return result, nil
    208 }
