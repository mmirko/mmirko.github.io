     58 			}

     60 			result += "	end\n"

     62 	}
     63 	result += "	// Memory Write Block  \n"
     64 	result += "	// Write Operation we = 1 \n"
     65 	result += "	always @ (posedge clk) \n"
     66 	result += "	begin : MEM_WRITE \n"
