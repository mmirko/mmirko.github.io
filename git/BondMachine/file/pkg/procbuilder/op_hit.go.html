     57 func (op Hit) Op_instruction_verilog_state_machine(conf *Config, arch *Arch, rg *bmreqs.ReqRoot, flavor string) string {
     58 	rom_word := arch.Max_word()
     59 	opbits := arch.Opcodes_bits()
     60 	bar_num := arch.Shared_num("barrier")
     61 
     62 	reg_num := 1 << arch.R
     63 
     64 	result := ""
     65 	result += "					HIT: begin\n"
     66 	if arch.R == 1 {
     67 		result += "						case (current_instruction[" + strconv.Itoa(rom_word-opbits-1) + "])\n"
     68 	} else {
     69 		result += "						case (current_instruction[" + strconv.Itoa(rom_word-opbits-1) + ":" + strconv.Itoa(rom_word-opbits-int(arch.R)) + "])\n"
     70 	}
     71 
     72 	for i := 0; i < reg_num; i++ {
     73 		result += "						" + strings.ToUpper(Get_register_name(i)) + " : begin\n"
     74 
     75 		for j := 0; j < bar_num; j++ {
     76 			result += "						" + strings.ToUpper(Get_register_name(j)) + " : begin\n"
     77 
     78 			result += "							$display(\"CLR " + strings.ToUpper(Get_register_name(j)) + "\");\n"
     79 			result += "						end\n"
     80 		}
     81 
     82 		result += "							$display(\"CLR " + strings.ToUpper(Get_register_name(i)) + "\");\n"
     83 		result += "						end\n"
     84 	}
     85 	result += "						endcase\n"
     86 	result += NextInstruction(conf, arch, 6, "_pc + 1'b1")
     87 	result += "					end\n"
     88 	return result
