     42 func (op Jz) Op_instruction_verilog_state_machine(conf *Config, arch *Arch, rg *bmreqs.ReqRoot, flavor string) string {
     43 	result := ""
     44 	rom_word := arch.Max_word()
     45 	opbits := arch.Opcodes_bits()
     46 
     47 	reg_num := 1 << arch.R
     48 
     49 	result += "					JZ: begin\n"
     50 	if arch.R == 1 {
     51 		result += "						case (current_instruction[" + strconv.Itoa(rom_word-opbits-1) + "])\n"
     52 	} else {
     53 		result += "						case (current_instruction[" + strconv.Itoa(rom_word-opbits-1) + ":" + strconv.Itoa(rom_word-opbits-int(arch.R)) + "])\n"
     54 	}
     55 	for i := 0; i < reg_num; i++ {
     56 
     57 		if IsHwOptimizationSet(conf.HwOptimizations, HwOptimizations(OnlyDestRegs)) {
     58 			cp := arch.Tag
     59 			req := rg.Requirement(bmreqs.ReqRequest{Node: "/bm:cps/id:" + cp + "/opcodes:jz", T: bmreqs.ObjectSet, Name: "destregs", Value: Get_register_name(i), Op: bmreqs.OpCheck})
     60 			if req.Value == "false" {
     61 				continue
     62 			}
     63 		}
     64 
     65 		result += "							" + strings.ToUpper(Get_register_name(i)) + " : begin\n"
     66 		result += "								if(_" + strings.ToLower(Get_register_name(i)) + " == 'b0) begin\n"
     67 		result += NextInstruction(conf, arch, 8, "current_instruction["+strconv.Itoa(rom_word-opbits-1-int(arch.R))+":"+strconv.Itoa(rom_word-opbits-int(arch.O)-int(arch.R))+"]")
     68 		result += "								end\n"
     69 		result += "								else begin\n"
     70 		result += NextInstruction(conf, arch, 9, "_pc + 1'b1")
     71 		result += "								end\n"
     72 		result += "								$display(\"JZ " + strings.ToUpper(Get_register_name(i)) + " \",_" + strings.ToLower(Get_register_name(i)) + ");\n"
     73 		result += "							end\n"
     74 	}
     75 	result += "						endcase\n"
     76 	result += "					end\n"

     86 func (op Jz) Assembler(arch *Arch, words []string) (string, error) {
     87 	opbits := arch.Opcodes_bits()
     88 	rom_word := arch.Max_word()
     89 	osize := int(arch.O)
     90 
     91 	reg_num := 1 << arch.R
     92 
     93 	if len(words) != 2 {
     94 		return "", Prerror{"Wrong arguments number"}
     95 	}
     96 
     97 	result := ""
     98 	for i := 0; i < reg_num; i++ {
     99 		if words[0] == strings.ToLower(Get_register_name(i)) {
    100 			result += zeros_prefix(int(arch.R), get_binary(i))
    101 			break
    102 		}
    103 	}
    104 
    105 	if result == "" {
    106 		return "", Prerror{"Unknown register name " + words[0]}
    107 	}
    108 
    109 	if partial, err := Process_number(words[1]); err == nil {
    110 		result += zeros_prefix(osize, partial)
    111 	} else {
    112 		return "", Prerror{err.Error()}
    113 	}
    114 
    115 	for i := opbits + int(arch.R) + osize; i < rom_word; i++ {
    116 		result += "0"
    117 	}
    118 
    119 	return result, nil
    120 
    121 }

    133 func (op Jz) Simulate(vm *VM, instr string) error {
    134 	reg_bits := vm.Mach.R
    135 	reg := get_id(instr[:reg_bits])
    136 	jumpTo := get_id(instr[reg_bits : reg_bits+vm.Mach.O])
    137 	switch vm.Mach.Rsize {
    138 	case 8:
    139 		if vm.Registers[reg] == uint8(0) {
    140 			vm.Pc = uint64(jumpTo)
    141 		} else {
    142 			vm.Pc++
    143 		}
    144 	case 16:
    145 		if vm.Registers[reg] == uint16(0) {
    146 			vm.Pc = uint64(jumpTo)
    147 		} else {
    148 			vm.Pc++
    149 		}
    150 	case 32:
    151 		if vm.Registers[reg] == uint32(0) {
    152 			vm.Pc = uint64(jumpTo)
    153 		} else {
    154 			vm.Pc++
    155 		}
    156 	case 64:
    157 		if vm.Registers[reg] == uint64(0) {
    158 			vm.Pc = uint64(jumpTo)
    159 		} else {
    160 			vm.Pc++
    161 		}
    162 	default:
    163 		return errors.New("go simulation only works on 8,16,32 or 64 bits registers")
    164 	}
    165 	return nil
    166 }
