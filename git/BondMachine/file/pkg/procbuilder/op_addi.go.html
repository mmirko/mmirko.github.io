    104 func (op Addi) Op_instruction_verilog_state_machine(conf *Config, arch *Arch, rg *bmreqs.ReqRoot, flavor string) string {
    105 	rom_word := arch.Max_word()
    106 	opbits := arch.Opcodes_bits()
    107 
    108 	reg_num := 1 << arch.R
    109 
    110 	result := ""
    111 	result += "					ADDI: begin\n"
    112 	if arch.N > 0 {
    113 		if arch.R == 1 {
    114 			result += "						case (current_instruction[" + strconv.Itoa(rom_word-opbits-1) + "])\n"
    115 		} else {
    116 			result += "						case (current_instruction[" + strconv.Itoa(rom_word-opbits-1) + ":" + strconv.Itoa(rom_word-opbits-int(arch.R)) + "])\n"
    117 		}
    118 		for i := 0; i < reg_num; i++ {
    119 			result += "						" + strings.ToUpper(Get_register_name(i)) + " : begin\n"
    120 
    121 			sumstring := ""
    122 			for j := 0; j < int(arch.N); j++ {
    123 				sumstring += " + " + strings.ToLower(Get_input_name(j))
    124 			}
    125 			result += "							_" + strings.ToLower(Get_register_name(i)) + " <= #1 " + sumstring[2:len(sumstring)] + ";\n"
    126 			result += "							$display(\"ADDI " + strings.ToUpper(Get_register_name(i)) + " " + sumstring[2:len(sumstring)] + "\");\n"
    127 			result += "						end\n"
    128 		}
    129 		result += "						endcase\n"
    130 	} else {
    131 		result += "						$display(\"NOP\");\n"
    132 	}
    133 	result += NextInstruction(conf, arch, 6, "_pc + 1'b1")
    134 	result += "					end\n"

    142 func (op Addi) Assembler(arch *Arch, words []string) (string, error) {
    143 	opbits := arch.Opcodes_bits()
    144 	rom_word := arch.Max_word()
    145 
    146 	reg_num := 1 << arch.R
    147 
    148 	if len(words) != 1 {
    149 		return "", Prerror{"Wrong arguments number"}
    150 	}
    151 
    152 	result := ""
    153 	for i := 0; i < reg_num; i++ {
    154 		if words[0] == strings.ToLower(Get_register_name(i)) {
    155 			result += zeros_prefix(int(arch.R), get_binary(i))
    156 			break
    157 		}
    158 	}
    159 
    160 	if result == "" {
    161 		return "", Prerror{"Unknown register name " + words[0]}
    162 	}
    163 
    164 	for i := opbits + int(arch.R); i < rom_word; i++ {
    165 		result += "0"
    166 	}
    167 
    168 	return result, nil
    169 }
