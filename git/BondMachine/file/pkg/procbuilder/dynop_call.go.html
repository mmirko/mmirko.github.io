    139 func (op Call) Op_instruction_verilog_state_machine(conf *Config, arch *Arch, rg *bmreqs.ReqRoot, flavor string) string {
    140 	romWord := arch.Max_word()
    141 	opBits := arch.Opcodes_bits()
    142 	stackName := "restack" + arch.Tag + "_" + strconv.Itoa(op.s) + op.sn
    143 	locationBits := arch.O
    144 
    145 	dataSize := 0
    146 	switch arch.Modes[0] {
    147 	case "ha":
    148 		locationBits = arch.O
    149 		dataSize = int(arch.O)
    150 	case "vn":
    151 		locationBits = arch.L
    152 		dataSize = int(arch.L)
    153 	case "hy":
    154 		if arch.O > arch.L {
    155 			locationBits = arch.O
    156 			dataSize = int(arch.O) + 1
    157 		} else {
    158 			locationBits = arch.L
    159 			dataSize = int(arch.L) + 1
    160 		}
    161 	}
    162 
    163 	result := ""
    164 	if op.opType == OP_CALLA || op.opType == OP_CALLO {
    165 		result += "					" + strings.ToUpper(op.callName) + ": begin\n"
    166 		result += "						case (" + stackName + "SM)\n"
    167 		result += "						CALL1: begin\n"
    168 		result += "							if (!" + stackName + "senderAck) begin\n"
    169 		if arch.Modes[0] == "hy" {
    170 			result += "							     " + stackName + "senderData[" + strconv.Itoa(dataSize-1) + ":0] <= #1 { exec_mode, _pc + 1 };\n"
    171 		} else if arch.Modes[0] == "ha" {
    172 			result += "							     " + stackName + "senderData[" + strconv.Itoa(dataSize-1) + ":0] <= #1 _pc + 1;\n"
    173 		}
    174 		result += "							     " + stackName + "senderWrite <= #1 1'b1;\n"
    175 		result += "							     " + stackName + "SM <= CALL2;\n"
    176 		result += "							end\n"
    177 		result += "						end\n"
    178 		result += "						CALL2: begin\n"
    179 		result += "							if (" + stackName + "senderAck) begin\n"
    180 		result += "								" + stackName + "senderWrite <= #1 1'b0;\n"
    181 		if arch.Modes[0] == "hy" {
    182 			if op.opType == OP_CALLO {
    183 				result += "								exec_mode <= #1 1'b0;\n"
    184 			} else {
    185 				result += "								exec_mode <= #1 1'b1;\n"
    186 			}
    187 		}
    188 		if locationBits == 1 {
    189 			result += "								_pc <= #1 current_instruction[" + strconv.Itoa(romWord-opBits-1) + "];\n"
    190 			result += "								$display(\"" + strings.ToUpper(op.callName) + " \", current_instruction[" + strconv.Itoa(romWord-opBits-1) + "]);\n"
    191 		} else {
    192 			result += "								_pc <= #1 current_instruction[" + strconv.Itoa(romWord-opBits-1) + ":" + strconv.Itoa(romWord-opBits-int(locationBits)) + "];\n"
    193 			result += "								$display(\"" + strings.ToUpper(op.callName) + " \", current_instruction[" + strconv.Itoa(romWord-opBits-1) + ":" + strconv.Itoa(romWord-opBits-int(locationBits)) + "]);\n"
    194 		}
    195 		result += "								" + stackName + "SM <= CALL1;\n"
    196 		result += "							end\n"
    197 		result += "						end\n"
    198 		result += "						endcase\n"
    199 		result += "					end\n"
    200 	} else {
    201 		result += "					" + strings.ToUpper(op.callName) + ": begin\n"
    202 		result += "						if (" + stackName + "receiverAck && " + stackName + "receiverRead) begin\n"
    203 		result += "							" + stackName + "receiverRead <= #1 1'b0;\n"
    204 		result += "							_pc[" + strconv.Itoa(int(locationBits)-1) + ":0] <= #1 " + stackName + "receiverData[" + strconv.Itoa(int(locationBits)-1) + ":0];\n"
    205 		if arch.Modes[0] == "hy" {
    206 			result += "							exec_mode <= #1 " + stackName + "receiverData[" + strconv.Itoa(int(locationBits)) + "];\n"
    207 		}
    208 		if arch.Modes[0] == "hy" || arch.Modes[0] == "vn" {
    209 			result += "							vn_state <= FETCH;\n"
    210 		}
    211 		result += "						end\n"
    212 		result += "						else begin\n"
    213 		result += "							" + stackName + "receiverRead <= #1 1'b1;\n"
    214 		result += "						end\n"
    215 		result += "					end\n"
    216 	}
    217 
    218 	return result

    225 func (op Call) Assembler(arch *Arch, words []string) (string, error) {
    226 	opBits := arch.Opcodes_bits()
    227 	romWord := arch.Max_word()
    228 
    229 	locationBits := arch.O
    230 
    231 	switch op.opType {
    232 	case OP_CALLO:
    233 		locationBits = arch.O
    234 		if len(words) != 1 {
    235 			return "", Prerror{"Wrong arguments number"}
    236 		}
    237 	case OP_CALLA:
    238 		locationBits = arch.L
    239 		if len(words) != 1 {
    240 			return "", Prerror{"Wrong arguments number"}
    241 		}
    242 	case OP_RET:
    243 		locationBits = 0
    244 		if len(words) != 0 {
    245 			return "", Prerror{"Wrong arguments number"}
    246 		}
    247 	}
    248 
    249 	result := ""
    250 	if op.opType != OP_RET {
    251 		if partial, err := Process_number(words[0]); err == nil {
    252 			result += zeros_prefix(int(locationBits), partial)
    253 		} else {
    254 			return "", Prerror{err.Error()}
    255 		}
    256 	}
    257 	for i := opBits + int(locationBits); i < romWord; i++ {
    258 		result += "0"
    259 	}
    260 	return result, nil
    261 }
