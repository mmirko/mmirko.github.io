    125 func (op R2m) Assembler(arch *Arch, words []string) (string, error) {
    126 	opbits := arch.Opcodes_bits()
    127 	rom_word := arch.Max_word()
    128 	ramDepth := int(arch.L)
    129 
    130 	reg_num := 1 << arch.R
    131 
    132 	if len(words) != 2 {
    133 		return "", Prerror{"Wrong arguments number"}
    134 	}
    135 
    136 	result := ""
    137 	for i := 0; i < reg_num; i++ {
    138 		if words[0] == strings.ToLower(Get_register_name(i)) {
    139 			result += zeros_prefix(int(arch.R), get_binary(i))
    140 			break
    141 		}
    142 	}
    143 
    144 	if result == "" {
    145 		return "", Prerror{"Unknown register name " + words[0]}
    146 	}
    147 
    148 	if partial, err := Process_number(words[1]); err == nil {
    149 		result += zeros_prefix(ramDepth, partial)
    150 	} else {
    151 		return "", Prerror{err.Error()}
    152 	}
    153 
    154 	for i := opbits + int(arch.R) + ramDepth; i < rom_word; i++ {
    155 		result += "0"
    156 	}
    157 
    158 	return result, nil
    159 }

    234 func (Op R2m) HLAssemblerNormalize(arch *Arch, rg *bmreqs.ReqRoot, node string, line *bmline.BasmLine) (*bmline.BasmLine, error) {
    235 	switch line.Operation.GetValue() {
    236 	case "r2m":
    237 		regVal := line.Elements[0].GetValue()
    238 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regVal, Op: bmreqs.OpAdd})
    239 		return line, nil
    240 	case "mov":
    241 		regVal := line.Elements[1].GetValue()
    242 		addressing := line.Elements[0].GetMeta("ramaddressing")
    243 		switch addressing {
    244 		case "immediate":
    245 			location := line.Elements[0].GetMeta("location")
    246 			rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regVal, Op: bmreqs.OpAdd})
    247 			if regVal != "" && location != "" {
    248 				newLine := new(bmline.BasmLine)
    249 				newOp := new(bmline.BasmElement)
    250 				newOp.SetValue("r2m")
    251 				newLine.Operation = newOp
    252 				newArgs := make([]*bmline.BasmElement, 2)
    253 				newArg0 := new(bmline.BasmElement)
    254 				newArg0.BasmMeta = newArg0.SetMeta("type", "reg")
    255 				newArg0.SetValue(regVal)
    256 				newArgs[0] = newArg0
    257 				newArg1 := new(bmline.BasmElement)
    258 				newArg1.SetValue(location)
    259 				newArg1.BasmMeta = newArg1.SetMeta("type", "number")
    260 				newArgs[1] = newArg1
    261 				newLine.Elements = newArgs
    262 				return newLine, nil
    263 			}
    264 		case "symbol":
    265 			// The mov is a r2m, the symbol is kept unchanged because it will be resolved by the symbol resolver
    266 			symbol := line.Elements[0].GetMeta("symbol")
    267 			rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regVal, Op: bmreqs.OpAdd})
    268 			if regVal != "" && symbol != "" {
    269 				newLine := new(bmline.BasmLine)
    270 				newOp := new(bmline.BasmElement)
    271 				newOp.SetValue("r2m")
    272 				newLine.Operation = newOp
    273 				newArgs := make([]*bmline.BasmElement, 2)
    274 				newArg0 := new(bmline.BasmElement)
    275 				newArg0.BasmMeta = newArg0.SetMeta("type", "reg")
    276 				newArg0.SetValue(regVal)
    277 				newArgs[0] = newArg0
    278 				newArg1 := new(bmline.BasmElement)
    279 				newArg1.SetValue(symbol)
    280 				newArg1.BasmMeta = newArg1.SetMeta("type", "ram")
    281 				newArg1.BasmMeta = newArg1.SetMeta("ramaddressing", "symbol")
    282 				newArg1.BasmMeta = newArg1.SetMeta("symbol", symbol)
    283 				newArgs[1] = newArg1
    284 				newLine.Elements = newArgs
    285 				return newLine, nil
    286 			}
    287 		default:
    288 			return nil, errors.New("unknown addressing mode")
    289 		}
    290 	}
    291 	return nil, errors.New("HL Assembly normalize failed")
    292 }
