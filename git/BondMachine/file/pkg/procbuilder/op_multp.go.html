     52 	rom_word := arch.Max_word()

     54 
     55 	reg_num := 1 << arch.R
     56 
     57 	result := ""
     58 	result += "					MULTP: begin\n"
     59 	if arch.R == 1 {
     60 		result += "						case (current_instruction[" + strconv.Itoa(rom_word-opbits-1) + "])\n"

     62 		result += "						case (current_instruction[" + strconv.Itoa(rom_word-opbits-1) + ":" + strconv.Itoa(rom_word-opbits-int(arch.R)) + "])\n"

     64 	for i := 0; i < reg_num; i++ {
     65

     67 			cp := arch.Tag
     68 			req := rg.Requirement(bmreqs.ReqRequest{Node: "/bm:cps/id:" + cp + "/opcodes:multp", T: bmreqs.ObjectSet, Name: "destregs", Value: Get_register_name(i), Op: bmreqs.OpCheck})
     69 			if req.Value == "false" {
     70 				continue
     71 			}
     72 		}
     73 
     74 		result += "						" + strings.ToUpper(Get_register_name(i)) + " : begin\n"
     75 
     76 		if arch.R == 1 {
     77 			result += "							case (current_instruction[" + strconv.Itoa(rom_word-opbits-int(arch.R)-1) + "])\n"
     78 		} else {
     79 			result += "							case (current_instruction[" + strconv.Itoa(rom_word-opbits-int(arch.R)-1) + ":" + strconv.Itoa(rom_word-opbits-int(arch.R)-int(arch.R)) + "])\n"
     80 		}
     81 
     82 		for j := 0; j < reg_num; j++ {
     83

     85 				cp := arch.Tag

     90 			}
     91 
     92 			result += "							" + strings.ToUpper(Get_register_name(j)) + " : begin\n"
     93 			result += "							case (multp_" + arch.Tag + "_state)\n"
     94 			result += "							multp_" + arch.Tag + "_put : begin\n"
     95 			result += "								multp_" + arch.Tag + "_input_a <= #1 _" + strings.ToLower(Get_register_name(i)) + ";\n"
     96 			result += "								multp_" + arch.Tag + "_input_b <= #1 _" + strings.ToLower(Get_register_name(j)) + ";\n"

     98 			result += "							end\n"
     99 			result += "							multp_" + arch.Tag + "_get : begin\n"
    100 			result += "								_" + strings.ToLower(Get_register_name(i)) + " <= #1 multp_" + arch.Tag + "_output_z;\n"
    101 			result += "								multp_" + arch.Tag + "_state <= #1 multp_" + arch.Tag + "_put;\n"

    104 			result += "							endcase\n"

    121 func (op Multp) Assembler(arch *Arch, words []string) (string, error) {
    122 	opbits := arch.Opcodes_bits()
    123 	rom_word := arch.Max_word()
    124 
    125 	reg_num := 1 << arch.R
    126 
    127 	if len(words) != 2 {
    128 		return "", Prerror{"Wrong arguments number"}
    129 	}
    130 
    131 	result := ""
    132 	for i := 0; i < reg_num; i++ {
    133 		if words[0] == strings.ToLower(Get_register_name(i)) {
    134 			result += zeros_prefix(int(arch.R), get_binary(i))
    135 			break
    136 		}
    137 	}
    138 
    139 	if result == "" {
    140 		return "", Prerror{"Unknown register name " + words[0]}
    141 	}
    142 
    143 	partial := ""
    144 	for i := 0; i < reg_num; i++ {
    145 		if words[1] == strings.ToLower(Get_register_name(i)) {
    146 			partial += zeros_prefix(int(arch.R), get_binary(i))
    147 			break
    148 		}
    149 	}
    150 
    151 	if partial == "" {
    152 		return "", Prerror{"Unknown register name " + words[1]}
    153 	}
    154 
    155 	result += partial
    156 
    157 	for i := opbits + 2*int(arch.R); i < rom_word; i++ {
    158 		result += "0"
    159 	}
    160 
    161 	return result, nil
    162 }

    172 func (op Multp) Simulate(vm *VM, instr string) error {
    173 	regBits := vm.Mach.R
    174 	regDest := get_id(instr[:regBits])
    175 	regSrc := get_id(instr[regBits : regBits*2])
    176 
    177 	if *op.pipeline {
    178 		switch vm.Mach.Rsize {
    179 		case 8:
    180 			vm.Registers[regDest] = vm.Registers[regDest].(uint8) * vm.Registers[regSrc].(uint8)
    181 		case 16:
    182 			vm.Registers[regDest] = vm.Registers[regDest].(uint16) * vm.Registers[regSrc].(uint16)
    183 		case 32:
    184 			vm.Registers[regDest] = vm.Registers[regDest].(uint32) * vm.Registers[regSrc].(uint32)
    185 		case 64:
    186 			vm.Registers[regDest] = vm.Registers[regDest].(uint64) * vm.Registers[regSrc].(uint64)
    187 		default:
    188 			return errors.New("invalid register size")
    189 		}
    190 		vm.Pc = vm.Pc + 1
    191 		*op.pipeline = false
    192 	} else {
    193 		*op.pipeline = true
    194 	}
    195 	return nil
    196 }
