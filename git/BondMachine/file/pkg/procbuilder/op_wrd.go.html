    146 func (op Wrd) Assembler(arch *Arch, words []string) (string, error) {
    147 	opbits := arch.Opcodes_bits()
    148 	chso := Channel{}
    149 	channum := arch.Shared_num(chso.Shr_get_name())
    150 	chanbits := arch.Shared_bits(chso.Shr_get_name())
    151 	shortname := chso.Shortname()
    152 	rom_word := arch.Max_word()
    153 
    154 	reg_num := 1 << arch.R
    155 
    156 	if len(words) != 2 {
    157 		return "", Prerror{"Wrong arguments number"}
    158 	}
    159 
    160 	result := ""
    161 	for i := 0; i < reg_num; i++ {
    162 		if words[0] == strings.ToLower(Get_register_name(i)) {
    163 			result += zeros_prefix(int(arch.R), get_binary(i))
    164 			break
    165 		}
    166 	}
    167 
    168 	if result == "" {
    169 		return "", Prerror{"Unknown register name " + words[0]}
    170 	}
    171 
    172 	if partial, err := Process_shared(shortname, words[1], channum); err == nil {
    173 		result += zeros_prefix(chanbits, partial)
    174 	} else {
    175 		return "", Prerror{err.Error()}
    176 	}
    177 
    178 	for i := opbits + int(arch.R) + chanbits; i < rom_word; i++ {
    179 		result += "0"
    180 	}
    181 
    182 	return result, nil
    183 }
