    172 func (op LinearQuantizer) Assembler(arch *Arch, words []string) (string, error) {
    173 	opBits := arch.Opcodes_bits()
    174 	rom_word := arch.Max_word()
    175 
    176 	reg_num := 1 << arch.R
    177 
    178 	if len(words) != 2 {
    179 		return "", Prerror{"Wrong arguments number"}
    180 	}
    181 
    182 	result := ""
    183 	for i := 0; i < reg_num; i++ {
    184 		if words[0] == strings.ToLower(Get_register_name(i)) {
    185 			result += zeros_prefix(int(arch.R), get_binary(i))
    186 			break
    187 		}
    188 	}
    189 
    190 	if result == "" {
    191 		return "", Prerror{"Unknown register name " + words[0]}
    192 	}
    193 
    194 	partial := ""
    195 	for i := 0; i < reg_num; i++ {
    196 		if words[1] == strings.ToLower(Get_register_name(i)) {
    197 			partial += zeros_prefix(int(arch.R), get_binary(i))
    198 			break
    199 		}
    200 	}
    201 
    202 	if partial == "" {
    203 		return "", Prerror{"Unknown register name " + words[1]}
    204 	}
    205 
    206 	result += partial
    207 
    208 	for i := opBits + 2*int(arch.R); i < rom_word; i++ {
    209 		result += "0"
    210 	}
    211 
    212 	return result, nil
    213 }

    223 func (op LinearQuantizer) Simulate(vm *VM, instr string) error {
    224 	regBits := vm.Mach.R
    225 	regDest := get_id(instr[:regBits])
    226 	regSrc := get_id(instr[regBits : regBits*2])
    227 
    228 	sd := float64(uint64(1) << (vm.Mach.Rsize - 1))
    229 	sn := float64(op.max)
    230 	s := sd / sn
    231 
    232 	switch *op.pipeline {
    233 	case LQPUT:
    234 		if op.opType == LQMULT || op.opType == LQDIV {
    235 			*op.pipeline = LQCORR
    236 		} else {
    237 			*op.pipeline = LQGET
    238 		}
    239 	case LQCORR:
    240 		*op.pipeline = LQGET
    241 	case LQGET:
    242 		switch op.opType {
    243 		case LQADD:
    244 			if vm.Mach.Rsize <= 8 {
    245 				vm.Registers[regDest] = Int8bits(Int8FromBits(vm.Registers[regDest].(uint8)) + Int8FromBits(vm.Registers[regSrc].(uint8)))
    246 			} else if vm.Mach.Rsize <= 16 {
    247 				vm.Registers[regDest] = Int16bits(Int16FromBits(vm.Registers[regDest].(uint16)) + Int16FromBits(vm.Registers[regSrc].(uint16)))
    248 			} else if vm.Mach.Rsize <= 32 {
    249 				vm.Registers[regDest] = Int32bits(Int32FromBits(vm.Registers[regDest].(uint32)) + Int32FromBits(vm.Registers[regSrc].(uint32)))
    250 			} else if vm.Mach.Rsize <= 64 {
    251 				vm.Registers[regDest] = Int64bits(Int64FromBits(vm.Registers[regDest].(uint64)) + Int64FromBits(vm.Registers[regSrc].(uint64)))
    252 			} else {
    253 				return errors.New("invalid register size, must be <= 64")
    254 			}
    255 		case LQMULT:
    256 			if vm.Mach.Rsize <= 8 {
    257 				vm.Registers[regDest] = Int8bits(Int8FromBits(vm.Registers[regDest].(uint8)) * Int8FromBits(vm.Registers[regSrc].(uint8)) / int8(s))
    258 			} else if vm.Mach.Rsize <= 16 {
    259 				vm.Registers[regDest] = Int16bits(Int16FromBits(vm.Registers[regDest].(uint16)) * Int16FromBits(vm.Registers[regSrc].(uint16)) / int16(s))
    260 			} else if vm.Mach.Rsize <= 32 {
    261 				vm.Registers[regDest] = Int32bits(Int32FromBits(vm.Registers[regDest].(uint32)) * Int32FromBits(vm.Registers[regSrc].(uint32)) / int32(s))
    262 			} else if vm.Mach.Rsize <= 64 {
    263 				vm.Registers[regDest] = Int64bits(Int64FromBits(vm.Registers[regDest].(uint64)) * Int64FromBits(vm.Registers[regSrc].(uint64)) / int64(s))
    264 			} else {
    265 				return errors.New("invalid register size, must be <= 64")
    266 			}
    267 		case LQDIV:
    268 			if vm.Mach.Rsize <= 8 {
    269 				vm.Registers[regDest] = Int8bits(Int8FromBits(vm.Registers[regDest].(uint8)) * int8(s) / Int8FromBits(vm.Registers[regSrc].(uint8)))
    270 			} else if vm.Mach.Rsize <= 16 {
    271 				vm.Registers[regDest] = Int16bits(Int16FromBits(vm.Registers[regDest].(uint16)) * int16(s) / Int16FromBits(vm.Registers[regSrc].(uint16)))
    272 			} else if vm.Mach.Rsize <= 32 {
    273 				vm.Registers[regDest] = Int32bits(Int32FromBits(vm.Registers[regDest].(uint32)) * int32(s) / Int32FromBits(vm.Registers[regSrc].(uint32)))
    274 			} else if vm.Mach.Rsize <= 64 {
    275 				vm.Registers[regDest] = Int64bits(Int64FromBits(vm.Registers[regDest].(uint64)) * int64(s) / Int64FromBits(vm.Registers[regSrc].(uint64)))
    276 			} else {
    277 				return errors.New("invalid register size, must be <= 64")
    278 			}
    279 		}
    280 		vm.Pc = vm.Pc + 1
    281 		*op.pipeline = LQPUT
    282 	}
    283 	return nil
    284 }

    321 func (op LinearQuantizer) Op_instruction_verilog_extra_modules(arch *Arch, flavor string) ([]string, []string) {
    322 	result := "\n\n"
    323 	var moduleName string
    324 
    325 	result += "module " + op.lqName + "_" + arch.Tag + "(\n"
    326 	result += "        input_a,\n"
    327 	result += "        input_b,\n"
    328 	result += "        output_z);\n"
    329 	result += "\n"
    330 	result += "  input signed    [" + strconv.Itoa(int(arch.Rsize)-1) + ":0] input_a;\n"
    331 	result += "  input signed    [" + strconv.Itoa(int(arch.Rsize)-1) + ":0] input_b;\n"
    332 	result += "  output signed   [" + strconv.Itoa(int(arch.Rsize)-1) + ":0] output_z;\n"
    333 	switch op.opType {
    334 	case LQADD:
    335 		result += "  assign output_z = input_a + input_b;\n"
    336 		moduleName = "adder"
    337 	case LQMULT:
    338 		result += "  assign output_z = input_a * input_b;\n"
    339 		moduleName = "multiplier"
    340 	case LQDIV:
    341 		result += "  assign output_z = input_a / input_b;\n"
    342 		moduleName = "divider"
    343 	}
    344 	result += "\n"
    345 	result += "endmodule\n"
    346 
    347 	moduleNames := []string{moduleName}
    348 	moduleCodes := []string{result}
    349 
    350 	if op.opType == LQDIV || op.opType == LQMULT {
    351 		var correctionName string
    352 		correction := "\n\n"
    353 		correction += "module " + op.lqName + "_correction_" + arch.Tag + "(\n"
    354 		correction += "        input_corr,\n"
    355 		correction += "        output_corr);\n"
    356 		correction += "\n"
    357 		correction += "  input signed    [" + strconv.Itoa(int(arch.Rsize)-1) + ":0] input_corr;\n"
    358 		correction += "  output signed   [" + strconv.Itoa(int(arch.Rsize)-1) + ":0] output_corr;\n"
    359 		sd := float64(uint64(1) << (arch.Rsize - 1))
    360 		sn := float64(op.max)
    361 		s := sd / sn
    362 		corr := fmt.Sprint(uint(s))
    363 		correction += "  parameter signed [" + strconv.Itoa(int(arch.Rsize)-1) + ":0] CORRECTION = " + strconv.Itoa(int(arch.Rsize)) + "'d" + corr + ";\n"
    364 		switch op.opType {
    365 		case LQMULT:
    366 			correction += "  assign output_corr = input_corr / CORRECTION;\n"
    367 			correctionName = "multiplier_correction"
    368 		case LQDIV:
    369 			correction += "  assign output_corr = input_corr * CORRECTION;\n"
    370 			correctionName = "divider_correction"
    371 		}
    372 		correction += "\n"
    373 		correction += "endmodule\n"
    374 		moduleNames = append(moduleNames, correctionName)
    375 		moduleCodes = append(moduleCodes, correction)
    376 	}
    377 
    378 	return moduleNames, moduleCodes
    379 }

    399 func (Op LinearQuantizer) HLAssemblerNormalize(arch *Arch, rg *bmreqs.ReqRoot, node string, line *bmline.BasmLine) (*bmline.BasmLine, error) {
    400 	switch line.Operation.GetValue() {
    401 	case Op.lqName:
    402 		regDst := line.Elements[0].GetValue()
    403 		regSrc := line.Elements[1].GetValue()
    404 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regDst, Op: bmreqs.OpAdd})
    405 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regSrc, Op: bmreqs.OpAdd})
    406 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "opcodes", Value: Op.lqName, Op: bmreqs.OpAdd})
    407 		rg.Requirement(bmreqs.ReqRequest{Node: node + "/opcodes:" + Op.lqName, T: bmreqs.ObjectSet, Name: "destregs", Value: regDst, Op: bmreqs.OpAdd})
    408 		rg.Requirement(bmreqs.ReqRequest{Node: node + "/opcodes:" + Op.lqName, T: bmreqs.ObjectSet, Name: "sourceregs", Value: regSrc, Op: bmreqs.OpAdd})
    409 		return line, nil
    410 	}
    411 	return nil, errors.New("HL Assembly normalize failed")
