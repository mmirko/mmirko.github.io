    152 func (op M2rri) Op_instruction_verilog_footer(arch *Arch, flavor string) string {
    153 	result := ""
    154 
    155 	rom_word := arch.Max_word()
    156 	opbits := arch.Opcodes_bits()

    159 	if arch.OnlyOne(op.Op_get_name(), []string{"r2mri", "r2m", "m2r", "m2rri"}) {

    175 		ramAddr = " (current_instruction[" + strconv.Itoa(rom_word-1) + ":" + strconv.Itoa(rom_word-opbits) + "]==M2R) ? addr_ram_m2r : " + ramAddr
    176 	}
    177 
    178 	if arch.HasOp("m2rri") {
    179 		ramAddr = " (current_instruction[" + strconv.Itoa(rom_word-1) + ":" + strconv.Itoa(rom_word-opbits) + "]==M2RRI) ? addr_ram_m2rri: " + ramAddr

    193 func (op M2rri) Assembler(arch *Arch, words []string) (string, error) {
    194 	opbits := arch.Opcodes_bits()
    195 	romWord := arch.Max_word()
    196 
    197 	regNum := 1 << arch.R
    198 
    199 	if len(words) != 2 {
    200 		return "", Prerror{"Wrong arguments number"}
    201 	}
    202 
    203 	result := ""
    204 	for i := 0; i < regNum; i++ {
    205 		if words[0] == strings.ToLower(Get_register_name(i)) {
    206 			result += zeros_prefix(int(arch.R), get_binary(i))
    207 			break
    208 		}
    209 	}
    210 
    211 	if result == "" {
    212 		return "", Prerror{"Unknown register name " + words[0]}
    213 	}
    214 
    215 	partial := ""
    216 	for i := 0; i < regNum; i++ {
    217 		if words[1] == strings.ToLower(Get_register_name(i)) {
    218 			partial += zeros_prefix(int(arch.R), get_binary(i))
    219 			break
    220 		}
    221 	}
    222 
    223 	if partial == "" {
    224 		return "", Prerror{"Unknown register name " + words[1]}
    225 	}
    226 
    227 	result += partial
    228 
    229 	for i := opbits + 2*int(arch.R); i < romWord; i++ {
    230 		result += "0"
    231 	}
    232 
    233 	return result, nil
    234 }

    244 func (op M2rri) Simulate(vm *VM, instr string) error {
    245 	reg_bits := vm.Mach.R
    246 	regDest := get_id(instr[:reg_bits])
    247 	regSrc := get_id(instr[reg_bits : reg_bits*2])
    248 
    249 	if vm.Mach.Rsize <= 8 {
    250 		loc := int(vm.Registers[regSrc].(uint8))
    251 		if loc < len(vm.Mach.Program.Slocs) {
    252 			vm.Registers[regDest] = uint8(get_id(vm.Mach.Program.Slocs[loc]))
    253 		} else {
    254 			vm.Registers[regDest] = uint8(get_id(vm.Mach.Data.Vars[loc-len(vm.Mach.Program.Slocs)]))
    255 		}
    256 	} else if vm.Mach.Rsize <= 16 {
    257 		loc := int(vm.Registers[regSrc].(uint16))
    258 		if loc < len(vm.Mach.Program.Slocs) {
    259 			vm.Registers[regDest] = uint16(get_id(vm.Mach.Program.Slocs[loc]))
    260 		} else {
    261 			vm.Registers[regDest] = uint16(get_id(vm.Mach.Data.Vars[loc-len(vm.Mach.Program.Slocs)]))
    262 		}
    263 	} else if vm.Mach.Rsize <= 32 {
    264 		loc := int(vm.Registers[regSrc].(uint32))
    265 		if loc < len(vm.Mach.Program.Slocs) {
    266 			vm.Registers[regDest] = uint32(get_id(vm.Mach.Program.Slocs[loc]))
    267 		} else {
    268 			vm.Registers[regDest] = uint32(get_id(vm.Mach.Data.Vars[loc-len(vm.Mach.Program.Slocs)]))
    269 		}
    270 	} else if vm.Mach.Rsize <= 64 {
    271 		loc := int(vm.Registers[regSrc].(uint64))
    272 		if loc < len(vm.Mach.Program.Slocs) {
    273 			vm.Registers[regDest] = uint64(get_id(vm.Mach.Program.Slocs[loc]))
    274 		} else {
    275 			vm.Registers[regDest] = uint64(get_id(vm.Mach.Data.Vars[loc-len(vm.Mach.Program.Slocs)]))
    276 		}
    277 	} else {
    278 		return errors.New("invalid register size, must be <= 64")
    279 	}
    280 
    281 	vm.Pc = vm.Pc + 1
    282 	return nil
    283 }

    305 func (Op M2rri) AbstractAssembler(arch *Arch, words []string) ([]UsageNotify, error) {
    306 	seq0, types0 := Sequence_to_0(words[0])
    307 	seq1, types1 := Sequence_to_0(words[1])
    308 
    309 	if len(seq0) > 0 && types0 == O_REGISTER && len(seq1) > 0 && types1 == O_INPUT {
    310 
    311 		result := make([]UsageNotify, 2+len(seq1))
    312 		newnot0 := UsageNotify{C_OPCODE, "m2rri", I_NIL}
    313 		result[0] = newnot0
    314 		newnot1 := UsageNotify{C_REGSIZE, S_NIL, len(seq0)}
    315 		result[1] = newnot1
    316 
    317 		for i, _ := range seq1 {
    318 			result[i+2] = UsageNotify{C_INPUT, S_NIL, i + 1}
    319 		}
    320 
    321 		return result, nil
    322 
    323 	}
    324 
    325 	return []UsageNotify{}, errors.New("Wrong parameters")
    326 }

    342 func (Op M2rri) HLAssemblerNormalize(arch *Arch, rg *bmreqs.ReqRoot, node string, line *bmline.BasmLine) (*bmline.BasmLine, error) {
    343 	switch line.Operation.GetValue() {
    344 	case "m2rri":
    345 		regNeed := line.Elements[1].GetValue()
    346 		regDest := line.Elements[0].GetValue()
    347 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regNeed, Op: bmreqs.OpAdd})
    348 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regDest, Op: bmreqs.OpAdd})
    349 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "opcodes", Value: "m2rri", Op: bmreqs.OpAdd})
    350 		rg.Requirement(bmreqs.ReqRequest{Node: node + "/opcodes:m2rri", T: bmreqs.ObjectSet, Name: "destregs", Value: regDest, Op: bmreqs.OpAdd})
    351 		rg.Requirement(bmreqs.ReqRequest{Node: node + "/opcodes:m2rri", T: bmreqs.ObjectSet, Name: "sourceregs", Value: regNeed, Op: bmreqs.OpAdd})
    352 		return line, nil
    353 	case "mov":
    354 		regDest := line.Elements[0].GetValue()
    355 		regNeed := line.Elements[1].GetMeta("ramregister")
    356 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regDest, Op: bmreqs.OpAdd})
    357 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regNeed, Op: bmreqs.OpAdd})
    358 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "opcodes", Value: "m2rri", Op: bmreqs.OpAdd})
    359 		rg.Requirement(bmreqs.ReqRequest{Node: node + "/opcodes:m2rri", T: bmreqs.ObjectSet, Name: "destregs", Value: regDest, Op: bmreqs.OpAdd})
    360 		rg.Requirement(bmreqs.ReqRequest{Node: node + "/opcodes:m2rri", T: bmreqs.ObjectSet, Name: "sourceregs", Value: regNeed, Op: bmreqs.OpAdd})
    361 		if regDest != "" && regNeed != "" {
    362 			newLine := new(bmline.BasmLine)
    363 			newOp := new(bmline.BasmElement)
    364 			newOp.SetValue("m2rri")
    365 			newLine.Operation = newOp
    366 			newArgs := make([]*bmline.BasmElement, 2)
    367 			newArg0 := new(bmline.BasmElement)
    368 			newArg0.BasmMeta = newArg0.SetMeta("type", "reg")
    369 			newArg0.SetValue(regDest)
    370 			newArgs[0] = newArg0
    371 			newArg1 := new(bmline.BasmElement)
    372 			newArg1.SetValue(regNeed)
    373 			newArg1.BasmMeta = newArg1.SetMeta("type", "reg")
    374 			newArgs[1] = newArg1
    375 			newLine.Elements = newArgs
    376 			return newLine, nil
    377 		}
    378 	}
    379 	return nil, errors.New("HL Assembly normalize failed")
    380 }
