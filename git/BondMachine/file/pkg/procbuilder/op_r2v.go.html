    164 func (op R2v) Assembler(arch *Arch, words []string) (string, error) {
    165 	opbits := arch.Opcodes_bits()
    166 	rom_word := arch.Max_word()
    167 	ramDepth := 8
    168 	boxes := soLists("vtextmem", arch.Shared_constraints, 5)
    169 loop:
    170 	for _, box := range boxes {
    171 		if box[0] == arch.Tag {
    172 			memWidth, _ := strconv.Atoi(box[3])
    173 			memHeight, _ := strconv.Atoi(box[4])
    174 			memDepth := memWidth * memHeight
    175 			ramDepth = Needed_bits(memDepth)
    176 			break loop
    177 		}
    178 	}
    179 	// fmt.Println("ramDepth", ramDepth)
    180 	reg_num := 1 << arch.R
    181 
    182 	if len(words) != 2 {
    183 		return "", Prerror{"Wrong arguments number"}
    184 	}
    185 
    186 	result := ""
    187 	for i := 0; i < reg_num; i++ {
    188 		if words[0] == strings.ToLower(Get_register_name(i)) {
    189 			result += zeros_prefix(int(arch.R), get_binary(i))
    190 			break
    191 		}
    192 	}
    193 
    194 	if result == "" {
    195 		return "", Prerror{"Unknown register name " + words[0]}
    196 	}
    197 
    198 	if partial, err := Process_number(words[1]); err == nil {
    199 		result += zeros_prefix(ramDepth, partial)
    200 	} else {
    201 		return "", Prerror{err.Error()}
    202 	}
    203 
    204 	for i := opbits + int(arch.R) + ramDepth; i < rom_word; i++ {
    205 		result += "0"
    206 	}
    207 
    208 	return result, nil
    209 }

    298 func (Op R2v) HLAssemblerNormalize(arch *Arch, rg *bmreqs.ReqRoot, node string, line *bmline.BasmLine) (*bmline.BasmLine, error) {
    299 	switch line.Operation.GetValue() {
    300 	case "r2v":
    301 		regNeed := line.Elements[0].GetValue()
    302 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regNeed, Op: bmreqs.OpAdd})
    303 		return line, nil
    304 	case "mov":
    305 		regVal := line.Elements[1].GetValue()
    306 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regVal, Op: bmreqs.OpAdd})
    307 		port := line.Elements[0].GetMeta("soport")
    308 		if regVal != "" && port != "" {
    309 			newLine := new(bmline.BasmLine)
    310 			newOp := new(bmline.BasmElement)
    311 			newOp.SetValue("r2v")
    312 			newLine.Operation = newOp
    313 			newArgs := make([]*bmline.BasmElement, 2)
    314 			newArg0 := new(bmline.BasmElement)
    315 			newArg0.BasmMeta = newArg0.SetMeta("type", "reg")
    316 			newArg0.SetValue(regVal)
    317 			newArgs[0] = newArg0
    318 			newArg1 := new(bmline.BasmElement)
    319 			newArg1.SetValue(port)
    320 			newArg1.BasmMeta = newArg1.SetMeta("type", "number")
    321 			newArgs[1] = newArg1
    322 			newLine.Elements = newArgs
    323 			return newLine, nil
    324 		}
    325 	}
    326 	return nil, errors.New("HL Assembly normalize failed")
    327 }
