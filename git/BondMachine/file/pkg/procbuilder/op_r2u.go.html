    137 func (op R2u) Assembler(arch *Arch, words []string) (string, error) {
    138 	opBits := arch.Opcodes_bits()
    139 	uSo := Uart{}
    140 	uartNum := arch.Shared_num(uSo.Shr_get_name())
    141 	uartBits := arch.Shared_bits(uSo.Shr_get_name())
    142 	shortName := uSo.Shortname()
    143 	romWord := arch.Max_word()
    144 
    145 	regNum := 1 << arch.R
    146 
    147 	if len(words) != 2 {
    148 		return "", Prerror{"Wrong arguments number"}
    149 	}
    150 
    151 	result := ""
    152 	for i := 0; i < regNum; i++ {
    153 		if words[0] == strings.ToLower(Get_register_name(i)) {
    154 			result += zeros_prefix(int(arch.R), get_binary(i))
    155 			break
    156 		}
    157 	}
    158 
    159 	if result == "" {
    160 		return "", Prerror{"Unknown register name " + words[0]}
    161 	}
    162 	if partial, err := Process_shared(shortName, words[1], uartNum); err == nil {
    163 		result += zeros_prefix(uartBits, partial)
    164 	} else {
    165 		return "", Prerror{err.Error()}
    166 	}
    167 
    168 	for i := opBits + int(arch.R) + uartBits; i < romWord; i++ {
    169 		result += "0"
    170 	}
    171 
    172 	return result, nil
    173 }

    264 func (Op R2u) HLAssemblerNormalize(arch *Arch, rg *bmreqs.ReqRoot, node string, line *bmline.BasmLine) (*bmline.BasmLine, error) {
    265 	switch line.Operation.GetValue() {
    266 	case "r2u":
    267 		regVal := line.Elements[0].GetValue()
    268 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regVal, Op: bmreqs.OpAdd})
    269 		soVal := line.Elements[0].GetValue()
    270 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "sos", Value: soVal, Op: bmreqs.OpAdd})
    271 		return line, nil
    272 	case "touart":
    273 		regVal := line.Elements[0].GetValue()
    274 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regVal, Op: bmreqs.OpAdd})
    275 		soVal := "u0" // Push implicitely uses the first stack
    276 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "sos", Value: soVal, Op: bmreqs.OpAdd})
    277 		if regVal != "" && soVal != "" {
    278 			newLine := new(bmline.BasmLine)
    279 			newOp := new(bmline.BasmElement)
    280 			newOp.SetValue("r2u")
    281 			newLine.Operation = newOp
    282 			newArgs := make([]*bmline.BasmElement, 2)
    283 			newArg0 := new(bmline.BasmElement)
    284 			newArg0.BasmMeta = newArg0.SetMeta("type", "reg")
    285 			newArg0.SetValue(regVal)
    286 			newArgs[0] = newArg0
    287 			newArg1 := new(bmline.BasmElement)
    288 			newArg1.SetValue(soVal)
    289 			newArg1.BasmMeta = newArg1.SetMeta("type", "somov")
    290 			newArg1.BasmMeta = newArg1.SetMeta("sotype", "u")
    291 			newArgs[1] = newArg1
    292 			newLine.Elements = newArgs
    293 			return newLine, nil
    294 		}
    295 	case "mov":
    296 		regVal := line.Elements[1].GetValue()
    297 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regVal, Op: bmreqs.OpAdd})
    298 		soVal := line.Elements[0].GetValue()
    299 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "sos", Value: soVal, Op: bmreqs.OpAdd})
    300 		if regVal != "" && soVal != "" {
    301 			newLine := new(bmline.BasmLine)
    302 			newOp := new(bmline.BasmElement)
    303 			newOp.SetValue("r2u")
    304 			newLine.Operation = newOp
    305 			newArgs := make([]*bmline.BasmElement, 2)
    306 			newArg0 := new(bmline.BasmElement)
    307 			newArg0.BasmMeta = newArg0.SetMeta("type", "reg")
    308 			newArg0.SetValue(regVal)
    309 			newArgs[0] = newArg0
    310 			newArg1 := new(bmline.BasmElement)
    311 			newArg1.SetValue(soVal)
    312 			newArg1.BasmMeta = newArg1.SetMeta("type", "somov")
    313 			newArg1.BasmMeta = newArg1.SetMeta("sotype", "u")
