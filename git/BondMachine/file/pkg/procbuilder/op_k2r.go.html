    129 func (op K2r) Assembler(arch *Arch, words []string) (string, error) {
    130 	opBits := arch.Opcodes_bits()
    131 	kSo := Kbd{}
    132 	kbdNum := arch.Shared_num(kSo.Shr_get_name())
    133 	kbdBits := arch.Shared_bits(kSo.Shr_get_name())
    134 	shortName := kSo.Shortname()
    135 	romWord := arch.Max_word()
    136 
    137 	regNum := 1 << arch.R
    138 
    139 	if len(words) != 2 {
    140 		return "", Prerror{"Wrong arguments number"}
    141 	}
    142 
    143 	result := ""
    144 	for i := 0; i < regNum; i++ {
    145 		if words[0] == strings.ToLower(Get_register_name(i)) {
    146 			result += zeros_prefix(int(arch.R), get_binary(i))
    147 			break
    148 		}
    149 	}
    150 
    151 	if result == "" {
    152 		return "", Prerror{"Unknown register name " + words[0]}
    153 	}
    154 
    155 	if partial, err := Process_shared(shortName, words[1], kbdNum); err == nil {
    156 		result += zeros_prefix(kbdBits, partial)
    157 	} else {
    158 		return "", Prerror{err.Error()}
    159 	}
    160 
    161 	for i := opBits + int(arch.R) + kbdBits; i < romWord; i++ {
    162 		result += "0"
    163 	}
    164 
    165 	return result, nil
    166 }
