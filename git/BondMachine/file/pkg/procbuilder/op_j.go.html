    114 func (op J) Assembler(arch *Arch, words []string) (string, error) {
    115 	opbits := arch.Opcodes_bits()
    116 	rom_word := arch.Max_word()
    117 
    118 	locationBits := arch.O
    119 
    120 	switch arch.Modes[0] {
    121 	case "ha":
    122 		locationBits = arch.O
    123 	case "vn":
    124 		locationBits = arch.L
    125 	case "hy":
    126 		if arch.O > arch.L {
    127 			locationBits = arch.O
    128 		} else {
    129 			locationBits = arch.L
    130 		}
    131 	}
    132 
    133 	if len(words) != 1 {
    134 		return "", Prerror{"Wrong arguments number"}
    135 	}
    136 
    137 	result := ""
    138 	if partial, err := Process_number(words[0]); err == nil {
    139 		result += zeros_prefix(int(locationBits), partial)
    140 	} else {
    141 		return "", Prerror{err.Error()}
    142 	}
    143 
    144 	for i := opbits + int(locationBits); i < rom_word; i++ {
    145 		result += "0"
    146 	}
    147 
    148 	return result, nil
    149 }
