    373 func (op Chw) Assembler(arch *Arch, words []string) (string, error) {
    374 	opbits := arch.Opcodes_bits()
    375 	rom_word := arch.Max_word()
    376 
    377 	reg_num := 1 << arch.R
    378 
    379 	if len(words) != 1 {
    380 		return "", Prerror{"Wrong arguments number"}
    381 	}
    382 
    383 	result := ""
    384 	for i := 0; i < reg_num; i++ {
    385 		if words[0] == strings.ToLower(Get_register_name(i)) {
    386 			result += zeros_prefix(int(arch.R), get_binary(i))
    387 			break
    388 		}
    389 	}
    390 
    391 	if result == "" {
    392 		return "", Prerror{"Unknown register name " + words[0]}
    393 	}
    394 
    395 	for i := opbits + int(arch.R); i < rom_word; i++ {
    396 		result += "0"
    397 	}
    398 
    399 	return result, nil
    400 }
