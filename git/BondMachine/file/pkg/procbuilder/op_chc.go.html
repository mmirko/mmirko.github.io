    398 func (op Chc) Assembler(arch *Arch, words []string) (string, error) {
    399 	opbits := arch.Opcodes_bits()
    400 	rom_word := arch.Max_word()
    401 
    402 	reg_num := 1 << arch.R
    403 
    404 	if len(words) != 2 {
    405 		return "", Prerror{"Wrong arguments number"}
    406 	}
    407 
    408 	result := ""
    409 	for i := 0; i < reg_num; i++ {
    410 		if words[0] == strings.ToLower(Get_register_name(i)) {
    411 			result += zeros_prefix(int(arch.R), get_binary(i))
    412 			break
    413 		}
    414 	}
    415 
    416 	if result == "" {
    417 		return "", Prerror{"Unknown register name " + words[0]}
    418 	}
    419 
    420 	partial := ""
    421 	for i := 0; i < reg_num; i++ {
    422 		if words[1] == strings.ToLower(Get_register_name(i)) {
    423 			partial += zeros_prefix(int(arch.R), get_binary(i))
    424 			break
    425 		}
    426 	}
    427 
    428 	if partial == "" {
    429 		return "", Prerror{"Unknown register name " + words[1]}
    430 	}
    431 
    432 	result += partial
    433 
    434 	for i := opbits + 2*int(arch.R); i < rom_word; i++ {
    435 		result += "0"
    436 	}
    437 
    438 	return result, nil
    439 }
