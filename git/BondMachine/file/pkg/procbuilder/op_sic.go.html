    120 func (op Sic) Assembler(arch *Arch, words []string) (string, error) {
    121 	opbits := arch.Opcodes_bits()
    122 	inpbits := arch.Inputs_bits()
    123 	rom_word := arch.Max_word()
    124 
    125 	reg_num := 2
    126 	reg_num = reg_num << (arch.R - 1)
    127 
    128 	if len(words) != 2 {
    129 		return "", Prerror{"Wrong arguments number"}
    130 	}
    131 
    132 	result := ""
    133 	for i := 0; i < reg_num; i++ {
    134 		if words[0] == strings.ToLower(Get_register_name(i)) {
    135 			result += zeros_prefix(int(arch.R), get_binary(i))
    136 			break
    137 		}
    138 	}
    139 
    140 	if result == "" {
    141 		return "", Prerror{"Unknown register name " + words[0]}
    142 	}
    143 
    144 	if partial, err := Process_input(words[1], int(arch.N)); err == nil {
    145 		result += zeros_prefix(inpbits, partial)
    146 	} else {
    147 		return "", Prerror{err.Error()}
    148 	}
    149 
    150 	for i := opbits + int(arch.R) + inpbits; i < rom_word; i++ {
    151 		result += "0"
    152 	}
    153 
    154 	return result, nil
    155 }
