     88 func (op Incc) Assembler(arch *Arch, words []string) (string, error) {
     89 	opbits := arch.Opcodes_bits()
     90 	rom_word := arch.Max_word()
     91 
     92 	reg_num := 1 << arch.R
     93 
     94 	if len(words) != 1 {
     95 		return "", Prerror{"Wrong arguments number"}
     96 	}
     97 
     98 	result := ""
     99 	for i := 0; i < reg_num; i++ {
    100 		if words[0] == strings.ToLower(Get_register_name(i)) {
    101 			result += zeros_prefix(int(arch.R), get_binary(i))
    102 			break
    103 		}
    104 	}
    105 
    106 	if result == "" {
    107 		return "", Prerror{"Unknown register name " + words[0]}
    108 	}
    109 
    110 	for i := opbits + int(arch.R); i < rom_word; i++ {
    111 		result += "0"
    112 	}
    113 
    114 	return result, nil
    115 }
