     83 func (op Or) Assembler(arch *Arch, words []string) (string, error) {
     84 	opbits := arch.Opcodes_bits()
     85 	rom_word := arch.Max_word()
     86 
     87 	reg_num := 1 << arch.R
     88 
     89 	if len(words) != 2 {
     90 		return "", Prerror{"Wrong arguments number"}
     91 	}
     92 
     93 	result := ""
     94 	for i := 0; i < reg_num; i++ {
     95 		if words[0] == strings.ToLower(Get_register_name(i)) {
     96 			result += zeros_prefix(int(arch.R), get_binary(i))
     97 			break
     98 		}
     99 	}
    100 
    101 	if result == "" {
    102 		return "", Prerror{"Unknown register name " + words[0]}
    103 	}
    104 
    105 	partial := ""
    106 	for i := 0; i < reg_num; i++ {
    107 		if words[1] == strings.ToLower(Get_register_name(i)) {
    108 			partial += zeros_prefix(int(arch.R), get_binary(i))
    109 			break
    110 		}
    111 	}
    112 
    113 	if partial == "" {
    114 		return "", Prerror{"Unknown register name " + words[1]}
    115 	}
    116 
    117 	result += partial
    118 
    119 	for i := opbits + 2*int(arch.R); i < rom_word; i++ {
    120 		result += "0"
    121 	}
    122 
    123 	return result, nil
    124 }
