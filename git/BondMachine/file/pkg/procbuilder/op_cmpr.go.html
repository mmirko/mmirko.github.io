     74 		}

    108 func (op Cmpr) Assembler(arch *Arch, words []string) (string, error) {
    109 	opBits := arch.Opcodes_bits()
    110 	rom_word := arch.Max_word()
    111 
    112 	reg_num := 1 << arch.R
    113 
    114 	if len(words) != 2 {
    115 		return "", Prerror{"Wrong arguments number"}
    116 	}
    117 
    118 	result := ""
    119 	for i := 0; i < reg_num; i++ {
    120 		if words[0] == strings.ToLower(Get_register_name(i)) {
    121 			result += zeros_prefix(int(arch.R), get_binary(i))
    122 			break
    123 		}
    124 	}
    125 
    126 	if result == "" {
    127 		return "", Prerror{"Unknown register name " + words[0]}
    128 	}
    129 
    130 	partial := ""
    131 	for i := 0; i < reg_num; i++ {
    132 		if words[1] == strings.ToLower(Get_register_name(i)) {
    133 			partial += zeros_prefix(int(arch.R), get_binary(i))
    134 			break
    135 		}
    136 	}
    137 
    138 	if partial == "" {
    139 		return "", Prerror{"Unknown register name " + words[1]}
    140 	}
    141 
    142 	result += partial
    143 
    144 	for i := opBits + 2*int(arch.R); i < rom_word; i++ {
    145 		result += "0"
    146 	}
    147 
    148 	return result, nil
    149 }

    228 		rg.Requirement(bmreqs.ReqRequest{Node: node + "/opcodes:cmpr", T: bmreqs.ObjectSet, Name: "destregs", Value: regDst, Op: bmreqs.OpAdd})
    229 		rg.Requirement(bmreqs.ReqRequest{Node: node + "/opcodes:cmpr", T: bmreqs.ObjectSet, Name: "sourceregs", Value: regSrc, Op: bmreqs.OpAdd})
