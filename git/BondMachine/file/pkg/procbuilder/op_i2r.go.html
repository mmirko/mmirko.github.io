    142 func (op I2r) Assembler(arch *Arch, words []string) (string, error) {
    143 	opbits := arch.Opcodes_bits()
    144 	inpbits := arch.Inputs_bits()
    145 	rom_word := arch.Max_word()
    146 
    147 	reg_num := 2
    148 	reg_num = reg_num << (arch.R - 1)
    149 
    150 	if len(words) != 2 {
    151 		return "", Prerror{"Wrong arguments number"}
    152 	}
    153 
    154 	result := ""
    155 	for i := 0; i < reg_num; i++ {
    156 		if words[0] == strings.ToLower(Get_register_name(i)) {
    157 			result += zeros_prefix(int(arch.R), get_binary(i))
    158 			break
    159 		}
    160 	}
    161 
    162 	if result == "" {
    163 		return "", Prerror{"Unknown register name " + words[0]}
    164 	}
    165 
    166 	if partial, err := Process_input(words[1], int(arch.N)); err == nil {
    167 		result += zeros_prefix(inpbits, partial)
    168 	} else {
    169 		return "", Prerror{err.Error()}
    170 	}
    171 
    172 	for i := opbits + int(arch.R) + inpbits; i < rom_word; i++ {
    173 		result += "0"
    174 	}
    175 
    176 	return result, nil
    177 }

    261 func (Op I2r) Op_instruction_verilog_extra_block(arch *Arch, flavor string, level uint8, blockname string, objects []string) string {
    262 	// TODO: ADDI and I2R are to be fixed in order to coexist
    263 	opbits := arch.Opcodes_bits()
    264 	inbits := arch.Inputs_bits()
    265 	rom_word := arch.Max_word()
    266 
    267 	result := ""
    268 
    269 	pref := strings.Repeat("\t", int(level))
    270 
    271 	switch blockname {
    272 	case "input_data_received":
    273 		result += pref + "I2R: begin\n"
    274 		if inbits == 1 {
    275 			result += pref + "\tcase (current_instruction[" + strconv.Itoa(rom_word-opbits-int(arch.R)-1) + "])\n"
    276 		} else {
    277 			result += pref + "\tcase (current_instruction[" + strconv.Itoa(rom_word-opbits-int(arch.R)-1) + ":" + strconv.Itoa(rom_word-opbits-int(arch.R)-int(inbits)) + "])\n"
    278 		}
    279 
    280 		result += pref + "\t" + strings.ToUpper(objects[0]) + " : begin\n"
    281 		result += pref + "\t\tif (" + strings.ToLower(objects[0]) + "_valid)\n"
    282 		result += pref + "\t\tbegin\n"
    283 		result += pref + "\t\t\t" + strings.ToLower(objects[0]) + "_recv <= #1 1'b1;\n"
    284 		result += pref + "\t\tend\n"
    285 		result += pref + "\tend\n"
    286 
    287 		result += pref + "\tdefault: begin\n"
    288 		result += pref + "\t\tif (!" + strings.ToLower(objects[0]) + "_valid)\n"
    289 		result += pref + "\t\tbegin\n"
    290 		result += pref + "\t\t\t" + strings.ToLower(objects[0]) + "_recv <= #1 1'b0;\n"
    291 		result += pref + "\t\tend\n"
    292 		result += pref + "\tend\n"
    293 
    294 		result += pref + "\tendcase\n"
    295 
    296 		result += pref + "end\n"
    297 	default:

    308 func (Op I2r) HLAssemblerNormalize(arch *Arch, rg *bmreqs.ReqRoot, node string, line *bmline.BasmLine) (*bmline.BasmLine, error) {
    309 	switch line.Operation.GetValue() {
    310 	case "i2r":
    311 		regNeed := line.Elements[0].GetValue()
    312 		inNeed := line.Elements[1].GetValue()
    313 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regNeed, Op: bmreqs.OpAdd})
    314 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "inputs", Value: inNeed, Op: bmreqs.OpAdd})
    315 		return line, nil
    316 	case "mov":
    317 		regVal := line.Elements[0].GetValue()
    318 		inVal := line.Elements[1].GetValue()
    319 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "registers", Value: regVal, Op: bmreqs.OpAdd})
    320 		rg.Requirement(bmreqs.ReqRequest{Node: node, T: bmreqs.ObjectSet, Name: "inputs", Value: inVal, Op: bmreqs.OpAdd})
    321 		if regVal != "" && inVal != "" {
    322 			newLine := new(bmline.BasmLine)
    323 			newOp := new(bmline.BasmElement)
    324 			newOp.SetValue("i2r")
    325 			newLine.Operation = newOp
    326 			newArgs := make([]*bmline.BasmElement, 2)
    327 			newArg0 := new(bmline.BasmElement)
    328 			newArg0.BasmMeta = newArg0.SetMeta("type", "reg")
    329 			newArg0.SetValue(regVal)
    330 			newArgs[0] = newArg0
    331 			newArg1 := new(bmline.BasmElement)
    332 			newArg1.SetValue(inVal)
    333 			newArg1.BasmMeta = newArg1.SetMeta("type", "input")
    334 			newArgs[1] = newArg1
    335 			newLine.Elements = newArgs
    336 			return newLine, nil
    337 		}
    338 	}
    339 	return nil, errors.New("HL Assembly normalize failed")
    340 }
