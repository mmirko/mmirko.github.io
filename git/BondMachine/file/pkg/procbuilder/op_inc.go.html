     56 func (op Inc) Op_instruction_verilog_state_machine(conf *Config, arch *Arch, rg *bmreqs.ReqRoot, flavor string) string {
     57 	// "reference": {"support_hdl":"ok"}
     58 	// "reference": {"support_mt":"ok"}
     59 	romWord := arch.Max_word()
     60 	opBits := arch.Opcodes_bits()
     61 	tabsNum := 5
     62 	regNum := 1 << arch.R
     63 
     64 	result := ""
     65 	result += tabs(tabsNum) + "INC: begin\n"
     66 	if th := ThreadInstructionStart(conf, arch, tabsNum+1); th != "" {
     67 		result += th
     68 		tabsNum++
     69 	}
     70 	if arch.R == 1 {
     71 		result += tabs(tabsNum+1) + "case (current_instruction[" + strconv.Itoa(romWord-opBits-1) + "])\n"
     72 	} else {
     73 		result += tabs(tabsNum+1) + "case (current_instruction[" + strconv.Itoa(romWord-opBits-1) + ":" + strconv.Itoa(romWord-opBits-int(arch.R)) + "])\n"
     74 	}
     75 	for i := 0; i < regNum; i++ {
     76 
     77 		if IsHwOptimizationSet(conf.HwOptimizations, HwOptimizations(OnlyDestRegs)) {
     78 			// "reference": {"hwoptimizations": "OnlyDestRegs"}
     79 			cp := arch.Tag
     80 			req := rg.Requirement(bmreqs.ReqRequest{Node: "/bm:cps/id:" + cp + "/opcodes:inc", T: bmreqs.ObjectSet, Name: "destregs", Value: Get_register_name(i), Op: bmreqs.OpCheck})
     81 			if req.Value == "false" {
     82 				continue
     83 			}
     84 		}
     85 		result += tabs(tabsNum+1) + strings.ToUpper(Get_register_name(i)) + " : begin\n"
     86 		result += tabs(tabsNum+2) + "_" + strings.ToLower(Get_register_name(i)) + " <= #1 _" + strings.ToLower(Get_register_name(i)) + " + 1'b1;\n"
     87 		result += tabs(tabsNum+2) + "$display(\"INC " + strings.ToUpper(Get_register_name(i)) + "\");\n"
     88 		result += tabs(tabsNum+1) + "end\n"
     89 	}
     90 	result += tabs(tabsNum+1) + "endcase\n"
     91 	result += NextInstruction(conf, arch, tabsNum+1, "_pc + 1'b1")
     92 	if th := ThreadInstructionEnd(conf, arch, tabsNum); th != "" {
     93 		result += th
     94 		tabsNum--
     95 	}
     96 	result += tabs(tabsNum) + "end\n"
     97 	return result
     98 }

    104 func (op Inc) Assembler(arch *Arch, words []string) (string, error) {
    105 	// "reference": {"support_asm": "ok"}
    106 	opBits := arch.Opcodes_bits()
    107 	romWord := arch.Max_word()
    108 
    109 	regNum := 1 << arch.R
    110 
    111 	if len(words) != 1 {
    112 		return "", errors.New("wrong arguments number")
    113 	}
    114 
    115 	result := ""
    116 	for i := 0; i < regNum; i++ {
    117 		if words[0] == strings.ToLower(Get_register_name(i)) {
    118 			result += zeros_prefix(int(arch.R), get_binary(i))
    119 			break
    120 		}
    121 	}
    122 
    123 	if result == "" {
    124 		return "", errors.New("unknown register name " + words[0])
    125 	}
    126 
    127 	for i := opBits + int(arch.R); i < romWord; i++ {
    128 		result += "0"
    129 	}
    130 
    131 	return result, nil
    132 }

    141 func (op Inc) Simulate(vm *VM, instr string) error {
    142 	// "reference": {"support_gosim":"ok"}
    143 	reg_bits := vm.Mach.R
    144 	reg := get_id(instr[:reg_bits])
    145 	switch vm.Mach.Rsize {
    146 	case 8:
    147 		vm.Registers[reg] = vm.Registers[reg].(uint8) + uint8(1)
    148 	case 16:
    149 		vm.Registers[reg] = vm.Registers[reg].(uint16) + uint16(1)
    150 	case 32:
    151 		vm.Registers[reg] = vm.Registers[reg].(uint32) + uint32(1)
    152 	case 64:
    153 		vm.Registers[reg] = vm.Registers[reg].(uint64) + uint64(1)
    154 	default:
    155 		return errors.New("go simulation only works on 8,16,32 or 64 bits registers")
    156 	}
    157 	vm.Pc = vm.Pc + 1
    158 	return nil
    159 }
