    508 		result += "	reg [2:0] " + threadStack + "SM;\n"
    509 		result += "	localparam	CTXSEND = 3'b000,\n" // Context switch send the current thread to the stack
    510 		result += "			CTXWSEND = 3'b001,\n"         // Wait for the stack to send the current thread
    511 		result += "			CTXREQNEW = 3'b010,\n"        // Request a new thread from the stack
    512 		result += "			CTXNEW = 3'b011,\n"           // New thread from the stack received
    513 		result += "			CTXCHKNICE = 3'b100,\n"       // Check if the nice is 0
    514 		result += "			CTXEXE = 3'b101;\n"           // Execute the thread
    515 
    516 		result += "\n"
    517 		result += "\treg [" + strconv.Itoa(dataSize-1) + ":0] " + threadStack + "senderData;\n"
    518 		result += "\treg " + threadStack + "senderWrite;\n"
    519 		result += "\twire " + threadStack + "senderAck;\n"
    520 		result += "\n"
    521 		result += "\twire [" + strconv.Itoa(dataSize-1) + ":0] " + threadStack + "receiverData;\n"
    522 		result += "\treg " + threadStack + "receiverRead;\n"
    523 		result += "\twire " + threadStack + "receiverAck;\n"
    524 		result += "\n"
    525 		result += "\twire " + threadStack + "empty;\n"
    526 		result += "\twire " + threadStack + "full;\n"
    527 		result += "\n"
    528 		result += "\t" + threadStack + " " + threadStack + "_inst (\n"
    529 		result += "\t\t.clk(clock_signal),\n"
    530 		result += "\t\t.reset(reset_signal),\n"
    531 		result += "\t\t.senderData(" + threadStack + "senderData),\n"
    532 		result += "\t\t.senderWrite(" + threadStack + "senderWrite),\n"
    533 		result += "\t\t.senderAck(" + threadStack + "senderAck),\n"
    534 		result += "\t\t.receiverData(" + threadStack + "receiverData),\n"
    535 		result += "\t\t.receiverRead(" + threadStack + "receiverRead),\n"
    536 		result += "\t\t.receiverAck(" + threadStack + "receiverAck),\n"
    537 		result += "\t\t.empty(" + threadStack + "empty),\n"
    538 		result += "\t\t.full(" + threadStack + "full)\n"
    539 		result += "\t);\n"
    540 		result += "\n"
    541 		result += "initial begin\n"
    542 		result += "	" + threadStack + "SM <= CTXEXE;\n"
    543 		result += "\tnice <= 8'b00000100;\n"
    544 		result += "\tnicecount <= 8'b00000100;\n"
    545 		result += "end\n"
    546 
    547 		s := bmstack.CreateBasicStack()
    548 		s.ModuleName = threadStack
    549 		s.DataSize = dataSize
    550 		s.Depth = proc.Threaded
    551 		s.MemType = "FIFO"
    552 		s.Senders = []string{"sender"}
    553 		s.Receivers = []string{"receiver"}
    554 
    555 		f, _ := os.Create(threadStack + "stack.v")
    556 		r, _ := s.WriteHDL()
    557 		f.WriteString(r)
    558 		f.Close()
    559 	}
    560 
    561 	// modes handling
    562 	switch arch.Modes[0] {
    563 	case "ha":
    564 		result += "\n"
    565 		result += "	wire [" + strconv.Itoa(int(rom_word)-1) + ":0] current_instruction;\n"
    566 		result += "	assign current_instruction=rom_value;\n"
    567 		result += "\n"
    568 	case "hy":
    569 		result += "\n"
    570 		result += "	wire [" + strconv.Itoa(int(rom_word)-1) + ":0] current_instruction;\n"
    571 		result += "	reg [" + strconv.Itoa(int(rom_word)-1) + ":0] ram_instruction;\n"
    572 		result += "	reg exec_mode; // 0 = harvard , 1=VN\n"
    573 		result += "	reg [1:0] vn_state;\n"
    574 		result += "	localparam FETCH=2'b00, WAIT=2'b10, EXECUTE=2'b01;\n"
    575 		result += "	assign current_instruction= (exec_mode==1'b0) ? rom_value : ram_instruction;\n"
    576 		result += "\n"
    577 		if !arch.HasOp("r2m") && !arch.HasOp("m2r") && !arch.HasOp("r2mri") && !arch.HasOp("m2rri") {
    578 			result += "	assign ram_addr = _pc;\n"
    579 			result += "	assign ram_en = 1'b1;\n"
    580 		}
    581 	case "vn":
    582 		result += "\n"
    583 		result += "	wire [" + strconv.Itoa(int(rom_word)-1) + ":0] current_instruction;\n"
    584 		result += "	reg [" + strconv.Itoa(int(rom_word)-1) + ":0] ram_instruction;\n"
    585 		result += "	reg [1:0] vn_state;\n"
    586 		result += "	localparam FETCH=2'b00, WAIT=2'b10, EXECUTE=2'b01;\n"
    587 		result += "	assign current_instruction=ram_instruction;\n"
    588 		result += "\n"
    589 		if !arch.HasOp("r2m") && !arch.HasOp("m2r") && !arch.HasOp("r2mri") && !arch.HasOp("m2rri") {
    590 			result += "	assign ram_addr = _pc;\n"
    591 			result += "	assign ram_en = 1'b1;\n"
    592 		}
    593 	}
    594 
    595 	for _, op := range proc.Op {
    596 		if conf.Commented_verilog {
    597 			result += "\n// Start of the component \"header\" for the opcode " + op.Op_get_name() + "\n\n"
    598 		}
    599 		result += op.OpInstructionVerilogHeader(conf, arch, flavor, processor_module_name)
    600 	}
    601 
    602 	result += "\n"
    603 	result += "	always @(posedge clock_signal, posedge reset_signal)\n"
    604 	result += "	begin\n"
    605 	result += "		if(reset_signal)\n"
    606 	result += "		begin\n"
    607 	switch arch.Modes[0] {
    608 	case "ha":
    609 		result += "			_pc <= #1 " + strconv.Itoa(int(arch.O)) + "'h0;\n"
    610 	case "hy":
    611 		if arch.L > arch.O {
    612 			result += "			_pc <= #1 " + strconv.Itoa(int(arch.L)) + "'h0;\n"
    613 		} else {
    614 			result += "			_pc <= #1 " + strconv.Itoa(int(arch.O)) + "'h0;\n"
    615 		}
    616 	case "vn":
    617 		result += "			_pc <= #1 " + strconv.Itoa(int(arch.L)) + "'h0;\n"
    618 	}
    619 
    620 	for i := 0; i < reg_num; i++ {
    621 		result += "			_" + strings.ToLower(Get_register_name(i)) + " <= #1 " + strconv.Itoa(int(arch.Rsize)) + "'h0;\n"
    622 	}
    623 
    624 	for _, op := range proc.Op {
    625 		if conf.Commented_verilog {
    626 			result += "\n// Start of the component \"reset\" for the opcode " + op.Op_get_name() + "\n\n"
    627 		}
    628 		result += op.Op_instruction_verilog_reset(arch, flavor)
    629 	}
    630 
    631 	result += "		end\n"
    632 	result += "		else begin\n"
    633 
    634 	switch arch.Modes[0] {
    635 	case "ha":
    636 		result += "			// ha placeholder\n"
    637 	case "hy":
    638 		result += "			if (exec_mode == 1'b1 && vn_state == FETCH) begin\n"
    639 		result += "				vn_state <= WAIT;\n"
    640 		result += "			end\n"
    641 		result += "			else if (exec_mode == 1'b1 && vn_state == WAIT) begin\n"
    642 		result += "				vn_state <= EXECUTE;\n"
    643 		result += "				ram_instruction <= ram_dout;\n"
    644 		result += "			end\n"
    645 		result += "			else begin\n"
    646 	case "vn":
    647 		result += "			case (vn_state)\n"
    648 		result += "			FETCH: begin\n"
    649 		result += "				vn_state <= WAIT;\n"
    650 		result += "			end\n"
    651 		result += "			WAIT: begin\n"
    652 		result += "				vn_state <= EXECUTE;\n"
    653 		result += "				ram_instruction <= ram_dout;\n"
    654 		result += "			end\n"
    655 		result += "			EXECUTE: begin\n"

    658 	result += "			$display(\"Program Counter:%d\", _pc);\n"
    659 	result += "			$display(\"Instruction:%b\", rom_value);\n"
    660 
    661 	format := ""
    662 	list := ""
    663 	for i := 0; i < reg_num; i++ {
    664 		format = format + strings.ToLower(Get_register_name(i)) + ":%b "
    665 		list = list + ", _" + strings.ToLower(Get_register_name(i))
    666 	}
    667 
    668 	result += "			$display(\"Registers " + format + "\"" + list + ");\n"
    669 
    670 	for _, op := range proc.Op {
    671 		if conf.Commented_verilog {
    672 			result += "\n// Start of the component \"internal state\" for the opcode " + op.Op_get_name() + "\n\n"
    673 		}
    674 		result += op.Op_instruction_verilog_internal_state(arch, flavor)
    675 	}
    676

    678 	//	result += "			else\n"
    679 	//	result += "				begin\n"

    683 			result += "\n// Start of the component \"default state\" for the opcode " + op.Op_get_name() + "\n\n"
    684 		}
    685 		result += op.Op_instruction_verilog_default_state(arch, flavor)
    686 	}

    688 	if opbits == 1 {
    689 		result += "				case(current_instruction[" + strconv.Itoa(rom_word-1) + "])\n"
    690 	} else {
    691 		result += "				case(current_instruction[" + strconv.Itoa(rom_word-1) + ":" + strconv.Itoa(rom_word-opbits) + "])\n"
    692 	}
    693 
    694 	for _, op := range proc.Op {
    695 		if conf.Commented_verilog {
    696 			result += "\n// Start of the component of the \"state machine\" for the opcode " + op.Op_get_name() + "\n\n"

    699 	}
    700 
    701 	result += "					default : begin\n"
    702 	result += "						$display(\"Unknown Opcode\");\n"
    703 	result += NextInstruction(conf, arch, 6, "_pc + 1'b1")
    704 	result += "					end\n"
    705 
    706 	result += "				endcase\n"
    707

    709 	//	result += "			end\n"
    710 
    711 	switch arch.Modes[0] {
    712 	case "ha":
    713 		result += "			// ha placeholder\n"
    714 	case "hy":
    715 		result += "			end\n"
    716 	case "vn":
    717 		result += "				end\n"
    718 		result += "			endcase\n"
    719 	}

    730 		result += "	assign " + Get_input_name(i) + "_received = " + Get_input_name(i) + "_recv;\n"
    731 	}

    733 	for i := 0; i < int(proc.M); i++ {
    734 		result += "	assign " + Get_output_name(i) + " = _aux" + Get_output_name(i) + ";\n"
    735 		result += "	assign " + Get_output_name(i) + "_valid = " + Get_output_name(i) + "_val;\n"
    736 	}
    737 
    738 	result += "endmodule\n"

    744 		for i, module := range modlist {

    746 				result += modcode[i]
    747 				doneextramod[module] = true
    748 			}
    749 		}
    750 
    751 		files, filesCode := op.ExtraFiles(arch)
    752 		for i, file := range files {
