    148 func (op Addf16) Assembler(arch *Arch, words []string) (string, error) {
    149 	opbits := arch.Opcodes_bits()
    150 	rom_word := arch.Max_word()
    151 
    152 	reg_num := 1 << arch.R
    153 
    154 	if len(words) != 2 {
    155 		return "", Prerror{"Wrong arguments number"}
    156 	}
    157 
    158 	result := ""
    159 	for i := 0; i < reg_num; i++ {
    160 		if words[0] == strings.ToLower(Get_register_name(i)) {
    161 			result += zeros_prefix(int(arch.R), get_binary(i))
    162 			break
    163 		}
    164 	}
    165 
    166 	if result == "" {
    167 		return "", Prerror{"Unknown register name " + words[0]}
    168 	}
    169 
    170 	partial := ""
    171 	for i := 0; i < reg_num; i++ {
    172 		if words[1] == strings.ToLower(Get_register_name(i)) {
    173 			partial += zeros_prefix(int(arch.R), get_binary(i))
    174 			break
    175 		}
    176 	}
    177 
    178 	if partial == "" {
    179 		return "", Prerror{"Unknown register name " + words[1]}
    180 	}
    181 
    182 	result += partial
    183 
    184 	for i := opbits + 2*int(arch.R); i < rom_word; i++ {
    185 		result += "0"
    186 	}
    187 
    188 	return result, nil
    189 }

    199 func (op Addf16) Simulate(vm *VM, instr string) error {
    200 	reg_bits := vm.Mach.R
    201 	regDest := get_id(instr[:reg_bits])
    202 	regSrc := get_id(instr[reg_bits : reg_bits*2])
    203 	switch vm.Mach.Rsize {
    204 	case 16:
    205 		var floatDest float16.Float16
    206 		var floatSrc float16.Float16
    207 		if v, ok := vm.Registers[regDest].(uint16); ok {
    208 			floatDest = float16.Frombits(v)
    209 		} else {
    210 			floatDest = float16.Float16(0)
    211 		}
    212 		if v, ok := vm.Registers[regSrc].(uint16); ok {
    213 			floatSrc = float16.Frombits(v)
    214 		} else {
    215 			floatSrc = float16.Float16(0)
    216 		}
    217 		operation := floatDest.Float32() + floatSrc.Float32()
    218 		vm.Registers[regDest] = float16.Fromfloat32(operation).Bits()
    219 	default:
    220 		return errors.New("invalid register size, for float registers has to be 16 bits")
    221 	}
    222 	vm.Pc = vm.Pc + 1
    223 	return nil
    224 }
