     90 func (mach *Machine) MelCopy() mel.Me3li {
     91 	newmach := new(Machine)

     94 	for i, op := range mach.Arch.Op {
     95 		newmach.Arch.Op[i] = op
     96 	}
     97 	newmach.Program.Slocs = make([]string, len(mach.Program.Slocs))
     98 	for i, sloc := range mach.Program.Slocs {
     99 		newmach.Program.Slocs[i] = sloc
    100 	}
