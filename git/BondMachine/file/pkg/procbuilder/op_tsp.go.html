     37 		} else {
     38 			result = "j [" + strconv.Itoa(int(arch.R)) + "(Reg)] [" + strconv.Itoa(int(arch.L)) + "(Location)] [ 8 (nice)]	// Start a thread at a specific location [" + strconv.Itoa(opBits+int(arch.L)+8) + "]\n"
     39 		}
     40 	}
     41 	return result

     77 func (op Tsp) Op_instruction_verilog_state_machine(conf *Config, arch *Arch, rg *bmreqs.ReqRoot, flavor string) string {
     78 
     79 	romWord := arch.Max_word()
     80 	opBits := arch.Opcodes_bits()
     81 	tabsNum := 5
     82 	regNum := 1 << arch.R
     83 	locationBits := arch.O
     84 
     85 	threadStack := "threadStack" + strconv.Itoa(arch.Threaded)
     86 	sm := threadStack + "SM"
     87 
     88 	switch arch.Modes[0] {
     89 	case "ha":
     90 		locationBits = arch.O
     91 	case "vn":
     92 		locationBits = arch.L
     93 	case "hy":
     94 		if arch.O > arch.L {
     95 			locationBits = arch.O
     96 		} else {
     97 			locationBits = arch.L
     98 		}
     99 	}
    100 
    101 	result := ""
    102 	result += tabs(tabsNum) + "TSP: begin\n"
    103 	if arch.R == 1 {
    104 		result += tabs(tabsNum+1) + "case (current_instruction[" + strconv.Itoa(romWord-opBits-1) + "])\n"
    105 	} else {
    106 		result += tabs(tabsNum+1) + "case (current_instruction[" + strconv.Itoa(romWord-opBits-1) + ":" + strconv.Itoa(romWord-opBits-int(arch.R)) + "])\n"
    107 	}
    108 	for i := 0; i < regNum; i++ {
    109 
    110 		if IsHwOptimizationSet(conf.HwOptimizations, HwOptimizations(OnlyDestRegs)) {
    111 			cp := arch.Tag
    112 			req := rg.Requirement(bmreqs.ReqRequest{Node: "/bm:cps/id:" + cp + "/opcodes:tsp", T: bmreqs.ObjectSet, Name: "destregs", Value: Get_register_name(i), Op: bmreqs.OpCheck})
    113 			if req.Value == "false" {
    114 				continue
    115 			}
    116 		}
    117 
    118 		result += tabs(tabsNum+2) + strings.ToUpper(Get_register_name(i)) + " : begin\n"
    119 
    120 		result += tabs(tabsNum+3) + "case (" + sm + ")\n"
    121 		result += tabs(tabsNum+3) + "CTXEXE: begin\n"
    122 		result += tabs(tabsNum+4) + sm + " <= CTXSEND;\n"
    123 		result += tabs(tabsNum+4) + "provpc <= current_instruction[" + strconv.Itoa(romWord-opBits-1-int(arch.R)) + ":" + strconv.Itoa(romWord-opBits-int(locationBits)-int(arch.R)) + "];\n"
    124 
    125 		result += tabs(tabsNum+3) + "end\n"
    126 		result += tabs(tabsNum+3) + "CTXSEND: begin\n"
    127 		result += tabs(tabsNum+5) + "if (!" + threadStack + "senderAck) begin\n"
    128 		regList := ""
    129 		for j := 0; j < regNum; j++ {
    130 			regList += "_" + strings.ToLower(Get_register_name(j)) + ", "
    131 		}
    132 		result += tabs(tabsNum+6) + threadStack + "senderData <= {ThreadID, 1'b0, provpc, " + regList + "current_instruction[" + strconv.Itoa(romWord-opBits-1-int(arch.R)-int(locationBits)) + ":" + strconv.Itoa(romWord-opBits-int(locationBits)-int(arch.R)-8) + "]};\n"
    133 		result += tabs(tabsNum+6) + threadStack + "senderWrite <= 1'b1;\n"
    134 		result += tabs(tabsNum+6) + sm + " <= CTXWSEND;\n"
    135 		result += tabs(tabsNum+5) + "end\n"
    136 		result += tabs(tabsNum+4) + "end\n"
    137 		result += tabs(tabsNum+3) + "CTXWSEND: begin\n"
    138 		result += tabs(tabsNum+4) + "if (" + threadStack + "senderAck) begin\n"
    139 		result += tabs(tabsNum+5) + threadStack + "senderWrite <= 1'b0;\n"
    140 		result += tabs(tabsNum+5) + sm + " <= CTXEXE;\n"
    141 		result += NextInstruction(conf, arch, tabsNum+5, "_pc + 1'b1")
    142 
    143 		result += tabs(tabsNum+2) + "\t\tend\n"
    144 		result += tabs(tabsNum+2) + "\tend\n"
    145 		result += tabs(tabsNum+2) + "\tendcase\n"
    146 
    147 		result += tabs(tabsNum+2) + "end\n"
    148 	}
    149 	result += tabs(tabsNum+1) + "endcase\n"
    150 	result += tabs(tabsNum) + "end\n"
    151 	return result
    152 }

    158 func (op Tsp) Assembler(arch *Arch, words []string) (string, error) {
    159 	reg_num := 1 << arch.R
    160 
    161 	locationBits := arch.O
    162 
    163 	switch arch.Modes[0] {
    164 	case "ha":
    165 		locationBits = arch.O
    166 	case "vn":
    167 		locationBits = arch.L
    168 	case "hy":
    169 		if arch.O > arch.L {
    170 			locationBits = arch.O
    171 		} else {
    172 			locationBits = arch.L
    173 		}
    174 	}
    175 
    176 	if len(words) != 3 {
    177 		return "", Prerror{"Wrong arguments number"}
    178 	}
    179 
    180 	result := ""
    181 	for i := 0; i < reg_num; i++ {
    182 		if words[0] == strings.ToLower(Get_register_name(i)) {
    183 			result += zeros_prefix(int(arch.R), get_binary(i))
    184 			break
    185 		}
    186 	}
    187 
    188 	if partial, err := Process_number(words[1]); err == nil {
    189 		result += zeros_prefix(int(locationBits), partial)
    190 	} else {
    191 		return "", Prerror{err.Error()}
    192 	}
    193 
    194 	if partial, err := Process_number(words[2]); err == nil {
    195 		result += zeros_prefix(8, partial)
    196 	} else {
    197 		return "", Prerror{err.Error()}
    198 	}
    199 
    200 	return result, nil
    201 }
