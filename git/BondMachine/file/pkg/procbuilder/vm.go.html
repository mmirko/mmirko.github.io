    159 func (vm *VM) Step(psc *Sim_config) (string, error) {
    160 	result := ""
    161 
    162 	if psc != nil {
    163 		if psc.Show_pc {
    164 			result += "\t\tPC: " + strconv.Itoa(int(vm.Pc)) + "\n"
    165 		}
    166 	}
    167 
    168 	//	reg_num := 1 << vm.Mach.R
    169 	num_instr := len(vm.Mach.Program.Slocs)
    170 	opBits := vm.Mach.Opcodes_bits()
    171 
    172 	if int(vm.Pc) > num_instr {
    173 		return "", Prerror{"Program counter outside limits"}
    174 	}
    175 
    176 	if int(vm.Pc) == num_instr {
    177 		// Halt computation
    178 		// vm.Pc = 0
    179 	} else {
    180 		instr := vm.Mach.Program.Slocs[vm.Pc]
    181 
    182 		if psc != nil {
    183 			if psc.Show_instruction {
    184 				result += "\t\tInstr: " + instr + "\n"
    185 			}
    186 		}
    187 
    188 		if opcode_id, err := vm.Mach.Conproc.Decode_opcode(instr); err == nil {
    189 			op := vm.Mach.Arch.Conproc.Op[opcode_id]
    190 
    191 			if psc != nil {
    192 				if psc.Show_disasm {
    193 					curline := "\t\tDisasm: " + op.Op_get_name() + " "
    194 					if disas, err := op.Disassembler(&vm.Mach.Arch, instr[opBits:]); err != nil {
    195 						return "", Prerror{"Disassembling falied"}
    196 					} else {
    197 						result += curline + disas + "\n"
    198 					}
    199 				}
    200 			}
    201 			if psc != nil {
    202 				if psc.Show_io_pre {
    203 					result += "\t\tPre-compute IO: " + vm.DumpIO() + "\n"
    204 				}
    205 				if psc.Show_regs_pre {
    206 					result += "\t\tPre-compute Regs: " + vm.DumpRegisters() + "\n"
    207 				}
    208 			}
    209 
    210 			if err := op.Simulate(vm, instr[opBits:]); err != nil {
    211 				return "", Prerror{"Simulation failed"}
    212 			}
    213 
    214 			if psc != nil {
    215 				if psc.Show_io_pre {
    216 					result += "\t\tPost-compute IO: " + vm.DumpIO() + "\n"
    217 				}
    218 				if psc.Show_regs_pre {
    219 					result += "\t\tPost-compute Regs: " + vm.DumpRegisters() + "\n"
    220 				}
    221 			}
    222 
    223 		} else {
    224 			return "", Prerror{"Unknown opcode"}
    225 		}
    226 	}
    227 
    228 	return result, nil
    229 }
