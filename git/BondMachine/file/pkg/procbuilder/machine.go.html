    238 func (mach *Machine) ConstraintCheck() (string, bool) {
    239 	result := ""
    240 	shared := make([]string, 0)
    241 	required := make([]string, 0)
    242 	forbidden := make([]string, 0)
    243 
    244 	for _, op := range mach.Op {
    245 		if present, modes := op.Required_modes(); present {
    246 			for _, mode := range modes {
    247 				checkpres := false
    248 				for _, curr := range required {
    249 					if curr == mode {
    250 						checkpres = true
    251 						break
    252 					}
    253 				}
    254 				if !checkpres {
    255 					required = append(required, mode)
    256 					result += "Added required mode: " + mode + "\n"
    257 				}
    258 			}
    259 		}
    260 		if present, modes := op.Forbidden_modes(); present {
    261 			for _, mode := range modes {
    262 				checkpres := false
    263 				for _, curr := range forbidden {
    264 					if curr == mode {
    265 						checkpres = true
    266 						break
    267 					}
    268 				}
    269 				if !checkpres {
    270 					forbidden = append(forbidden, mode)
    271 					result += "Added forbidden mode: " + mode + "\n"
    272 
    273 				}
    274 			}
    275 
    276 		}
    277 		if present, sos := op.Required_shared(); present {
    278 			for _, so := range sos {
    279 				checkpres := false
    280 				for _, curr := range shared {
    281 					if curr == so {
    282 						checkpres = true
    283 						break
    284 					}
    285 				}
    286 				if !checkpres {
    287 					shared = append(shared, so)
    288 					result += "Added required shared object: " + so + "\n"
    289 
    290 				}
    291 			}
    292 
    293 		}
    294 	}
    295 
    296 	// Checking required modes constraints
    297 	// TODO Finish
    298 	for _, mode := range required {
    299 		switch mode {
    300 		case "ha":
    301 		case "vn":
    302 		case "hy":
    303 		case "ramabs":
    304 		case "romabs":
    305 		case "ramind":
    306 			if mach.Rsize != mach.L {
    307 				return "Register size nd RAM depth mismatch", false
    308 			}
    309 		case "romind":
    310 			if mach.Rsize != mach.O {
    311 				return "Register size nd ROM depth mismatch", false
    312 			}
    313 		}
    314 	}
    315 
    316 	// Checking conflicting modes
    317 	for _, mode := range forbidden {
    318 		for _, req := range required {
    319 			if mode == req {
    320 				return "", false
    321 			}
    322 		}
    323 	}
    324 
    325 	// TODO Finish shared checks
    326 
    327 	return result, true
