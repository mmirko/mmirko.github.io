    189 func (op DynOpStack) Assembler(arch *Arch, words []string) (string, error) {
    190 	opBits := arch.Opcodes_bits()
    191 	romWord := arch.Max_word()
    192 
    193 	regNum := 1 << arch.R
    194 
    195 	if len(words) != 1 {
    196 		return "", Prerror{"Wrong arguments number"}
    197 	}
    198 
    199 	result := ""
    200 	for i := 0; i < regNum; i++ {
    201 		if words[0] == strings.ToLower(Get_register_name(i)) {
    202 			result += zeros_prefix(int(arch.R), get_binary(i))
    203 			break
    204 		}
    205 	}
    206 
    207 	if result == "" {
    208 		return "", Prerror{"Unknown register name " + words[0]}
    209 	}
    210 
    211 	for i := opBits + int(arch.R); i < romWord; i++ {
    212 		result += "0"
    213 	}
    214 
    215 	return result, nil
    216 }

    253 	suffix := strconv.Itoa(op.s) + op.sn
    254 	if arch.OnlyOne(op.Op_get_name(), []string{"pull" + suffix, "push" + suffix}) {
    255 
    256 		s := bmstack.CreateBasicStack()
    257 		s.ModuleName = "regstack" + arch.Tag + "_" + suffix
    258 		s.DataSize = int(arch.Rsize)
    259 		s.Depth = op.s
    260 		s.MemType = "LIFO"
    261 		s.Senders = []string{"sender"}
    262 		s.Receivers = []string{"receiver"}
    263 
    264 		r, _ := s.WriteHDL()
    265 
    266 		result := r
    267 		return []string{s.ModuleName}, []string{result}
    268 	}
    269 	return []string{}, []string{}
