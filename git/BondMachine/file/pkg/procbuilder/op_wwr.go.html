    195 func (op Wwr) Assembler(arch *Arch, words []string) (string, error) {
    196 	opbits := arch.Opcodes_bits()
    197 	chso := Channel{}
    198 	channum := arch.Shared_num(chso.Shr_get_name())
    199 	chanbits := arch.Shared_bits(chso.Shr_get_name())
    200 	shortname := chso.Shortname()
    201 	rom_word := arch.Max_word()
    202 
    203 	reg_num := 1 << arch.R
    204 
    205 	if len(words) != 2 {
    206 		return "", Prerror{"Wrong arguments number"}
    207 	}
    208 
    209 	result := ""
    210 	for i := 0; i < reg_num; i++ {
    211 		if words[0] == strings.ToLower(Get_register_name(i)) {
    212 			result += zeros_prefix(int(arch.R), get_binary(i))
    213 			break
    214 		}
    215 	}
    216 
    217 	if result == "" {
    218 		return "", Prerror{"Unknown register name " + words[0]}
    219 	}
    220 
    221 	if partial, err := Process_shared(shortname, words[1], channum); err == nil {
    222 		result += zeros_prefix(chanbits, partial)
    223 	} else {
    224 		return "", Prerror{err.Error()}
    225 	}
    226 
    227 	for i := opbits + int(arch.R) + chanbits; i < rom_word; i++ {
    228 		result += "0"
    229 	}
    230 
    231 	return result, nil
    232 }
