     40 func main() {
     41 	g := graphviz.New()
     42 
     43 	var data []byte
     44 
     45 	// Load a graph from a file
     46 	if *graphFile != "" {
     47 		var err error
     48 		data, err = os.ReadFile(*graphFile)
     49 		if err != nil {
     50 			panic(err)
     51 		}
     52 	} else {
     53 		log.Fatalln("No graph file specified")
     54 	}
     55 
     56 	graph, err := graphviz.ParseBytes(data)
     57 	if err != nil {
     58 		panic(err)
     59 	}
     60 
     61 	defer func() {
     62 		if err := graph.Close(); err != nil {
     63 			panic(err)
     64 		}
     65 		g.Close()
     66 	}()
     67 
     68 	// Create the config struct
     69 	config := new(bmgraph.Config)
     70 
     71 	// Load net from a JSON file the configuration
     72 	if *configFile != "" {
     73 		if netFileJSON, err := os.ReadFile(*configFile); err == nil {
     74 			if err := json.Unmarshal(netFileJSON, config); err != nil {
     75 				panic(err)
     76 			}
     77 		} else {
     78 			panic(err)
     79 		}
     80 	} else {
     81 		config.Debug = *debug
     82 		config.Verbose = *verbose
     83 		config.Params = make(map[string]string)
     84 	}
     85 
     86 	// Load or create the Info file
     87 	config.BMinfo = new(bminfo.BMinfo)
     88 
     89 	if *bmInfoFile != "" {
     90 		if bmInfoJSON, err := os.ReadFile(*bmInfoFile); err == nil {
     91 			if err := json.Unmarshal(bmInfoJSON, config.BMinfo); err != nil {
     92 				panic(err)
     93 			}
     94 		} else {
     95 			panic(err)
     96 		}
     97 	}
     98 
     99 	if config.Params == nil {
    100 		config.Params = make(map[string]string)
    101 	}
    102 	if config.List == nil {
    103 		config.List = make(map[string]string)
    104 	}
    105 	if config.Pruned == nil {
    106 		config.Pruned = make([]string, 0)
    107 	}
    108 
    109 	if *useNilNeuron {
    110 		config.UseNilNeuron = true
    111 	}
    112 
    113 	if *neuronLibPath != "" {
    114 		config.NeuronLibPath = *neuronLibPath
    115 	} else {
    116 		// panic("No neuron library path specified")
    117 	}
    118 
    119 	if *dataType != "" {
    120 		found := false
    121 		for _, tpy := range bmnumbers.AllTypes {
    122 			if tpy.GetName() == *dataType {
    123 				for opType, opName := range tpy.ShowInstructions() {
    124 					config.Params[opType] = opName
    125 				}
    126 				config.DataType = *dataType
    127 				config.TypePrefix = tpy.ShowPrefix()
    128 				config.Params["typeprefix"] = tpy.ShowPrefix()
    129 				found = true
    130 				break
    131 			}
    132 		}
    133 		if !found {
    134 			if created, err := bmnumbers.EventuallyCreateType(*dataType, nil); err == nil {
    135 				if created {
    136 					for _, tpy := range bmnumbers.AllTypes {
    137 						if tpy.GetName() == *dataType {
    138 							for opType, opName := range tpy.ShowInstructions() {
    139 								config.Params[opType] = opName
    140 							}
    141 							config.DataType = *dataType
    142 							config.TypePrefix = tpy.ShowPrefix()
    143 							config.Params["typeprefix"] = tpy.ShowPrefix()
    144 							break
    145 						}
    146 					}
    147 				} else {
    148 					panic("Unknown data type")
    149 				}
    150 
    151 			} else {
    152 				panic(err)
    153 			}
    154 		}
    155 	} else {
    156 		if config.DataType == "" {
    157 			panic("No data type specified")
    158 		}
    159 	}
    160 
    161 	bg := new(bmgraph.Graph)
    162 	bg.Graph = graph
    163 	bg.Config = config
    164 
    165 	bg.RegisterSize = *registerSize
    166 
    167 	switch *iomode {
    168 	case "async":
    169 		bg.IOMode = bmgraph.ASYNC
    170 	case "sync":
    171 		bg.IOMode = bmgraph.SYNC
    172 	default:
    173 		panic("Unknown IO mode")
    174 	}
    175 
    176 	if *saveBasm != "" {
    177 		if basmFile, err := bg.WriteBasm(); err == nil {
    178 			os.WriteFile(*saveBasm, []byte(basmFile), 0644)
    179 		} else {
    180 			panic(err)
    181 		}
    182 	}
    183 
    184 	if *bmInfoFile != "" {
    185 		// Write the info file
    186 		if bmInfoFileJSON, err := json.MarshalIndent(config.BMinfo, "", "  "); err == nil {
    187 			os.WriteFile(*bmInfoFile, bmInfoFileJSON, 0644)
    188 		} else {
    189 			panic(err)
    190 		}
    191 	}
    192 
    193 	// Remove the info file from the config prior to saving it
    194 	config.BMinfo = nil
    195 	if *configFile != "" {
    196 		// Write the eventually updated config file
    197 		if configFileJSON, err := json.MarshalIndent(config, "", "  "); err == nil {
    198 			os.WriteFile(*configFile, configFileJSON, 0644)
    199 		} else {
    200 			panic(err)
    201 		}
    202 	}
    203 
    204 }
