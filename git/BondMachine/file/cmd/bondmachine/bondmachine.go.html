    235 func lastAddr(n *net.IPNet) (net.IP, error) { // works when the n is a prefix, otherwise...
    236 	if n.IP.To4() == nil {
    237 		return net.IP{}, errors.New("does not support IPv6 addresses.")
    238 	}
    239 	ip := make(net.IP, len(n.IP.To4()))
    240 	binary.BigEndian.PutUint32(ip, binary.BigEndian.Uint32(n.IP.To4())|^binary.BigEndian.Uint32(net.IP(n.Mask).To4()))
    241 	return ip, nil
    242 }
