     76 func init() {
     77 	flag.Var(&swaps, "swap", "Swap qbits in the circuit")
     78 	flag.Parse()
     79 
     80 	// if *debug {
     81 	// 	fmt.Println("basm init")
     82 	// }
     83 
     84 	numOp := 0
     85 	if *buildFullHardwareHardcoded {
     86 		numOp++
     87 	}
     88 	if *buildMatrixSeqHardcoded {
     89 		numOp++
     90 	}
     91 	if *buildMatrixSeq || *buildMatrixSeqCompiled {
     92 		numOp++
     93 	}
     94 	if *softwareSimulation {
     95 		numOp++
     96 	}
     97 	if numOp == 0 {
     98 		log.Fatal("No build mode selected")
     99 	}
    100 	if numOp > 1 {
    101 		log.Fatal("Only one build mode can be selected among: build-full-hw-hardcoded, build-matrix-seq(_compiled), build-matrix-seq-hardcoded, software-simulation")
    102 	}
    103 
    104 	if *linearDataRange != "" {
    105 		if err := bmnumbers.LoadLinearDataRangesFromFile(*linearDataRange); err != nil {
    106 			log.Fatal(err)
    107 		}
    108 
    109 		var lqRanges *map[int]bmnumbers.LinearDataRange
    110 		for _, t := range bmnumbers.AllDynamicalTypes {
    111 			if t.GetName() == "dyn_linear_quantizer" {
    112 				lqRanges = t.(bmnumbers.DynLinearQuantizer).Ranges
    113 			}
    114 		}
    115 
    116 		for i, t := range procbuilder.AllDynamicalInstructions {
    117 			if t.GetName() == "dyn_linear_quantizer" {
    118 				dynIst := t.(procbuilder.DynLinearQuantizer)
    119 				dynIst.Ranges = lqRanges
    120 				procbuilder.AllDynamicalInstructions[i] = dynIst
    121 			}
    122 		}
    123 	}
    124 }
